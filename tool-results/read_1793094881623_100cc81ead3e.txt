    22→export async function GET(req: Request, { params }: { params: Promise<{ symbol: string }> }) {
    23→  const start = Date.now();
    24→  const { symbol } = await params;
    25→  const sym = symbol.toUpperCase();
    26→  const stock = MOCK_STOCKS.find((s) => s.symbol === sym);
    27→
    28→  let stockName = stock?.name || sym;
    29→  let currentPrice = stock?.price || 0;
    30→
    31→  if (!stock) {
    32→    try {
    33→      const ohlcvText = await fs.readFile(
    34→        path.join(process.cwd(), "data", "ohlcv_all", `${sym}.csv`),
    35→        "utf-8"
    36→      );
    37→      const lines = ohlcvText.trim().split("\n");
    38→      if (lines.length > 1) {
    39→        const lastLine = lines[lines.length - 1].split(",");
    40→        currentPrice = parseFloat(lastLine[4]) || 0;
    41→      }
    42→    } catch {
    43→      try {
    44→        const finText = await fs.readFile(
    45→          path.join(process.cwd(), "data", "financials", `${sym}.json`),
    46→          "utf-8"
    47→        );
    48→        const finData = JSON.parse(finText);
    49→        stockName = finData.company_name || finData.symbol || sym;
    50→      } catch {}
    51→    }
    52→  }
    53→
    54→  const url = new URL(req.url);
    55→  const refresh = url.searchParams.get("refresh") === "1";
    56→  if (refresh) await cacheDelete(CK(sym));
    57→  const cached = refresh ? null : await cacheGetJSON<any>(CK(sym));
    58→  if (cached) return NextResponse.json({ ...cached, cached: true, latencyMs: Date.now() - start });
    59→
    60→  let multi: any = null;
    61→  try {
    62→    multi = JSON.parse(
    63→      await fs.readFile(path.join(process.cwd(), "data", "ml_predictions_multi.json"), "utf-8")
    64→    );
    65→  } catch {
    66→    return NextResponse.json(
    67→      { error: "Run: python3 scripts/train_all_with_backtesting.py" },
    68→      { status: 503 }
    69→    );
    70→  }
    71→
    72→  // Fallback: if we still don't have a currentPrice, pull from the predictions file
    73→  if (!currentPrice) {
    74→    const clPred = multi?.cl_ensemble_predictions?.[sym];
    75→    if (clPred && typeof clPred.currentPrice === "number" && clPred.currentPrice > 0) {
    76→      currentPrice = clPred.currentPrice;
    77→    } else {
    78→      for (const m of Object.values(multi?.models || {}) as any[]) {
    79→        if (!m?.trained) continue;
    80→        const p = m?.predictions?.[sym];
    81→        if (p && typeof p.currentPrice === "number" && p.currentPrice > 0) {
    82→          currentPrice = p.currentPrice;
    83→          break;
    84→        }
    85→      }
    86→    }
    87→  }
    88→
    89→  // Build per-model predictions array.
    90→  // The lookup must handle three naming conventions for the model key in
    91→  // ml_predictions_multi.json's `models` dict:
    92→  //   1. "Random Forest"      → meta.name = "Random Forest" (direct match)
    93→  //   2. "RandomForest"        → meta.name with spaces stripped (nk)
    94→  //   3. "XGBoost (CL)"       → lookupName strips " (CL)" → "XGBoost"
    95→  //   4. "LightGBM-A (CL-2)"  → new CL-2 suffix — the .replace(" (CL)", "")
    96→  //                              doesn't strip " (CL-2)" so lookupName stays
    97→  //                              as "LightGBM-A (CL-2)" which matches the
    98→  //                              JSON key directly.
    99→  const predictions: ModelPrediction[] = ALL_MODELS.map((meta) => {
   100→    const lookupName = meta.name.replace(" (CL)", "");
   101→    const nk = lookupName.replace(/\s+/g, "");
   102→    const md = multi?.models?.[lookupName] || multi?.models?.[nk] || multi?.models?.[meta.name];
   103→    if (!md?.trained)
   104→      return { ...meta, trained: false, untrainedReason: meta.untrainedReason || multi?.untrained_models?.[meta.name] };
   105→    const p = md.predictions?.[sym];
   106→    if (!p)
   107→      return { ...meta, trained: true, untrainedReason: `No prediction for ${sym}` };
   108→    const bt = multi?.backtesting?.[lookupName] || multi?.backtesting?.[nk] || multi?.backtesting?.[meta.name];
   109→    return {
   110→      ...meta,
   111→      trained: true,
   112→      direction: p.direction,
   113→      signal: p.signal,
   114→      confidence: p.confidence,
   115→      currentPrice: p.currentPrice,
   116→      entry: p.entry,
   117→      sl: p.sl,
   118→      tp1: p.tp1,
   119→      tp2: p.tp2,
   120→      tp3: p.tp3,
   121→      avgReturn: p.avgReturn,
   122→      // Pass-through extras for CL-2 models (raw_prob, exp_win_rate, agreement)
   123→      // — the UI can optionally display these.
   124→      ...(p.raw_prob !== undefined ? { raw_prob: p.raw_prob } : {}),
   125→      ...(p.exp_win_rate !== undefined ? { exp_win_rate: p.exp_win_rate } : {}),
   126→      ...(p.agreement !== undefined ? { agreement: p.agreement } : {}),
   127→      metrics: md.metrics,

   160→          update: { predictionsJson: json, dataSourceDate: built.dataSourceDate, computedAt: new Date() },
   161→        });
   162→        cacheRow = { symbol: sym, predictionsJson: json, computedAt: new Date(), dataSourceDate: built.dataSourceDate } as any;
   163→      }
   164→    }
   165→    if (cacheRow) {
   166→      const fb = JSON.parse(cacheRow.predictionsJson || "{}") as {
   167→        models?: Record<string, any>;
   168→        ensembles?: Record<string, any>;
   169→      };
   170→      fbModels = fb.models || {};
   171→      fbEnsembles = fb.ensembles || {};

   178→          fbModels[meta.name] ||
   179→          fbModels[meta.name.replace(" (CL)", "")] ||
   180→          fbModels[meta.name.replace(" (X)", "")] ||
   181→          fbModels[meta.name.replace(/\s+/g, "")];

   183→          predictions[i] = {
   184→            ...meta,
   185→            trained: true,
   186→            extended: true,
   187→            untrainedReason: undefined,
   188→            signal: fp.signal,
   189→            direction: fp.direction,

   195→            tp2: fp.tp2,

   200→            ...(fp.agreement !== undefined ? { agreement: fp.agreement } : {}),

   202→          extendedCount++;
   203→        }
   204→      }
   205→    }
   206→  } catch (e) {

   208→  }

   489→
   490→  // Issue 3: TECHNICAL FALLBACK for stocks NOT in the 361-stock ML universe
   491→  // (e.g. TISL — Tasdeeq Limited — has no ML predictions, so every model shows
   492→  // 0). For those, generate a real signal from the live OHLCV using EMA/RSI/ATR
   493→  // so the user gets a BUY/SELL + SL/TP1/2/3 + confidence instead of zeros.
   494→  let fallbackNote: string | null = null;
   495→  if (ensembles.length === 0 || (!finalConsensus.entry || finalConsensus.entry <= 0)) {
   496→    try {
   497→      // 1. Load OHLCV — DB backup first, then live psxdata.
   498→      let bars: { date: string; open: number; high: number; low: number; close: number; volume: number }[] = [];
   499→      try {
   500→        const rows = await db.ohlcvBar.findMany({ where: { symbol: sym }, orderBy: { date: "asc" }, take: 400 });
   501→        bars = rows.map((r) => ({ date: r.date, open: r.open, high: r.high, low: r.low, close: r.close, volume: r.volume }));
   502→      } catch {}
   503→      if (bars.length < 30) {
   504→        const hist = await getHistorical(sym);
   505→        bars = (hist || []).map((b: any) => ({ date: b.date, open: Number(b.open), high: Number(b.high), low: Number(b.low), close: Number(b.close), volume: Number(b.volume || 0) }));
   506→      }
   507→      if (bars.length >= 30) {
   508→        const closes = bars.map((b) => b.close);
   509→        const entry = closes[closes.length - 1];
   510→        if (!currentPrice || currentPrice <= 0) currentPrice = entry;
   511→        // ATR(14)
   512→        const trs: number[] = [];
   513→        for (let i = 1; i < bars.length; i++) {
   514→          trs.push(Math.max(bars[i].high - bars[i].low, Math.abs(bars[i].high - bars[i - 1].close), Math.abs(bars[i].low - bars[i - 1].close)));
   515→        }
   516→        const atrVal = trs.slice(-14).reduce((s, v) => s + v, 0) / Math.min(14, trs.length || 1) || entry * 0.02;
   517→        // EMA20, EMA50, RSI(14)
   518→        const ema = (period: number) => {
   519→          const k = 2 / (period + 1);
   520→          let e = closes[0];
   521→          for (let i = 1; i < closes.length; i++) e = closes[i] * k + e * (1 - k);
   522→          return e;
   523→        };
   524→        const ema20 = ema(20), ema50 = ema(50);
   525→        const rsiCalc = () => {
   526→          let g = 0, l = 0;
   527→          for (let i = closes.length - 14; i < closes.length; i++) {
   528→            if (i < 1) continue;
   529→            const ch = closes[i] - closes[i - 1];
   530→            if (ch > 0) g += ch; else l -= ch;
   531→          }
   532→          const ag = g / 14, al = l / 14;
   533→          if (al === 0) return 100;
   534→          return 100 - 100 / (1 + ag / al);
   535→        };
   536→        const rsi = rsiCalc();
   537→        // Direction: BUY on uptrend (EMA20>EMA50) OR oversold bounce (RSI<30);
   538→        // SELL on downtrend (EMA20<EMA50) OR overbought (RSI>70); else HOLD.
   539→        let signal: "BUY" | "SELL" | "HOLD" = "HOLD";
   540→        let direction: "LONG" | "SHORT" = "LONG";
   541→        let confidence = 40;
   542→        if (ema20 > ema50 && rsi < 70) { signal = "BUY"; direction = "LONG"; confidence = 55 + Math.min(30, Math.abs((rsi - 50))); }
   543→        else if (ema20 < ema50 && rsi > 30) { signal = "SELL"; direction = "SHORT"; confidence = 55 + Math.min(30, Math.abs((rsi - 50))); }
   544→        else if (rsi < 30) { signal = "BUY"; direction = "LONG"; confidence = 50; }
   545→        else if (rsi > 70) { signal = "SELL"; direction = "SHORT"; confidence = 50; }
   546→        const sl = signal === "BUY" ? entry - 1.5 * atrVal : entry + 1.5 * atrVal;
   547→        const tp1 = signal === "BUY" ? entry + 1 * atrVal : entry - 1 * atrVal;
   548→        const tp2 = signal === "BUY" ? entry + 2 * atrVal : entry - 2 * atrVal;
   549→        const tp3 = signal === "BUY" ? entry + 3 * atrVal : entry - 3 * atrVal;
   550→        // Overwrite finalConsensus with the technical signal
   551→        const fcAny = finalConsensus as any;
   552→        fcAny.signal = signal;
   553→        fcAny.direction = direction;
   554→        fcAny.confidence = Math.round(confidence);
   555→        fcAny.entry = entry;
   556→        fcAny.sl = sl;
   557→        fcAny.tp1 = tp1;
   558→        fcAny.tp2 = tp2;
   559→        fcAny.tp3 = tp3;
   560→        fcAny.predictedPriceRange = { min: Math.min(tp1, tp2, tp3), max: Math.max(tp1, tp2, tp3) };
   561→        fcAny.agreement = signal === "HOLD" ? 0 : 100;
   562→        // Add a Technical Fallback ensemble so the UI shows the signal
   563→        ensembles.push({
   564→          method: "TECHNICAL_FALLBACK", methodLabel: "Technical Fallback (EMA/RSI/ATR — no ML prediction)",
   565→          signal, direction, confidence: Math.round(confidence),
   566→          predictedPrice: tp3, entry, sl, tp1, tp2, tp3,
   567→          modelCount: 1, buyCount: signal === "BUY" ? 1 : 0, sellCount: signal === "SELL" ? 1 : 0, holdCount: signal === "HOLD" ? 1 : 0,
   568→        } as any);
   569→        fallbackNote = `Technical fallback — ${sym} is outside the 361-stock ML-trained universe, so no model produced a prediction. This signal is generated from live OHLCV using EMA(20/50) trend + RSI(14) + ATR(14) SL/TP. EMA20=${ema20.toFixed(2)}, EMA50=${ema50.toFixed(2)}, RSI=${rsi.toFixed(1)}, ATR=${atrVal.toFixed(2)}.`;
   570→      } else {
   571→        fallbackNote = `${sym} has no ML prediction and not enough OHLCV history (< 30 bars) to generate a technical fallback signal.`;
   572→      }
   573→    } catch (e: any) {
   574→      fallbackNote = `Technical fallback failed: ${e?.message || e}`;
   575→    }
   576→  }
   577→
   578→  // Build tiers — include TIER_GIAN
   579→  const tiers = (
   580→    ["TIER_1", "TIER_2", "TIER_3", "TIER_4", "TIER_CL", "TIER_CL_2", "TIER_PI", "TIER_GM", "TIER_CG", "TIER_X", "TIER_GIAN"] as const
   581→  ).map((t) => ({
   582→    tier: t,
   583→    label: ALL_MODELS.find((m) => m.tier === t)?.tierLabel || t,
   584→    models: predictions.filter((p) => p.tier === t),
   585→  }));
   586→
   587→  // Issue 2: merge DB-cached extended-universe backtests (cover ALL stocks,
   588→  // incl. new / low-data listings) — DB values take precedence over the file.
   589→  let dbModelBacktests: Record<string, any> = {};
   590→  let dbEnsBacktests: Record<string, any> = {};
   591→  try {
   592→    const btRows = await db.modelBacktestCache.findMany();
   593→    for (const r of btRows) {
   594→      const stats = JSON.parse(r.payloadJson);
   595→      if (r.kind === "ensemble") dbEnsBacktests[r.key] = stats;
   596→      else dbModelBacktests[r.key] = stats;
   597→    }
   598→  } catch {}
   599→
   600→  const resp = {
   601→    symbol: sym,
   602→    stockName,
   603→    currentPrice,
   604→    tiers,
   605→    ensembles,
   606→    finalConsensus,
   607→    fallbackNote,
   608→    // Issue 2: extended-training transparency
   609→    extendedCount,
   610→    extendedAt,
   611→    trainedModelCount: predictions.filter((p) => p.trained).length,
   612→    totalModelCount: predictions.length,
   613→    trainedAt: multi?.trained_at,
   614→    backtesting: { ...(multi?.backtesting || {}), ...dbModelBacktests },
   615→    ensembleBacktesting: { ...(multi?.ensemble_backtesting || {}), ...dbEnsBacktests },
   616→    backtestSource: Object.keys(dbModelBacktests).length > 0 ? "db-extended-universe" : "file",
   617→    walkForward: multi?.walk_forward_backtesting || {},
   618→    clEnsemble: multi?.cl_ensemble_backtest || {},
   619→    clEnsembleAll: multi?.cl_ensemble_backtest_all || {},
   620→    clExtender: multi?.cl_extender_backtest || {},
   621→    clEnsemblePredictions: multi?.cl_ensemble_predictions || {},
   622→    clExtenderPredictions: multi?.cl_extender_predictions || {},
   623→    // CL-2 ensemble predictions + metadata
   624→    cl2EnsemblePredictions: multi?.cl2_ensemble_predictions || {},
   625→    cl2Metadata: multi?.cl2_metadata || {},
   626→    // PI-Ensemble predictions + metadata
   627→    piEnsemblePredictions: multi?.pi_ensemble_predictions || {},
   628→    piMetadata: multi?.pi_metadata || {},
   629→    // GM-Model predictions + metadata
   630→    gmModelPredictions: multi?.gm_model_predictions || {},
   631→    gmMetadata: multi?.gm_metadata || {},
   632→    // CG-Ensemble predictions + metadata
   633→    cgEnsemblePredictions: multi?.cg_ensemble_predictions || {},
   634→    cgMetadata: multi?.cg_metadata || {},
   635→    // X-Ensemble predictions + metadata (NEW)
   636→    xEnsemblePredictions: multi?.x_ensemble_predictions || {},
   637→    xMetadata: multi?.x_metadata || {},
   638→    // Gian-Ensemble + Gian-Extender predictions + metadata
   639→    gianEnsemblePredictions: multi?.gian_ensemble_predictions || {},
   640→    gianExtenderPredictions: multi?.gian_extender_predictions || {},
   641→    gianMetadata: multi?.gian_metadata || {},
   642→    gianExtenderMetadata: multi?.gian_extender_metadata || {},
   643→    // M-Ensemble predictions + metadata
   644→    mEnsemblePredictions: multi?.m_ensemble_predictions || {},
   645→    mMetadata: multi?.m_metadata || {},
   646→  };
   647→
   648→  await cacheSetJSON(CK(sym), resp, 3600);
   649→  return NextResponse.json({ ...resp, cached: false, latencyMs: Date.now() - start });
   650→}
