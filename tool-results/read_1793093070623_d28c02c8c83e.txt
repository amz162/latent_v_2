    49→def compute_rsi(p, n=14):
    50→    d = p.diff(); g = d.where(d > 0, 0); l = -d.where(d < 0, 0)
    51→    ag = g.ewm(alpha=1/n, min_periods=n).mean(); al = l.ewm(alpha=1/n, min_periods=n).mean()
    52→    return (100 - 100 / (1 + ag / al.replace(0, np.nan))).fillna(50)
    53→
    54→
    55→def compute_macd(p):
    56→    e12 = p.ewm(span=12, adjust=False).mean(); e26 = p.ewm(span=26, adjust=False).mean()
    57→    m = e12 - e26; s = m.ewm(span=9, adjust=False).mean(); return m, s, m - s
    58→

    60→def compute_bb(p, w=20):
    61→    sma = p.rolling(w).mean(); sd = p.rolling(w).std()
    62→    return ((sma + 2*sd - (sma - 2*sd)) / sma).fillna(0), \
    63→           ((p - (sma - 2*sd)) / (4*sd).replace(0, np.nan)).fillna(0.5)

    66→def compute_atr(h, l, c, n=14):
    67→    tr = pd.concat([h - l, (h - c.shift()).abs(), (l - c.shift()).abs()], axis=1).max(axis=1)
    68→    return tr.rolling(n).mean().fillna(0)

    71→def compute_obv(c, v):
    72→    s = c.diff().apply(lambda x: 1 if x > 0 else (-1 if x < 0 else 0))
    73→    return (s * v).cumsum()

   126→def apply_atr_rules(signal, current_price, atr):
   127→    """Return (entry, sl, tp1, tp2, tp3) following ATR rules.
   128→
   129→    For BUY: entry = current_price, SL = entry - 1.5×ATR (below), TPs above
   130→    For SELL: entry = current_price, SL = entry + 1.5×ATR (above), TPs below
   131→    For HOLD: returns zeros (model not tradeable)
   132→    """
   133→    if signal == "BUY":
   134→        entry = current_price
   135→        sl = entry - SL_MULT * atr
   136→        tp1 = entry + TP1_MULT * atr
   137→        tp2 = entry + TP2_MULT * atr
   138→        tp3 = entry + TP3_MULT * atr
   139→    elif signal == "SELL":
   140→        entry = current_price
   141→        sl = entry + SL_MULT * atr  # ABOVE entry (fixed!)
   142→        tp1 = entry - TP1_MULT * atr  # below entry (target on downside)
   143→        tp2 = entry - TP2_MULT * atr
   144→        tp3 = entry - TP3_MULT * atr
   145→    else:  # HOLD
   146→        entry = current_price
   147→        sl = current_price
   148→        tp1 = current_price
   149→        tp2 = current_price
   150→        tp3 = current_price
   151→    return entry, sl, tp1, tp2, tp3

   154→def signal_from_avg_return(avg_return):
   155→    """Convert avgReturn to BUY/SELL/HOLD using relaxed thresholds."""
   156→    if avg_return > 0.005:  # Lowered from 0.01 to 0.005 — captures more signals
   157→        return "BUY", "LONG"
   158→    elif avg_return < -0.005:
   159→        return "SELL", "SHORT"
   160→    else:
   161→        return "HOLD", "LONG"
