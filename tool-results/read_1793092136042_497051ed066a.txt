    48→def compute_rsi_vec(close, n=14):
    49→    """Vectorized RSI computation for a single stock's close prices."""
    50→    diff = close.diff()
    51→    gain = diff.where(diff > 0, 0.0)
    52→    loss = -diff.where(diff < 0, 0.0)
    53→    avg_gain = gain.ewm(alpha=1/n, min_periods=n).mean()
    54→    avg_loss = loss.ewm(alpha=1/n, min_periods=n).mean()
    55→    rs = avg_gain / avg_loss.replace(0, np.nan)
    56→    return (100 - 100 / (1 + rs)).fillna(50)
    57→
    58→def compute_macd_vec(close):
    59→    """Vectorized MACD."""
    60→    e12 = close.ewm(span=12, adjust=False).mean()
    61→    e26 = close.ewm(span=26, adjust=False).mean()
    62→    macd = e12 - e26
    63→    signal = macd.ewm(span=9, adjust=False).mean()
    64→    return macd, signal, macd - signal
    65→
    66→def compute_bb_vec(close, w=20):
    67→    """Vectorized Bollinger Bands."""
    68→    sma = close.rolling(w).mean()
    69→    sd = close.rolling(w).std()
    70→    width = ((sma + 2*sd - (sma - 2*sd)) / sma).fillna(0)
    71→    pct_b = ((close - (sma - 2*sd)) / (4 * sd).replace(0, np.nan)).fillna(0.5)
    72→    return width, pct_b
    73→
    74→def compute_atr_vec(high, low, close, n=14):
    75→    """Vectorized ATR."""
    76→    tr = pd.concat([
    77→        high - low,
    78→        (high - close.shift()).abs(),
    79→        (low - close.shift()).abs(),
    80→    ], axis=1).max(axis=1)
    81→    return tr.rolling(n).mean().fillna(0)
    82→
    83→def compute_obv_vec(close, volume):
    84→    """Vectorized OBV."""
    85→    direction = close.diff().apply(lambda x: 1 if x > 0 else (-1 if x < 0 else 0))
    86→    return (direction * volume).cumsum()

    88→def add_features_grouped(df):
    89→    """Add ALL technical features to the DataFrame using groupby.
    90→    This is the KEY optimization — processes ALL stocks at once instead of
    91→    looping through 883 stocks with df[df["symbol"]==s] filters.
    92→    """
    93→    df = df.copy().sort_values(["symbol", "date"]).reset_index(drop=True)
    94→
    95→    # Group by symbol and compute features VECTORIZED per group
    96→    # This is ~50× faster than filtering 883 times
    97→    groups = df.groupby("symbol", sort=False)
    98→
    99→    # RSI
   100→    df["rsi_14"] = groups["close"].transform(compute_rsi_vec)
   101→
   102→    # MACD
   103→    macd = groups["close"].transform(lambda c: compute_macd_vec(c)[0])
   104→    df["macd"] = macd
   105→    df["macd_signal"] = groups["close"].transform(lambda c: compute_macd_vec(c)[1])
   106→    df["macd_hist"] = groups["close"].transform(lambda c: compute_macd_vec(c)[2])
   107→
   108→    # Bollinger Bands
   109→    df["bb_width"] = groups["close"].transform(lambda c: compute_bb_vec(c)[0])
   110→    df["bb_pct_b"] = groups["close"].transform(lambda c: compute_bb_vec(c)[1])
   111→
   112→    # ATR
   113→    df["atr_14"] = groups.apply(lambda g: compute_atr_vec(g["high"], g["low"], g["close"])).reset_index(level=0, drop=True)
   114→
   115→    # OBV
   116→    obv = groups.apply(lambda g: compute_obv_vec(g["close"], g["volume"])).reset_index(level=0, drop=True)
   117→    df["obv_change_pct"] = obv.groupby(df["symbol"]).pct_change(periods=14).replace([np.inf, -np.inf], 0).fillna(0)
   118→
   119→    # SMA ratios
   120→    for w in [20, 50, 200]:
   121→        sma = groups["close"].transform(lambda c, w=w: c.rolling(w, min_periods=1).mean())
   122→        df[f"price_sma_{w}_ratio"] = df["close"] / sma
   123→
   124→    # Volume SMA ratio
   125→    vol_sma = groups["volume"].transform(lambda v: v.rolling(20, min_periods=1).mean().replace(0, 1))
   126→    df["volume_sma_20_ratio"] = df["volume"] / vol_sma
   127→
   128→    # Returns
   129→    for p in [1, 5, 7, 14, 30]:
   130→        df[f"return_{p}d"] = groups["close"].pct_change(periods=p).replace([np.inf, -np.inf], 0).fillna(0)
   131→
   132→    # HL range
   133→    df["hl_range_pct"] = ((df["high"] - df["low"]) / df["close"].replace(0, np.nan)).fillna(0)
   134→
   135→    # Body size
   136→    df["body_size_pct"] = ((df["close"] - df["open"]).abs() / df["close"].replace(0, np.nan)).fillna(0)
   137→
   138→    # Volatility
   139→    df["volatility_14d"] = groups["close"].pct_change().groupby(df["symbol"]).rolling(14).std().reset_index(level=0, drop=True).fillna(0)
   140→
   141→    # Labels (forward returns)
   142→    for hz in [5, 10, 21]:
   143→        df[f"target_{hz}d"] = groups["close"].shift(-hz)
   144→    df["target_5d"] = (df.groupby("symbol")["close"].shift(-5) - df["close"]) / df["close"]
   145→    df["target_10d"] = (df.groupby("symbol")["close"].shift(-10) - df["close"]) / df["close"]
   146→    df["target_21d"] = (df.groupby("symbol")["close"].shift(-21) - df["close"]) / df["close"]
   147→
   148→    return df

   196→# ─── SL/TP rules + signal generation (same as existing scripts) ────────────
   197→
   198→def atr_rules(sig, cp, atr):
   199→    if sig == "BUY":
   200→        return cp, cp - SL_MULT * atr, cp + TP1_MULT * atr, cp + TP2_MULT * atr, cp + TP3_MULT * atr
   201→    if sig == "SELL":
   202→        return cp, cp + SL_MULT * atr, cp - TP1_MULT * atr, cp - TP2_MULT * atr, cp - TP3_MULT * atr
   203→    return cp, cp, cp, cp, cp
   204→
   205→def sig_from_avg(ar):
   206→    if ar > SIG_THRESHOLD:
   207→        return "BUY", "LONG"
   208→    if ar < -SIG_THRESHOLD:
   209→        return "SELL", "SHORT"
   210→    return "HOLD", "LONG"
   211→
   212→def si(v):
   213→    try:
   214→        if np.isnan(v):
   215→            return 0
   216→        return min(100, int(abs(v) * 1000))
   217→    except:
   218→        return 0
   219→
   220→def mk_pred(cp, atr, avg):
   221→    sg, d = sig_from_avg(avg)
   222→    e, sl, t1, t2, t3 = atr_rules(sg, cp, atr)
   223→    return {
   224→        "signal": sg, "direction": d, "confidence": si(avg),
   225→        "currentPrice": round(cp, 2), "entry": round(e, 2),
   226→        "tp1": round(t1, 2), "tp2": round(t2, 2), "tp3": round(t3, 2),
   227→        "sl": round(sl, 2), "avgReturn": round(float(avg) if not np.isnan(avg) else 0.0, 4),
   228→    }

   232→def train_base_regressors(df, atr_per, stocks):
   233→    """Train 5 base regressor models (XGBoost, LightGBM, CatBoost, RF, Ridge).
   234→    Uses the SAME training config as the existing retrain script.
   235→    """
   236→    import xgboost as xgb
   237→    import lightgbm as lgb
   238→    import catboost as cb
   239→    from sklearn.ensemble import RandomForestRegressor
   240→    from sklearn.linear_model import Ridge
   241→
   242→    print("\n[Base Models] Training 5 regressors...")
   243→    targets = ["target_5d", "target_10d", "target_21d"]
   244→    dft = df.sample(n=min(100000, len(df)), random_state=42).dropna(subset=FC + targets)
   245→    sp = int(len(dft) * 0.8)
   246→    Xtr = dft.iloc[:sp][FC].fillna(0).values
   247→
   248→    # Latest features per stock (for prediction)
   249→    lat = {}
   250→    for s in stocks:
   251→        sd = df[df["symbol"] == s].sort_values("date").reset_index(drop=True)
   252→        if len(sd) < 30:
   253→            continue
   254→        lat[s] = {"f": sd.iloc[-1][FC].fillna(0).values.reshape(1, -1), "cp": float(sd.iloc[-1]["close"])}
   255→
   256→    cfg = {
   257→        "XGBoost": lambda: xgb.XGBRegressor(n_estimators=50, max_depth=3, learning_rate=0.1, random_state=42, n_jobs=-1),
   258→        "LightGBM": lambda: lgb.LGBMRegressor(n_estimators=50, num_leaves=15, learning_rate=0.1, random_state=42, verbose=-1, n_jobs=-1),
   259→        "CatBoost": lambda: cb.CatBoostRegressor(iterations=50, depth=3, learning_rate=0.1, random_state=42, verbose=0, allow_writing_files=False),
   260→        "Random Forest": lambda: RandomForestRegressor(n_estimators=50, max_depth=5, min_samples_leaf=15, max_features=0.7, random_state=42, n_jobs=-1),
   261→        "Ridge Regression": lambda: Ridge(alpha=10.0, random_state=42),
   262→    }
   263→
   264→    res = {}
   265→    for nm, fn in cfg.items():
   266→        print(f"  {nm}...", end=" ", flush=True)
   267→        pr = {}
   268→        for tc in targets:
   269→            hz = int(tc.split("_")[1].replace("d", ""))
   270→            m = fn()
   271→            m.fit(Xtr, dft.iloc[:sp][tc].values)
   272→            for s, lf in lat.items():
   273→                p = float(m.predict(lf["f"])[0])
   274→                if s not in pr:
   275→                    pr[s] = {"currentPrice": lf["cp"], "entry": lf["cp"]}
   276→                pr[s][f"ret_{hz}d"] = p
   277→        for s, pd_ in pr.items():
   278→            avg = (pd_.get("ret_5d", 0) + pd_.get("ret_10d", 0) + pd_.get("ret_21d", 0)) / 3
   279→            atr = atr_per.get(s, {"atr": pd_["currentPrice"] * 0.02})["atr"]
   280→            pd_.update(mk_pred(pd_["currentPrice"], atr, avg))
   281→        res[nm] = {"trained": True, "version": "3.0.0", "predictions": pr}
   282→        print(f"done ({len(pr)})")
   283→    return res

   285→# ─── Classical models (ARIMA, Holt-Winters, GARCH, Prophet, Kalman, GRU) ───

   287→def train_classical_models(df, atr_per, stocks):
   288→    """Train 6 classical time-series models (same analog formulas as existing)."""
   289→    print("\n[Classical Models] Training 6 models...")
   290→    try:
   291→        from statsmodels.tsa.arima.model import ARIMA
   292→        from statsmodels.tsa.holtwinters import ExponentialSmoothing
   293→        has_sm = True
   294→    except:
   295→        has_sm = False
   296→
   297→    def _ar(sd, cp, atr):
   298→        try:
   299→            if has_sm:
   300→                fc = ARIMA(sd["close"].tail(60).values, order=(2,1,0)).fit().forecast(steps=21)
   301→                avg = ((float(fc[4]) + float(fc[9]) + float(fc[20])) / 3 - cp) / cp
   302→            else:
   303→                avg = float(sd.iloc[-1]["return_5d"]) * 0.6 + float(sd.iloc[-1]["return_30d"]) * 0.4
   304→        except:
   305→            avg = float(sd.iloc[-1]["return_5d"]) * 0.6 + float(sd.iloc[-1]["return_30d"]) * 0.4
   306→        return mk_pred(cp, atr, avg)
   307→
   308→    def _hw(sd, cp, atr):
   309→        try:
   310→            if has_sm:
   311→                fc = ExponentialSmoothing(sd["close"].tail(60).values, trend="add", seasonal=None).fit().forecast(21)
   312→                avg = ((float(fc[4]) + float(fc[9]) + float(fc[20])) / 3 - cp) / cp
   313→            else:
   314→                avg = float(sd.iloc[-1]["return_5d"])
   315→        except:
   316→            avg = float(sd.iloc[-1]["return_5d"])
   317→        return mk_pred(cp, atr, avg)
   318→
   319→    def _ga(sd, cp, atr):
   320→        rv = float(sd["return_1d"].tail(14).std())
   321→        hv = float(sd["return_1d"].tail(60).std())
   322→        vr = rv / max(hv, 0.001)
   323→        r5 = float(sd.iloc[-1]["return_5d"])
   324→        avg = 0.015 if vr < 0.9 and r5 > -0.01 else (-0.015 if vr > 1.2 and r5 < 0.01 else r5 * 0.5)
   325→        return mk_pred(cp, atr, avg)
   326→
   327→    def _pr(sd, cp, atr):
   328→        try:
   329→            pr = pd.Series(sd["close"].tail(60).values)
   330→            sl = np.polyfit(np.arange(len(pr)), pr.values, 1)[0]
   331→            avg = ((sl * 21) / cp + float(np.mean(pr.pct_change(5).fillna(0).tail(10).values)) * 21) / 2
   332→        except:
   333→            avg = float(sd.iloc[-1]["return_30d"])
   334→        return mk_pred(cp, atr, avg)
   335→
   336→    def _kf(sd, cp, atr):
   337→        return mk_pred(cp, atr, float(sd.iloc[-1]["return_5d"]) * 0.7 + float(sd.iloc[-1]["return_30d"]) * 0.3)
   338→
   339→    def _gru(sd, cp, atr):
   340→        return mk_pred(cp, atr, (float(sd.iloc[-1]["return_1d"]) * 0.5 + float(sd.iloc[-1]["return_7d"]) * 0.5 + float(sd.iloc[-1]["return_7d"]) * 0.7 + float(sd.iloc[-1]["return_1d"]) * 0.3 + float(sd.iloc[-1]["return_7d"])) / 3)
   341→
   342→    res = {}
   343→    for nm, fn in [("ARIMA", _ar), ("Holt-Winters", _hw), ("GARCH", _ga), ("Prophet", _pr), ("Kalman Filter", _kf), ("GRU", _gru)]:
   344→        print(f"  {nm}...", end=" ", flush=True)
   345→        pr = {}
   346→        for s in stocks:
   347→            sd = df[df["symbol"] == s].sort_values("date").reset_index(drop=True)
   348→            if len(sd) < 60:
   349→                continue
   350→            cp = atr_per.get(s, {}).get("cp", float(sd.iloc[-1]["close"]))
   351→            atr = atr_per.get(s, {"atr": cp * 0.02})["atr"]
   352→            p = fn(sd, cp, atr)
   353→            if p:
   354→                pr[s] = p
   355→        res[nm] = {"trained": True, "version": "3.0.0", "predictions": pr}
   356→        print(f"done ({len(pr)})")
   357→    return res

   425→def compute_all_ensembles(all_models, atr_per, stocks):
   426→    """Compute 14 ensemble predictions from the 22 base models."""
   427→    print("\n[Ensembles] Computing 14 ensembles...")
   428→    preds = {}
   429→
   430→    # Helper: get a model's predictions
   431→    def mp(name):
   432→        return all_models.get(name, {}).get("predictions", {})
   433→
   434→    # 1. CL_ENSEMBLE: LightGBM 70% + Ridge 30%
   435→    lgb_p = mp("LightGBM")
   436→    rdg_p = mp("Ridge Regression")
   437→    cl_ens = {}
   438→    for s in stocks:
   439→        ar = 0.7 * lgb_p.get(s, {}).get("avgReturn", 0) + 0.3 * rdg_p.get(s, {}).get("avgReturn", 0)
   440→        cp = lgb_p.get(s, {}).get("currentPrice", 0)
   441→        atr = atr_per.get(s, {"atr": cp * 0.02})["atr"]
   442→        cl_ens[s] = mk_pred(cp, atr, ar)
   443→    preds["cl_ensemble"] = cl_ens
   444→    print(f"  CL_ENSEMBLE: {sum(1 for v in cl_ens.values() if v['signal']!='HOLD')} signals")
   445→
   446→    # 2. CL_EXTENDER: 8-model weighted vote
   447→    ep = {n: all_models.get(n, {}).get("predictions", {}) for n in EXT_W}
   448→    cl_ext = {}
   449→    for s in stocks:
   450→        ars = []
   451→        for mn, w in EXT_W.items():
   452→            p = ep.get(mn, {}).get(s, {})
   453→            ars.append(p.get("avgReturn", 0) * w)
   454→        ar = sum(ars)
   455→        cp = ep.get("LightGBM", {}).get(s, {}).get("currentPrice", 0)
   456→        atr = atr_per.get(s, {"atr": cp * 0.02})["atr"]
   457→        cl_ext[s] = mk_pred(cp, atr, ar)
   458→    preds["cl_extender"] = cl_ext
   459→    print(f"  CL_EXTENDER: {sum(1 for v in cl_ext.values() if v['signal']!='HOLD')} signals")
   460→
   461→    # 3-8. NEW ensembles (weighted avgReturn combinations)
   462→    ensemble_defs = {
   463→        "cl2_ensemble": [("LightGBM", 0.50), ("CatBoost", 0.30), ("Ridge Regression", 0.20)],
   464→        "pi_ensemble": [("ARIMA", 0.40), ("Holt-Winters", 0.30), ("Prophet", 0.30)],
   465→        "gm_model": [("XGBoost", 0.35), ("LightGBM", 0.30), ("CatBoost", 0.20), ("Random Forest", 0.15)],
   466→        "cg_ensemble": [("LightGBM", 0.50), ("XGBoost", 0.50)],
   467→        "x_ensemble": [("XGBoost", 0.20), ("LightGBM", 0.20), ("CatBoost", 0.20), ("Random Forest", 0.20), ("Ridge Regression", 0.20)],
   468→        "gian_ensemble": [("LSTM", 0.333), ("CNN-LSTM", 0.333), ("Transformer-LSTM", 0.334)],
   469→        "gian_extender": [("PatchTST", 0.25), ("N-BEATS", 0.25), ("TimesNet", 0.25), ("Informer", 0.25)],
   470→        "m_ensemble": [("Chronos (Amazon)", 0.25), ("Lag-Llama", 0.25), ("Moirai (Salesforce)", 0.25), ("Time-LLM", 0.25)],
   471→    }
   472→    for ens_name, ens_def in ensemble_defs.items():
   473→        ens_preds = {}
   474→        for s in stocks:
   475→            ar_w = 0.0
   476→            cp = 0.0
   477→            for mn, w in ens_def:
   478→                p = mp(mn).get(s, {})
   479→                ar_w += p.get("avgReturn", 0) * w
   480→                cp = p.get("currentPrice", cp)
   481→            atr = atr_per.get(s, {"atr": cp * 0.02})["atr"]
   482→            ens_preds[s] = mk_pred(cp, atr, ar_w)
   483→        preds[ens_name] = ens_preds
   484→        sig_count = sum(1 for v in ens_preds.values() if v["signal"] != "HOLD")
   485→        print(f"  {ens_name.upper()}: {sig_count} signals")
   486→
   487→    # 9-14. Basic ensembles (Majority Vote, Probability-Weighted, etc.)
   488→    all_preds = {n: mp(n) for n in all_models}
   489→
   490→    def majority_vote(s):
   491→        votes = {"BUY": 0, "SELL": 0, "HOLD": 0}
   492→        for mn in all_models:
   493→            sig = all_preds[mn].get(s, {}).get("signal", "HOLD")
   494→            votes[sig] = votes.get(sig, 0) + 1
   495→        return max(votes, key=votes.get)
   496→
   497→    def prob_weighted(s):
   498→        total_w = 0.0
   499→        for mn in all_models:
   500→            conf = all_preds[mn].get(s, {}).get("confidence", 0)
   501→            total_w += conf
   502→        buy_w = sum(all_preds[mn].get(s, {}).get("confidence", 0) for mn in all_models if all_preds[mn].get(s, {}).get("signal") == "BUY")
   503→        sell_w = sum(all_preds[mn].get(s, {}).get("confidence", 0) for mn in all_models if all_preds[mn].get(s, {}).get("signal") == "SELL")
   504→        if total_w == 0:
   505→            return "HOLD"
   506→        return "BUY" if buy_w > sell_w else ("SELL" if sell_w > buy_w else "HOLD")
   507→
   508→    TIER_W = {"TIER_1": 0.35, "TIER_2": 0.25, "TIER_3": 0.30, "TIER_4": 0.10}
   509→    tier_map = {"PatchTST": "TIER_1", "N-BEATS": "TIER_1", "TimesNet": "TIER_1", "Informer": "TIER_1",
   510→                "Chronos (Amazon)": "TIER_2", "Lag-Llama": "TIER_2", "Moirai (Salesforce)": "TIER_2", "Time-LLM": "TIER_2",
   511→                "XGBoost": "TIER_3", "LightGBM": "TIER_3", "CatBoost": "TIER_3", "Random Forest": "TIER_3", "Ridge Regression": "TIER_3",
   512→                "LSTM": "TIER_4", "CNN-LSTM": "TIER_4", "Transformer-LSTM": "TIER_4"}
   513→
   514→    def tier_weighted(s):
   515→        votes = {"BUY": 0.0, "SELL": 0.0, "HOLD": 0.0}
   516→        for mn in all_models:
   517→            tier = tier_map.get(mn, "TIER_3")
   518→            w = TIER_W.get(tier, 0.10)
   519→            sig = all_preds[mn].get(s, {}).get("signal", "HOLD")
   520→            votes[sig] = votes.get(sig, 0) + w
   521→        return max(votes, key=votes.get)
   522→
   523→    def final_consensus(s):
   524→        mv = majority_vote(s)
   525→        pw = prob_weighted(s)
   526→        tw = tier_weighted(s)
   527→        votes = {"BUY": 0, "SELL": 0, "HOLD": 0}
   528→        votes[mv] += 1
   529→        votes[pw] += 1
   530→        votes[tw] += 1
   531→        return max(votes, key=votes.get)
   532→
   533→    for ens_name, fn in [("majority_vote", majority_vote), ("probability_weighted", prob_weighted),
   534→                         ("tier_weighted", tier_weighted), ("final_consensus", final_consensus)]:
   535→        ens_preds = {}
   536→        for s in stocks:
   537→            sig = fn(s)
   538→            cp = atr_per.get(s, {}).get("cp", 0)
   539→            atr = atr_per.get(s, {"atr": cp * 0.02})["atr"]
   540→            # Use weighted avgReturn for SL/TP
   541→            ar = 0.0
   542→            n = 0
   543→            for mn in all_models:
   544→                p = all_preds[mn].get(s, {})
   545→                if p:
   546→                    ar += p.get("avgReturn", 0)
   547→                    n += 1
   548→            ar = ar / max(n, 1)
   549→            ens_preds[s] = mk_pred(cp, atr, ar)
   550→            ens_preds[s]["signal"] = sig
   551→            ens_preds[s]["direction"] = "SHORT" if sig == "SELL" else "LONG"
   552→        preds[ens_name] = ens_preds
   553→        sig_count = sum(1 for v in ens_preds.values() if v["signal"] != "HOLD")
   554→        print(f"  {ens_name.upper()}: {sig_count} signals")
   555→
   556→    return preds

   560→def backtest_model(df, preds, atr_per, name):
   561→    """Backtest a model/ensemble's predictions on 20% holdout per stock."""
   562→    b = s_ = c = cs = 0
   563→    returns = []
   564→    for sym in preds:
   565→        sig = preds[sym].get("signal", "HOLD")
   566→        if sig == "HOLD":
   567→            continue
   568→        sd = df[df["symbol"] == sym].sort_values("date").reset_index(drop=True)
   569→        if len(sd) < 30:
   570→            continue
   571→        sp = int(len(sd) * 0.8)
   572→        st = sd.iloc[sp:]
   573→        if len(st) < 5:
   574→            continue
   575→        cp = atr_per.get(sym, {}).get("cp", float(sd.iloc[-1]["close"]))
   576→        atr = atr_per.get(sym, {"atr": cp * 0.02})["atr"]
   577→        e, sl, t1, t2, t3 = atr_rules(sig, cp, atr)
   578→        for i in range(0, max(1, len(st) - min(21, len(st) - 1)), 5):
   579→            w = min(21, len(st) - i)
   580→            ep = float(st.iloc[min(i + w, len(st) - 1)]["close"])
   581→            for j in range(i, min(i + w, len(st))):
   582→                r = st.iloc[j]
   583→                nh = float(r["high"])
   584→                nl = float(r["low"])
   585→                if sig == "BUY":
   586→                    if nl <= sl:
   587→                        ep = sl
   588→                        break
   589→                    if nh >= t3:
   590→                        ep = t3
   591→                        break
   592→                    if nh >= t2:
   593→                        ep = t2
   594→                    elif nh >= t1:
   595→                        ep = t1
   596→                else:
   597→                    if nh >= sl:
   598→                        ep = sl
   599→                        break
   600→                    if nl <= t3:
   601→                        ep = t3
   602→                        break
   603→                    if nl <= t2:
   604→                        ep = t2
   605→                    elif nl <= t1:
   606→                        ep = t1
   607→            if sig == "BUY":
   608→                b += 1
   609→                a = (ep - e) / e
   610→                returns.append(a)
   611→                c += a > 0
   612→            else:
   613→                s_ += 1
   614→                a = (e - ep) / e
   615→                returns.append(a)
   616→                cs += a > 0
   617→    tt = b + s_
   618→    wr = (c + cs) / max(tt, 1) * 100
   619→    ap = float(np.mean(returns) * 100) if returns else 0
   620→    if np.isnan(ap):
   621→        ap = 0
   622→    return {
   623→        "totalTrades": int(tt), "buySignals": int(b), "sellSignals": int(s_),
   624→        "correctBuys": int(c), "correctSells": int(cs),
   625→        "winRate": round(float(wr if not np.isnan(wr) else 0), 1),
   626→        "avgReturnPct": round(float(ap), 2),
   627→        "maxReturnPct": round(float(max(returns) * 100 if returns else 0), 2),
   628→        "minReturnPct": round(float(min(returns) * 100 if returns else 0), 2),
   629→        "stars": int(min(5, max(1, int(wr / 20) + 1))) if tt > 0 else 0,
   630→    }
