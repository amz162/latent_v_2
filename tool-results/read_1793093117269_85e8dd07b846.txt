    43→def compute_rsi(p, n=14):
    44→    d = p.diff(); g = d.where(d > 0, 0); l = -d.where(d < 0, 0)
    45→    ag = g.ewm(alpha=1/n, min_periods=n).mean(); al = l.ewm(alpha=1/n, min_periods=n).mean()
    46→    return (100 - 100 / (1 + ag / al.replace(0, np.nan))).fillna(50)
    47→
    48→
    49→def compute_macd(p):
    50→    e12 = p.ewm(span=12, adjust=False).mean(); e26 = p.ewm(span=26, adjust=False).mean()
    51→    m = e12 - e26; s = m.ewm(span=9, adjust=False).mean(); return m, s, m - s
    52→
    53→
    54→def compute_bb(p, w=20):
    55→    sma = p.rolling(w).mean(); sd = p.rolling(w).std()
    56→    return ((sma + 2*sd - (sma - 2*sd)) / sma).fillna(0), \
    57→           ((p - (sma - 2*sd)) / (4*sd).replace(0, np.nan)).fillna(0.5)
    58→

    60→def compute_atr(h, l, c, n=14):
    61→    tr = pd.concat([h - l, (h - c.shift()).abs(), (l - c.shift()).abs()], axis=1).max(axis=1)
    62→    return tr.rolling(n).mean().fillna(0)
    63→
    64→
    65→def compute_obv(c, v):
    66→    s = c.diff().apply(lambda x: 1 if x > 0 else (-1 if x < 0 else 0))
    67→    return (s * v).cumsum()

    70→def add_features(df):
    71→    df = df.copy().sort_values("date").reset_index(drop=True)
    72→    c = df["close"]; h = df["high"]; l = df["low"]; v = df["volume"]
    73→    df["rsi_14"] = compute_rsi(c); m, s, hh = compute_macd(c)
    74→    df["macd"] = m; df["macd_signal"] = s; df["macd_hist"] = hh
    75→    bw, pb = compute_bb(c); df["bb_width"] = bw; df["bb_pct_b"] = pb
    76→    df["atr_14"] = compute_atr(h, l, c)
    77→    obv = compute_obv(c, v)
    78→    df["obv_change_pct"] = obv.pct_change(periods=14).replace([np.inf, -np.inf], 0).fillna(0)
    79→    for w in [20, 50, 200]:
    80→        df[f"price_sma_{w}_ratio"] = c / c.rolling(w, min_periods=1).mean()
    81→    df["volume_sma_20_ratio"] = v / v.rolling(20, min_periods=1).mean().replace(0, 1)
    82→    for p in [1, 7, 14, 30]:
    83→        df[f"return_{p}d"] = c.pct_change(periods=p).replace([np.inf, -np.inf], 0).fillna(0)
    84→    df["hl_range_pct"] = ((h - l) / c.replace(0, np.nan)).fillna(0)
    85→    df["body_size_pct"] = ((df["close"] - df["open"]).abs() / c.replace(0, np.nan)).fillna(0)
    86→    df["volatility_14d"] = c.pct_change().rolling(14).std().fillna(0)
    87→    return df

   129→def _signal_from_avg(r5, r10, r21):
   130→    avg = (r5 + r10 + r21) / 3
   131→    sig = "BUY" if avg > 0.01 else "SELL" if avg < -0.01 else "HOLD"
   132→    d = "LONG" if sig != "SELL" else "SHORT"
   133→    return sig, d, avg

   136→def _pred_record(cp, r5, r10, r21, sl_low_14):

   140→        "confidence": min(100, int(abs(avg) * 1000)),
   141→        "currentPrice": round(cp, 2), "entry": round(cp, 2),

   150→# ────────────────────────────────────────────────────────────────────
   151→# 1. Chronos (Amazon) — quantile-binned tokenize-then-forecast analog
   152→# ────────────────────────────────────────────────────────────────────
   153→def train_chronos(df):
   154→    """Chronos analog: tokenize returns into quantile bins (T5-style), then
   155→    forecast next-bin via ridge regression on the bin indices."""
   156→    print("  Training Chronos (Amazon) [quantile-binned analog]...")
   157→    targets = ["target_5d", "target_10d", "target_21d"]
   158→    # ML integrity (Task 2): chronological_split replaces the leaky global row
   159→    # split on the concatenated df (mixed train/test across stocks in stock-
   160→    # concat order, not date order). Per-horizon mask inside the loop.
   161→    tr, va, te = chronological_split(df)
   162→
   163→    # Quantile-bin the 1d returns into 100 tokens (Chronos uses 1024, we use 100 for speed)
   164→    bins = np.linspace(-0.05, 0.05, 101)  # 100 bins
   165→    bin_centers = (bins[:-1] + bins[1:]) / 2
   166→
   167→    def tokenize(returns):
   168→        idx = np.digitize(returns, bins) - 1
   169→        return np.clip(idx, 0, len(bin_centers) - 1)
   170→
   171→    # Build lag features from tokenized returns (Chronos uses lag tokens)
   172→    rets_tr = tr["return_1d"].values
   173→    tokens_tr = tokenize(rets_tr)
   174→    # Use last 10 tokens as features (lag window)
   175→    LAG = 10
   176→    Xtr_full = np.array([tokens_tr[i:i+LAG] for i in range(len(tokens_tr) - LAG)])
   177→    ytr_5_full = tr["target_5d"].values[LAG:]
   178→    ytr_10_full = tr["target_10d"].values[LAG:]
   179→    ytr_21_full = tr["target_21d"].values[LAG:]
   180→
   181→    # Pre-build test arrays (Xte only depends on tokens_te, shared across horizons).
   182→    rets_te = te["return_1d"].values
   183→    toks_te = tokenize(rets_te)
   184→    Xte_full = np.array([toks_te[i:i+LAG] for i in range(len(toks_te) - LAG)])
   185→    yte_5_full = te["target_5d"].values[LAG:]
   186→    yte_10_full = te["target_10d"].values[LAG:]
   187→    yte_21_full = te["target_21d"].values[LAG:]
   188→
   189→    if len(Xtr_full) < 50:
   190→        return {"trained": False, "error": "Not enough tokenized sequences"}
   191→
   192→    # Three ridge models, one per horizon
   193→    metrics = {}; models = {}
   194→    for tc, ytr_full, yte_full in zip(targets,
   195→                                       [ytr_5_full, ytr_10_full, ytr_21_full],
   196→                                       [yte_5_full, yte_10_full, yte_21_full]):
   197→        hz = int(tc.split("_")[1].replace("d", ""))
   198→        # ML integrity (Phase A1): per-horizon drop — mask rows whose THIS-horizon
   199→        # target is NaN/inf (preserves shorter-horizon rows). Keeps X and y aligned.
   200→        mask_tr = np.isfinite(ytr_full)
   201→        mask_te = np.isfinite(yte_full)
   202→        Xtr = Xtr_full[mask_tr]
   203→        ytr = ytr_full[mask_tr]
   204→        Xte = Xte_full[mask_te]
   205→        yte = yte_full[mask_te]
   206→        m = Ridge(alpha=1.0, random_state=42)
   207→        m.fit(Xtr, ytr)
   208→        if len(Xte) >= 5 and len(yte) >= 5:
   209→            yp = m.predict(Xte)
   210→            metrics[f"{hz}d"] = _metrics(yte, yp)
   211→        else:
   212→            metrics[f"{hz}d"] = {"mse": 0.0, "mae": 0.0, "r2": 0.0}
   213→        models[hz] = m
   214→
   215→    # Per-stock predictions
   216→    preds = {}
   217→    for s in STOCKS:
   218→        sdf = df[df["symbol"] == s].sort_values("date").reset_index(drop=True)
   219→        if len(sdf) < LAG + 21:
   220→            continue
   221→        cp = float(sdf.iloc[-1]["close"])
   222→        rets = sdf["return_1d"].values
   223→        toks = tokenize(rets)
   224→        X = toks[-LAG:].reshape(1, -1)
   225→        r5 = float(models[5].predict(X)[0])
   226→        r10 = float(models[10].predict(X)[0])
   227→        r21 = float(models[21].predict(X)[0])
   228→        sl = np.min(sdf["low"].tail(14).values)
   229→        preds[s] = _pred_record(cp, r5, r10, r21, sl)
   230→
   231→    print(f"    Done — {len(preds)} stocks, R²(5d)={metrics['5d']['r2']:.3f}")
   232→    return {"trained": True, "version": "2.0.0-foundation-analog",
   233→            "metrics": metrics, "predictions": preds,
   234→            "analog": "quantile-binned tokenize-then-forecast (T5-style)"}
   235→

   237→# ────────────────────────────────────────────────────────────────────
   238→# 2. Lag-Llama — lag-feature autoregression with gradient boosting
   239→# ────────────────────────────────────────────────────────────────────
   240→def train_lag_llama(df):
   241→    """Lag-Llama analog: lag-based attention → use lagged features + GBR."""
   242→    print("  Training Lag-Llama [lag-feature GBR analog]...")
   243→    targets = ["target_5d", "target_10d", "target_21d"]
   244→    # ML integrity (Task 2): chronological_split replaces the leaky global row split.
   245→    tr0, va0, te0 = chronological_split(df)
   246→    train_end = tr0["date"].max(); val_end = va0["date"].max()
   247→
   248→    # Build lag features: 7 lags of return_1d, plus RSI/MACD lags
   249→    LAGS = [1, 2, 3, 5, 7, 14, 21]
   250→    base_cols = ["return_1d", "rsi_14", "macd", "macd_signal"]
   251→    lag_features = []
   252→    for col in base_cols:
   253→        for L in LAGS:
   254→            lag_col = f"{col}_lag_{L}"
   255→            df[lag_col] = df.groupby("symbol")[col].shift(L).fillna(0)
   256→            lag_features.append(lag_col)
   257→
   258→    # Re-pull chronological partitions from the updated df (date-based masks).
   259→    tr = df[df["date"] <= train_end].copy()
   260→    te = df[df["date"] > val_end].copy()
   261→
   262→    metrics = {}; models = {}
   263→    for tc in targets:
   264→        hz = int(tc.split("_")[1].replace("d", ""))
   265→        # ML integrity (Phase A1): per-horizon drop — drop ONLY rows whose
   266→        # THIS-horizon target is NaN (preserves shorter-horizon rows).
   267→        tr_h = tr.dropna(subset=[tc]); te_h = te.dropna(subset=[tc])
   268→        Xtr = tr_h[lag_features].values; Xte = te_h[lag_features].values
   269→        ytr = tr_h[tc].values; yte = te_h[tc].values
   270→        m = GradientBoostingRegressor(n_estimators=80, max_depth=3,
   271→                                       learning_rate=0.05, random_state=42)
   272→        m.fit(Xtr, ytr)
   273→        yp = m.predict(Xte)
   274→        metrics[f"{hz}d"] = _metrics(yte, yp)
   275→        models[hz] = m
   276→
   277→    # Per-stock predictions
   278→    preds = {}
   279→    for s in STOCKS:
   280→        sdf = df[df["symbol"] == s].sort_values("date").reset_index(drop=True)
   281→        if len(sdf) < 30:
   282→            continue
   283→        cp = float(sdf.iloc[-1]["close"])
   284→        X = sdf[lag_features].iloc[-1:].values
   285→        r5 = float(models[5].predict(X)[0])
   286→        r10 = float(models[10].predict(X)[0])
   287→        r21 = float(models[21].predict(X)[0])
   288→        sl = np.min(sdf["low"].tail(14).values)
   289→        preds[s] = _pred_record(cp, r5, r10, r21, sl)
   290→
   291→    print(f"    Done — {len(preds)} stocks, R²(5d)={metrics['5d']['r2']:.3f}")
   292→    return {"trained": True, "version": "2.0.0-foundation-analog",
   293→            "metrics": metrics, "predictions": preds,
   294→            "analog": "lag-feature gradient boosting (Lag-Llama lag-attention)"}
   295→

   297→# ────────────────────────────────────────────────────────────────────
   298→# 3. Moirai (Salesforce) — any-variate multivariate regression
   299→# ────────────────────────────────────────────────────────────────────
   300→def train_moirai(df):
   301→    """Moirai analog: any-variate forecasting → multivariate regression that
   302→    treats all features jointly with shared weights (any-variate analog)."""
   303→    print("  Training Moirai (Salesforce) [any-variate analog]...")
   304→    targets = ["target_5d", "target_10d", "target_21d"]
   305→    # ML integrity (Task 2): chronological_split replaces the leaky global row
   306→    # split. The scaler fit is ALREADY correct (fit on tr, transform te) — only
   307→    # the underlying split was leaky; the per-horizon dropna is new (Phase A1).
   308→    tr, va, te = chronological_split(df)
   309→
   310→    # Any-variate: standardize all features jointly (fit on train only).
   311→    sc = StandardScaler()
   312→    sc.fit(tr[FC].values)
   313→
   314→    metrics = {}; models = {}
   315→    for tc in targets:
   316→        hz = int(tc.split("_")[1].replace("d", ""))
   317→        # ML integrity (Phase A1): per-horizon drop — re-pull X per-horizon
   318→        # (row counts differ across horizons after dropna); use the train-fit
   319→        # scaler via transform (no refit).
   320→        tr_h = tr.dropna(subset=[tc]); te_h = te.dropna(subset=[tc])
   321→        Xtr = sc.transform(tr_h[FC].values)
   322→        Xte = sc.transform(te_h[FC].values)
   323→        ytr = tr_h[tc].values; yte = te_h[tc].values
   324→        # Multivariate (any-variate) linear regression with strong regularization
   325→        m = Ridge(alpha=10.0, random_state=42)
   326→        m.fit(Xtr, ytr)
   327→        yp = m.predict(Xte)
   328→        metrics[f"{hz}d"] = _metrics(yte, yp)
   329→        models[hz] = (m, sc)
   330→
   331→    # Per-stock predictions
   332→    preds = {}
   333→    for s in STOCKS:
   334→        sdf = df[df["symbol"] == s].sort_values("date").reset_index(drop=True)
   335→        if len(sdf) < 30:
   336→            continue
   337→        cp = float(sdf.iloc[-1]["close"])
   338→        X = sdf[FC].iloc[-1:].values
   339→        Xs = sc.transform(X)
   340→        r5 = float(models[5][0].predict(Xs)[0])
   341→        r10 = float(models[10][0].predict(Xs)[0])
   342→        r21 = float(models[21][0].predict(Xs)[0])
   343→        sl = np.min(sdf["low"].tail(14).values)
   344→        preds[s] = _pred_record(cp, r5, r10, r21, sl)
   345→
   346→    print(f"    Done — {len(preds)} stocks, R²(5d)={metrics['5d']['r2']:.3f}")
   347→    return {"trained": True, "version": "2.0.0-foundation-analog",
   348→            "metrics": metrics, "predictions": preds,
   349→            "analog": "any-variate multivariate ridge (Moirai any-variate)"}
   350→

   352→# ────────────────────────────────────────────────────────────────────
   353→# 4. Time-LLM — reprogramming via SVD projection of base predictions
   354→# ────────────────────────────────────────────────────────────────────
   355→def train_time_llm(df, existing_models):
   356→    """Time-LLM analog: reprogramming of pretrained source via projection.
   357→    Here we project existing model predictions through an SVD basis to mimic
   358→    Time-LLM's reprogramming layer."""
   359→    print("  Training Time-LLM [SVD-reprogramming analog]...")
   360→    targets = ["target_5d", "target_10d", "target_21d"]
   361→    # ML integrity (Task 2): chronological_split replaces the leaky global row
   362→    # split. Compute train_end/val_end dates, then apply as row masks on the
   363→    # original (stock-concat-order) df so src and tr/te stay aligned.
   364→    tr0, va0, te0 = chronological_split(df)
   365→    train_end = tr0["date"].max(); val_end = va0["date"].max()
   366→    train_mask = (df["date"] <= train_end).values
   367→    test_mask = (df["date"] > val_end).values
   368→    tr = df[train_mask].copy(); te = df[test_mask].copy()
   369→
   370→    # Collect base model predictions for each (symbol, date) row
   371→    base_model_names = [n for n, m in existing_models.items()
   372→                        if m.get("trained") and n not in
   373→                        ("Chronos (Amazon)", "Lag-Llama",
   374→                         "Moirai (Salesforce)", "Time-LLM")]
   375→    print(f"    Using {len(base_model_names)} base models as 'pretrained source'")
   376→
   377→    # For each base model, predict per-row 5d/10d/21d return using their per-stock
   378→    # latest predictions — but we need a per-row version. As a lightweight stand-in,
   379→    # we use the base model's avgReturn applied per-stock as a constant prediction.
   380→    # This is the "pretrained source activations" in Time-LLM terms.
   381→
   382→    # Build a matrix of base-model predicted returns per row (reprogramming source)
   383→    base_returns = {}
   384→    for bn in base_model_names:
   385→        m = existing_models[bn]
   386→        if not m.get("trained"):
   387→            continue
   388→        per_stock = {s: m.get("predictions", {}).get(s, {}).get("avgReturn", 0)
   389→                     for s in STOCKS}
   390→        base_returns[bn] = df["symbol"].map(per_stock).fillna(0).values
   391→
   392→    if not base_returns:
   393→        return {"trained": False, "error": "No base models to reprogram"}
   394→
   395→    # Stack: shape (n_samples, n_base_models)
   396→    src = np.column_stack([base_returns[bn] for bn in base_model_names])
   397→    # SVD-based reprogramming: project source to a lower-dim space then linearly map to target
   398→    # For training, fit SVD on training source, then ridge regression on the projected source
   399→    src_tr = src[train_mask]; src_te = src[test_mask]
   400→    # Cap components to min(n_base, n_train_samples)
   401→    n_comp = min(len(base_model_names), 5, src_tr.shape[0] - 1)
   402→    n_comp = max(1, n_comp)
   403→    # SVD via truncated SVD on the source matrix
   404→    U_tr, S_tr, Vt_tr = np.linalg.svd(src_tr, full_matrices=False)
   405→    V_k = Vt_tr[:n_comp]  # projection basis (n_comp × n_base)
   406→    src_tr_proj = src_tr @ V_k.T  # (n_train × n_comp)
   407→    src_te_proj = src_te @ V_k.T
   408→
   409→    metrics = {}; models = {}
   410→    for tc in targets:
   411→        hz = int(tc.split("_")[1].replace("d", ""))
   412→        # ML integrity (Phase A1): per-horizon drop — mask rows whose THIS-horizon
   413→        # target is NaN/inf. Keeps src_tr_proj and ytr aligned (same row order).
   414→        ytr_full = tr[tc].values
   415→        yte_full = te[tc].values
   416→        mask_tr = np.isfinite(ytr_full)
   417→        mask_te = np.isfinite(yte_full)
   418→        ytr = ytr_full[mask_tr]
   419→        yte = yte_full[mask_te]
   420→        src_tr_proj_h = src_tr_proj[mask_tr]
   421→        src_te_proj_h = src_te_proj[mask_te]
   422→        if src_tr_proj_h.shape[0] < 5 or src_tr_proj_h.shape[1] < 1:
   423→            metrics[f"{hz}d"] = {"mse": 0.0, "mae": 0.0, "r2": 0.0}
   424→            models[hz] = (V_k, Ridge(alpha=1.0).fit(src_tr_proj_h, ytr))
   425→            continue
   426→        m = Ridge(alpha=1.0, random_state=42)
   427→        m.fit(src_tr_proj_h, ytr)
   428→        if len(yte) > 0:
   429→            yp = m.predict(src_te_proj_h)
   430→            metrics[f"{hz}d"] = _metrics(yte, yp)
   431→        else:
   432→            metrics[f"{hz}d"] = {"mse": 0.0, "mae": 0.0, "r2": 0.0}
   433→        models[hz] = (V_k, m)
   434→
   435→    # Per-stock predictions: take base-model avgReturns for the latest row per stock,
   436→    # project, then predict
   437→    preds = {}
   438→    for s in STOCKS:
   439→        sdf = df[df["symbol"] == s].sort_values("date").reset_index(drop=True)
   440→        if len(sdf) < 30:
   441→            continue
   442→        cp = float(sdf.iloc[-1]["close"])
   443→        # Build source vector for this stock from base models
   444→        src_v = np.array([existing_models[bn].get("predictions", {}).get(s, {}).get("avgReturn", 0)
   445→                          for bn in base_model_names])
   446→        src_proj = src_v @ V_k.T  # (n_comp,)
   447→        r5 = float(models[5][1].predict(src_proj.reshape(1, -1))[0])
   448→        r10 = float(models[10][1].predict(src_proj.reshape(1, -1))[0])
   449→        r21 = float(models[21][1].predict(src_proj.reshape(1, -1))[0])
   450→        sl = np.min(sdf["low"].tail(14).values)
   451→        preds[s] = _pred_record(cp, r5, r10, r21, sl)
   452→
   453→    print(f"    Done — {len(preds)} stocks, R²(5d)={metrics['5d']['r2']:.3f}")
   454→    return {"trained": True, "version": "2.0.0-foundation-analog",
   455→            "metrics": metrics, "predictions": preds,
   456→            "analog": "SVD-reprogrammed base-prediction ensemble (Time-LLM reprogramming)"}

   460→# Backtest a single model on the 20% test set (matches train_all_with_backtesting.py)
   461→# ────────────────────────────────────────────────────────────────────
   462→def backtest_model(df, model_pred):
   463→    """Backtest a single model on the 20% test set (matches train_all_with_backtesting.py).
   464→
   465→    ML-integrity note (Phase B Pattern #10): SIMPLIFIED backtest — applies
   466→    the model's LATEST per-stock signal to every historical test row in the
   467→    chronological test partition; NOT a walk-forward re-prediction. Test
   468→    partition is date-based chronological thanks to chronological_split.
   469→    """
   470→    _tr, _va, te = chronological_split(df); te = te.copy()
   471→    buys = sells = cb_ = cs = 0; rets = []
   472→    for s in STOCKS:
   473→        ste = te[te["symbol"] == s].sort_values("date").reset_index(drop=True)
   474→        if len(ste) < 21:
   475→            continue
   476→        pr = model_pred.get(s, {})
   477→        sig = pr.get("signal", "HOLD")
   478→        # Use the model's signal applied to all test rows for this stock
   479→        for i in range(0, len(ste) - 21, 5):
   480→            ar = (ste.iloc[min(i + 21, len(ste) - 1)]["close"] - ste.iloc[i]["close"]) / ste.iloc[i]["close"]
   481→            if sig == "BUY":
   482→                buys += 1; rets.append(ar); cb_ += ar > 0
   483→            elif sig == "SELL":
   484→                sells += 1; rets.append(-ar); cs += ar < 0
   485→    tt = buys + sells
   486→    wr = (cb_ + cs) / max(tt, 1) * 100
   487→    ar_ = float(np.mean(rets)) * 100 if rets else 0
   488→    return {
   489→        "totalTrades": int(tt), "buySignals": int(buys), "sellSignals": int(sells),
   490→        "correctBuys": int(cb_), "correctSells": int(cs),
   491→        "winRate": round(float(wr), 1),
   492→        "avgReturnPct": round(float(ar_), 2),
   493→        "maxReturnPct": round(float(max(rets) * 100 if rets else 0), 2),
   494→        "minReturnPct": round(float(min(rets) * 100 if rets else 0), 2),
   495→        "stars": int(min(5, max(1, int(wr / 20) + 1))),
   496→    }
