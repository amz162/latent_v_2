    28→# ─── Features (reused) ───
    29→def compute_rsi(p,n=14):
    30→ d=p.diff();g=d.where(d>0,0);l=-d.where(d<0,0);ag=g.ewm(alpha=1/n,min_periods=n).mean();al=l.ewm(alpha=1/n,min_periods=n).mean()
    31→ return(100-100/(1+ag/al.replace(0,np.nan))).fillna(50)
    32→def compute_macd(p):
    33→ e12=p.ewm(span=12,adjust=False).mean();e26=p.ewm(span=26,adjust=False).mean();m=e12-e26;s=m.ewm(span=9,adjust=False).mean();return m,s,m-s
    34→def compute_bb(p,w=20):
    35→ sma=p.rolling(w).mean();sd=p.rolling(w).std();return((sma+2*sd-(sma-2*sd))/sma.replace(0,1)).fillna(0),((p-(sma-2*sd))/(4*sd).replace(0,np.nan)).fillna(0.5)
    36→def compute_atr(h,l,c,n=14):
    37→ tr=pd.concat([(h-l),(h-c.shift()).abs(),(l-c.shift()).abs()],axis=1).max(axis=1);return tr.rolling(n).mean().fillna(0)
    38→def compute_obv(c,v):
    39→ s=c.diff().apply(lambda x:1 if x>0 else(-1 if x<0 else 0));return(s*v).cumsum()

   129→def train_kalman(sdf):
   130→ cp=float(sdf.iloc[-1]["close"]);sl_low=float(sdf["low"].tail(14).min())
   131→ c=sdf["close"].tail(50).values
   132→ # Simple Kalman-like filter: exponential smoothing + trend
   133→ alpha=0.3;state=c[0];trend=0
   134→ for i in range(1,len(c)):
   135→  pred=state+trend;state=alpha*c[i]+(1-alpha)*pred;trend=alpha*(c[i]-state)+(1-alpha)*trend
   136→ p5=state+5*trend;p10=state+10*trend;p21=state+21*trend
   137→ r5=(p5-cp)/cp;r10=(p10-cp)/cp;r21=(p21-cp)/cp
   138→ return gen_pred(cp,r5,r10,r21,sl_low,0.02*cp)

   254→  "currentPrice":round(cp,2),"entry":round(cp,2),"tp1":round(cp+atr,2),"tp2":round(cp+2*atr,2),
   255→  "tp3":round(cp+3*atr,2),"sl":round(sl_low,2),"avgReturn":0.0,"gian_levels":[s1,r1]}
   256→def gian_extender(sdf):
   257→ cp=float(sdf.iloc[-1]["close"]);sl_low=float(sdf["low"].tail(14).min())
   258→ atr=float(sdf["atr_14"].tail(14).mean() or cp*0.02)
   259→ # GANN square-of-9 levels
   260→ levels=[round((n**0.5)**2,2) for n in[1,9,25,49,81,121,169,225,289,361]]
   261→ nearest=min(levels,key=lambda x:abs(x-cp))
   262→ sig="BUY" if cp<nearest*0.98 else"SELL" if cp>nearest*1.02 else"HOLD"
   263→ return{"signal":sig,"direction":"LONG" if sig!="SELL" else"SHORT","confidence":50,
   264→  "currentPrice":round(cp,2),"entry":round(cp,2),"tp1":round(cp+atr,2),"tp2":round(cp+2*atr,2),
   265→  "tp3":round(cp+3*atr,2),"sl":round(sl_low,2),"avgReturn":0.0,"gann_levels":levels}
   266→def m_ensemble(sdf):
   267→ cp=float(sdf.iloc[-1]["close"]);sl_low=float(sdf["low"].tail(14).min())
   268→ atr=float(sdf["atr_14"].tail(14).mean() or cp*0.02)
   269→ # Historical price-touch S/R: find most-touched levels
   270→ highs=sdf["high"].tail(100).values;lows=sdf["low"].tail(100).values
   271→ all_levels=np.concatenate([highs,lows])
   272→ hist=np.histogram(all_levels,bins=20)
   273→ top_idx=np.argsort(hist[0])[-3:];top_levels=hist[1][top_idx]
   274→ nearest=top_levels[np.argmin(np.abs(top_levels-cp))]
   275→ sig="BUY" if cp<nearest*0.99 else"SELL" if cp>nearest*1.01 else"HOLD"
   276→ return{"signal":sig,"direction":"LONG" if sig!="SELL" else"SHORT","confidence":50,
   277→  "currentPrice":round(cp,2),"entry":round(cp,2),"tp1":round(cp+atr,2),"tp2":round(cp+2*atr,2),
   278→  "tp3":round(cp+3*atr,2),"sl":round(sl_low,2),"avgReturn":0.0,"m_levels":top_levels.tolist()}

   329→# ─── Ensemble computation ───
   330→def compute_ensembles(models_out):
   331→ ensembles={}
   332→ all_syms=set()
   333→ for m in models_out.values():
   334→  if isinstance(m,dict) and"predictions"in m:all_syms.update(m["predictions"].keys())
   335→ for sym in all_syms:
   336→  mp=[]
   337→  for mn,md in models_out.items():
   338→   if isinstance(md,dict) and"predictions"in md:
   339→    p=md["predictions"].get(sym)
   340→    if p and p.get("signal")in("BUY","SELL"):mp.append(p)
   341→  if not mp:continue
   342→  bc=sum(1 for p in mp if p["signal"]=="BUY");sc=sum(1 for p in mp if p["signal"]=="SELL")
   343→  hc=len(mp)-bc-sc
   344→  sig="BUY" if bc>sc and bc>hc else"SELL" if sc>bc and sc>hc else"HOLD"
   345→  ar=sum(p.get("avgReturn",0) for p in mp)/len(mp)
   346→  pw_sig="BUY" if ar>0.01 else"SELL" if ar<-0.01 else"HOLD"
   347→  cl_sig="BUY" if bc>=len(mp)/2 else"SELL" if sc>=len(mp)/2 else"HOLD"
   348→  cp=mp[0].get("entry",0)
   349→  sls=[p["sl"] for p in mp if p.get("sl",0)>0]
   350→  tp1s=[p["tp1"] for p in mp if p.get("tp1",0)>0]
   351→  tp2s=[p["tp2"] for p in mp if p.get("tp2",0)>0]
   352→  tp3s=[p["tp3"] for p in mp if p.get("tp3",0)>0]
   353→  sl=np.mean(sls) if sls else cp*0.97
   354→  tp1=np.mean(tp1s) if tp1s else cp*1.03
   355→  tp2=np.mean(tp2s) if tp2s else cp*1.06
   356→  tp3=np.mean(tp3s) if tp3s else cp*1.09
   357→  for en,es in[("majority_vote",sig),("probability_weighted",pw_sig),("cl_ensemble",cl_sig),("cl_extender",cl_sig)]:
   358→   if es not in("BUY","SELL"):continue
   359→   ensembles.setdefault(en,{})
   360→   ensembles[en][sym]={"signal":es,"direction":"LONG" if es!="SELL" else"SHORT",
   361→    "confidence":min(100,int(abs(ar)*1000)),"currentPrice":round(cp,2),"entry":round(cp,2),
   362→    "tp1":round(float(tp1),2),"tp2":round(float(tp2),2),"tp3":round(float(tp3),2),
   363→    "sl":round(float(sl),2),"avgReturn":round(float(ar),4)}
   364→ return ensembles
   365→# ─── Backtest ───
   366→def backtest(preds_dict):
   367→ """Compute predicted-positive-rate backtest metrics.
   368→
   369→ ML-integrity note (Phase B Pattern #11): this is NOT a realized-outcome
   370→ backtest. `wins = sum(1 for p in preds.values() if p.get('avgReturn',0)>0)`
   371→ uses the model's PREDICTED avgReturn being positive — i.e. the predicted-
   372→ positive rate, NOT whether a trade taken at the prediction date actually
   373→ made money. Documented as a known limitation; for honest walk-forward
   374→ realized-outcome backtesting see scripts/walk_forward_backtesting.py.
   375→ """
   376→ results={}
   377→ for name,model_preds in preds_dict.items():
   378→  if not model_preds:continue
   379→  total=len(model_preds);wins=sum(1 for p in model_preds.values() if p.get("avgReturn",0)>0)
   380→  returns=[p.get("avgReturn",0) for p in model_preds.values()]
   381→  results[name]={"totalTrades":total,"wins":wins,"winRate":round(wins/total*100,1) if total else 0,
   382→   "avgReturnPct":round(float(np.mean(returns))*100,2) if returns else 0,
   383→   "maxReturnPct":round(float(max(returns))*100,2) if returns else 0,
   384→   "minReturnPct":round(float(min(returns))*100,2) if returns else 0}
   385→ return results
