    44→function fmtNum(n: number): string {
    45→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}B`;
    46→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}M`;
    47→  return n.toFixed(0);
    48→}
    49→
    50→function fyOf(period: string): string {
    51→  const m = period.match(/FY\s?(\d+)/i);
    52→  return m ? m[1] : period.slice(-2);
    53→}

   266→function StockDeepDive({
   267→  results,
   268→  symbol,
   269→  company,
   270→  onBack,
   271→}: {
   272→  results: QuarterlyResult[];
   273→  symbol: string;
   274→  company: string;
   275→  onBack: () => void;
   276→}) {
   277→  const patterns = detectPatterns(results);
   278→  const quarters = generateQuarterlySummary(results, symbol);
   279→  const yearsBetween =
   280→    results.length >= 2
   281→      ? (new Date(results[results.length - 1].periodEnd).getTime() -
   282→          new Date(results[0].periodEnd).getTime()) /
   283→        (365 * 24 * 60 * 60 * 1000)
   284→      : 0;
   285→  const revCAGR = computeCAGR(results[0].metrics.revenue, results[results.length - 1].metrics.revenue, yearsBetween);
   286→  const profCAGR = computeCAGR(
   287→    results[0].metrics.profitAfterTax,
   288→    results[results.length - 1].metrics.profitAfterTax,
   289→    yearsBetween
   290→  );
   291→  const epsCAGR = computeCAGR(results[0].metrics.eps, results[results.length - 1].metrics.eps, yearsBetween);
   292→  const totalDPS = results.reduce((s, r) => s + r.metrics.dividendPerShare, 0);
   293→  const volatility = computeStdDev(results.map((r) => r.metrics.epsYoYGrowth));
   294→  const revVolatility = computeStdDev(results.map((r) => r.metrics.revenueYoYGrowth));
   295→  const marginSlope = computeSlope(results.map((r) => r.metrics.netMargin));
   296→
   297→  const stats: { label: string; value: string; sub: string; tone?: "pos" | "neg" | "neutral"; explanation: string }[] = [
   298→    { label: "Revenue CAGR", value: `${revCAGR.toFixed(1)}%`, sub: `${yearsBetween.toFixed(1)}y compound`, tone: revCAGR >= 0 ? "pos" : "neg", explanation: "How fast sales grew per year, on average, over the entire period" },
   299→    { label: "Profit CAGR", value: `${profCAGR.toFixed(1)}%`, sub: "PAT growth", tone: profCAGR >= 0 ? "pos" : "neg", explanation: "How fast profit grew per year, on average" },
   300→    { label: "EPS CAGR", value: `${epsCAGR.toFixed(1)}%`, sub: "per-share profit", tone: epsCAGR >= 0 ? "pos" : "neg", explanation: "How fast per-share profit grew per year" },
   301→    { label: "Total DPS", value: `Rs. ${totalDPS.toFixed(2)}`, sub: `${results.length} quarters`, tone: "neutral", explanation: "Total cash dividends paid per share over the entire period" },
   302→    { label: "EPS volatility", value: `${volatility.toFixed(1)}pp`, sub: "EPS YoY std dev", tone: volatility > 15 ? "neg" : "neutral", explanation: "How unpredictable the profit growth is. Higher = more risky/uncertain" },
   303→    { label: "Revenue volatility", value: `${revVolatility.toFixed(1)}pp`, sub: "Rev YoY std dev", tone: revVolatility > 15 ? "neg" : "neutral", explanation: "How unpredictable the sales growth is" },
   304→    { label: "Margin trend slope", value: `${marginSlope >= 0 ? "+" : ""}${marginSlope.toFixed(2)}pp/q`, sub: "net margin trend", tone: marginSlope >= 0 ? "pos" : "neg", explanation: "Whether profit margin is improving (positive) or declining (negative) over time" },
   305→    { label: "Patterns detected", value: `${patterns.length}`, sub: "fundamental signals", tone: "neutral", explanation: "Number of fundamental patterns found in the data" },
   306→  ];
   307→
   308→  return (
   309→    <div className="space-y-4">
   310→      {/* Header with back button */}
   311→      <Card className="bg-[#131829] border-white/[0.08]">
   312→        <CardContent className="p-4">
   313→          <div className="flex items-start gap-3">
   314→            <Button
   315→              variant="outline"
   316→              size="sm"
   317→              onClick={onBack}
   318→              className="border-white/[0.12] text-[#94A3B8] hover:text-[#F4F7FB] shrink-0"
   319→            >
   320→              <ArrowLeft className="h-3.5 w-3.5 mr-1.5" /> Back
   321→            </Button>
   322→            <div className="flex-1 min-w-0">
   323→              <div className="flex items-baseline gap-2 flex-wrap">
   324→                <h2 className="text-xl font-bold text-[#F4F7FB]">{symbol}</h2>
   325→                <span className="text-sm text-[#94A3B8] truncate">{company}</span>
   326→              </div>
   327→              <p className="text-xs text-[#64748B] mt-1">
   328→                Deep dive on {results.length} quarters ·{" "}
   329→                {new Date(results[0].periodEnd).toLocaleDateString("en-PK", { year: "numeric", month: "short" })} →{" "}
   330→                {new Date(results[results.length - 1].periodEnd).toLocaleDateString("en-PK", { year: "numeric", month: "short" })}
   331→              </p>
   332→            </div>
   333→          </div>
   334→        </CardContent>
   335→      </Card>
   336→
   337→      {/* 8 summary stats */}
   338→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   339→        {stats.map((s, i) => (
   340→          <SummaryStat key={i} label={s.label} value={s.value} sub={s.sub} tone={s.tone} explanation={s.explanation} />
   341→        ))}
   342→      </div>
   343→
   344→      {/* Multi-line chart */}
   345→      <Card className="bg-[#131829] border-white/[0.08]">
   346→        <CardHeader>
   347→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   348→            <BarChart3 className="h-4 w-4 text-[#00E676]" />
   349→            Revenue · Profit · EPS trend (normalized 0–100)
   350→          </CardTitle>
   351→        </CardHeader>
   352→        <CardContent>
   353→          <MultiLineChart results={results} />
   354→        </CardContent>
   355→      </Card>
   356→
   357→      {/* Quarterly seasonal pattern */}
   358→      <Card className="bg-[#131829] border-white/[0.08]">
   359→        <CardHeader>
   360→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   361→            <Calendar className="h-4 w-4 text-[#00E676]" />
   362→            Quarterly seasonal pattern
   363→          </CardTitle>
   364→        </CardHeader>
   365→        <CardContent>
   366→          {quarters.length === 0 ? (
   367→            <p className="text-xs text-[#64748B]">
   368→              Need at least 4 quarters (ideally spanning multiple FYs) to compute seasonality.
   369→            </p>

   382→        <CardHeader>
   383→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   384→            <TrendingUp className="h-4 w-4 text-[#00E676]" />
   385→            Revenue by quarter (PKR, color-coded by FY)
   386→          </CardTitle>
   387→        </CardHeader>
   388→        <CardContent>
   389→          <RevenueBars results={results} />
   390→        </CardContent>
   391→      </Card>
   392→
   393→      {/* EPS bars + net margin overlay */}
   394→      <Card className="bg-[#131829] border-white/[0.08]">
   395→        <CardHeader>
   396→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">

   408→        <CardHeader>
   409→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">

   424→            Detected patterns ({patterns.length})
   425→          </CardTitle>
   426→        </CardHeader>
   427→        <CardContent>
   428→          {patterns.length === 0 ? (
   429→            <p className="text-xs text-[#64748B]">
   430→              No strong patterns detected yet — need more quarters or larger moves to trigger detection rules.
   431→            </p>
   432→          ) : (

   434→              {patterns.map((p, i) => (
   435→                <PatternRow key={i} pattern={p} />

   437→            </div>

   439→        </CardContent>

   442→      {/* Full quarterly history table */}
   443→      <Card className="bg-[#131829] border-white/[0.08]">
   444→        <CardHeader>
   445→          <CardTitle className="text-base text-[#F4F7FB]">Full quarterly history</CardTitle>

   642→// ---- Pure-SVG multi-line chart: revenue (green), profit (blue dashed), EPS (yellow dotted)
   643→function MultiLineChart({ results }: { results: QuarterlyResult[] }) {
   644→  const W = 800;
   645→  const H = 280;
   646→  const P = { top: 20, right: 30, bottom: 36, left: 40 };
   647→  const innerW = W - P.left - P.right;
   648→  const innerH = H - P.top - P.bottom;
   649→
   650→  const revs = results.map((r) => r.metrics.revenue);
   651→  const profits = results.map((r) => r.metrics.profitAfterTax);
   652→  const epss = results.map((r) => r.metrics.eps);
   653→
   654→  const rangeOf = (arr: number[]) => {
   655→    const mn = Math.min(...arr);
   656→    const mx = Math.max(...arr);
   657→    return { mn, mx: mx === mn ? mn + 1 : mx };
   658→  };
   659→  const revR = rangeOf(revs);
   660→  const profR = rangeOf(profits);
   661→  const epsR = rangeOf(epss);
   662→
   663→  const norm = (v: number, r: { mn: number; mx: number }) =>
   664→    r.mx === r.mn ? 0.5 : (v - r.mn) / (r.mx - r.mn);
   665→
   666→  const xAt = (i: number) =>
   667→    P.left + (results.length === 1 ? innerW / 2 : (i / (results.length - 1)) * innerW);
   668→  const yAt = (n: number) => P.top + (1 - n) * innerH;
   669→
   670→  const linePath = (vals: number[], r: { mn: number; mx: number }) =>
   671→    vals
   672→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   673→      .join(" ");
   674→
   675→  const areaPath = (vals: number[], r: { mn: number; mx: number }) => {
   676→    if (vals.length === 0) return "";
   677→    const top = vals
   678→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   679→      .join(" ");
   680→    return `${top} L ${xAt(vals.length - 1).toFixed(1)} ${(P.top + innerH).toFixed(1)} L ${xAt(0).toFixed(
   681→      1
   682→    )} ${(P.top + innerH).toFixed(1)} Z`;
   683→  };
   684→
   685→  const gridLines = Array.from({ length: 5 }, (_, i) => P.top + (i / 4) * innerH);
   686→  const labelStep = Math.max(1, Math.ceil(results.length / 8));
   687→
   688→  const lastRev = revs[revs.length - 1];
   689→  const lastProf = profits[profits.length - 1];
   690→  const lastEps = epss[epss.length - 1];
   691→
   692→  return (
   693→    <div className="w-full">
   694→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-auto" preserveAspectRatio="xMidYMid meet">
   695→        <defs>
   696→          <linearGradient id="revFill" x1="0" y1="0" x2="0" y2="1">
   697→            <stop offset="0%" stopColor="#00E676" stopOpacity="0.25" />
   698→            <stop offset="100%" stopColor="#00E676" stopOpacity="0" />
   699→          </linearGradient>
   700→        </defs>
   701→        {/* Grid */}
   702→        {gridLines.map((y, i) => (
   703→          <line
   704→            key={i}
   705→            x1={P.left}
   706→            y1={y}
   707→            x2={P.left + innerW}
   708→            y2={y}
   709→            stroke="#1E2536"
   710→            strokeWidth="1"
   711→            strokeDasharray="2 4"
   712→          />
   713→        ))}
   714→        {/* X-axis labels */}
   715→        {results.map((r, i) =>
   716→          i % labelStep === 0 ? (
   717→            <text
   718→              key={i}
   719→              x={xAt(i)}
   720→              y={H - 8}
   721→              fill="#64748B"
   722→              fontSize="9"
   723→              textAnchor="middle"
   724→              fontFamily="monospace"
   725→            >
   726→              {r.period}
   727→            </text>
   728→          ) : null
   729→        )}
   730→        {/* Filled area under revenue */}
   731→        <path d={areaPath(revs, revR)} fill="url(#revFill)" />
   732→        {/* Revenue line (green solid) */}
   733→        <path d={linePath(revs, revR)} fill="none" stroke="#00E676" strokeWidth="2" />
   734→        {/* Profit line (blue dashed) */}
   735→        <path d={linePath(profits, profR)} fill="none" stroke="#3B82F6" strokeWidth="2" strokeDasharray="6 4" />
   736→        {/* EPS line (yellow dotted) */}
   737→        <path d={linePath(epss, epsR)} fill="none" stroke="#FBBF24" strokeWidth="2" strokeDasharray="2 3" />
   738→        {/* Endpoint markers */}
   739→        {revs.length > 0 && (
   740→          <circle cx={xAt(revs.length - 1)} cy={yAt(norm(lastRev, revR))} r="3.5" fill="#00E676" />
   741→        )}
   742→        {profits.length > 0 && (
   743→          <circle cx={xAt(profits.length - 1)} cy={yAt(norm(lastProf, profR))} r="3.5" fill="#3B82F6" />
   744→        )}
   745→        {epss.length > 0 && (
   746→          <circle cx={xAt(epss.length - 1)} cy={yAt(norm(lastEps, epsR))} r="3.5" fill="#FBBF24" />
   747→        )}
   748→      </svg>
   749→      {/* Legend */}
   750→      <div className="flex items-center gap-4 mt-3 text-[10px] text-[#94A3B8] flex-wrap justify-center">
   751→        <span className="flex items-center gap-1.5">
   752→          <span className="h-0.5 w-4 bg-[#00E676] inline-block" />
   753→          Revenue (PKR)
   754→        </span>
   755→        <span className="flex items-center gap-1.5">
   756→          <span className="h-0.5 w-4 inline-block" style={{ borderBottom: "2px dashed #3B82F6" }} />
   757→          Profit after tax (PKR)
   758→        </span>
   759→        <span className="flex items-center gap-1.5">
   760→          <span className="h-0.5 w-4 inline-block" style={{ borderBottom: "2px dotted #FBBF24" }} />
   761→          EPS (PKR)
   762→        </span>
   763→        <span className="text-[#64748B]">Each line scaled to its own min–max for trend comparison.</span>
   764→      </div>
   765→    </div>
   766→  );
   767→}
