    80→function fmtPct(n: number, digits = 1): string {
    81→  const s = n.toFixed(digits);
    82→  return n > 0 ? `+${s}%` : `${s}%`;
    83→}
    84→
    85→function fmtNum(n: number): string {
    86→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}M`;
    87→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    88→  return n.toFixed(0);
    89→}
    90→
    91→function confidenceOf(occurrences: number): "high" | "medium" | "low" {
    92→  if (occurrences >= 12) return "high";
    93→  if (occurrences >= 6) return "medium";
    94→  return "low";
    95→}

   586→function PatternChart({
   587→  data,
   588→  showEvidence,
   589→  filteredPatterns,
   590→}: {
   591→  data: ApiResult;
   592→  showEvidence: boolean;
   593→  filteredPatterns: PricePattern[];
   594→}) {
   595→  const [hover, setHover] = useState<{ i: number; x: number } | null>(null);
   596→  const svgRef = useRef<SVGSVGElement>(null);
   597→  const chart = data.chart;
   598→
   599→  const model = useMemo(() => {
   600→    if (chart.length < 2) return null;
   601→    const closes = chart.map((c) => c.close);
   602→    let min = Math.min(...closes);
   603→    let max = Math.max(...closes);
   604→    const zones = (data.keyZones || []).slice(0, 5).map((z) => ({ ...z, show: z.high >= min && z.low <= max }));
   605→    for (const z of zones) {
   606→      if (z.show) {
   607→        min = Math.min(min, z.low);
   608→        max = Math.max(max, z.high);
   609→      }
   610→    }
   611→    const pad = (max - min) * 0.05 || 1;
   612→    min -= pad;
   613→    max += pad;
   614→    const plotW = VB_W - PAD.left - PAD.right;
   615→    const plotH = VB_H - PAD.top - PAD.bottom;
   616→    const x = (i: number) => PAD.left + (i / (chart.length - 1)) * plotW;
   617→    const y = (v: number) => PAD.top + (1 - (v - min) / (max - min)) * plotH;
   618→    const line = chart.map((c, i) => `${i === 0 ? "M" : "L"}${x(i).toFixed(2)},${y(c.close).toFixed(2)}`).join(" ");
   619→    const area = `${line} L${x(chart.length - 1).toFixed(2)},${(VB_H - PAD.bottom).toFixed(2)} L${PAD.left},${(VB_H - PAD.bottom).toFixed(2)} Z`;
   620→    const dateIdx = new Map<string, number>();
   621→    chart.forEach((c, i) => dateIdx.set(c.date, i));
   622→    const evidencePatterns = filteredPatterns
   623→      .filter((p) => showEvidence && p.evidence_dates.length > 0)
   624→      .slice(0, 3)
   625→      .map((p) => ({
   626→        pattern: p,
   627→        color: TYPE_META[p.type]?.color ?? "#94A3B8",
   628→        points: p.evidence_dates
   629→          .map((d) => {
   630→            const i = dateIdx.get(d);
   631→            return i !== undefined ? { i, x: x(i), y: y(chart[i].close) } : null;
   632→          })
   633→          .filter((v): v is { i: number; x: number; y: number } => v !== null),
   634→      }))
   635→      .filter((g) => g.points.length > 0);
   636→    const gridValues = [0, 0.25, 0.5, 0.75, 1].map((f) => min + f * (max - min));
   637→    return { min, max, x, y, line, area, zones, evidencePatterns, gridValues, plotW, plotH, dateIdx };
   638→  }, [chart, data.keyZones, filteredPatterns, showEvidence]);
   639→
   640→  if (!model || chart.length < 2) return null;
   641→
   642→  const xLabelIdxs = [0, Math.floor(chart.length * 0.2), Math.floor(chart.length * 0.4), Math.floor(chart.length * 0.6), Math.floor(chart.length * 0.8), chart.length - 1];
   643→
   644→  const onMove = (e: React.MouseEvent<SVGSVGElement>) => {
   645→    const rect = svgRef.current?.getBoundingClientRect();
   646→    if (!rect) return;
   647→    const vx = ((e.clientX - rect.left) / rect.width) * VB_W;
   648→    const frac = (vx - PAD.left) / model.plotW;
   649→    const i = Math.round(frac * (chart.length - 1));
   650→    if (i >= 0 && i < chart.length) setHover({ i, x: model.x(i) });
   651→  };
   652→
   653→  const hoverPoint = hover ? chart[hover.i] : null;
   654→
   655→  return (
   656→    <Card className="bg-[#131829] border-white/[0.06]">
   657→      <CardHeader className="pb-2 px-4 md:px-5 pt-4">
   658→        <CardTitle className="text-sm font-semibold text-[#F4F7FB] flex items-center gap-2 flex-wrap">
   659→          <GitBranch className="h-4 w-4 text-[#00E676]" /> Price history with pattern overlays
   660→          <span className="text-[10px] font-normal text-[#64748B]">daily closes (last ~2 years) · green bands = key price zones · dots = pattern evidence dates</span>
   661→        </CardTitle>
   662→      </CardHeader>
   663→      <CardContent className="px-2 md:px-3 pb-4">
   664→        <div className="overflow-x-auto custom-scroll">
   665→          <div className="min-w-[720px] relative">
   666→            <svg
   667→              ref={svgRef}
   668→              viewBox={`0 0 ${VB_W} ${VB_H}`}
   669→              className="w-full h-auto select-none"
   670→              onMouseMove={onMove}
   671→              onMouseLeave={() => setHover(null)}
   672→              role="img"
   673→              aria-label={`Daily closing price for ${data.symbol} with key price zones and pattern evidence markers`}
   674→            >
   675→              <defs>
   676→                <linearGradient id="ppArea" x1="0" y1="0" x2="0" y2="1">
   677→                  <stop offset="0%" stopColor="#00E676" stopOpacity="0.18" />
   678→                  <stop offset="100%" stopColor="#00E676" stopOpacity="0" />
   679→                </linearGradient>
   680→              </defs>
   681→
   682→              {/* Key price zone bands */}
   683→              {model.zones.filter((z) => z.show).map((z, idx) => {
   684→                const yTop = model.y(z.high);
   685→                const yBot = model.y(z.low);
   686→                return (
   687→                  <g key={`zone-${idx}`}>
   688→                    <rect x={PAD.left} y={yTop} width={VB_W - PAD.left - PAD.right} height={Math.max(2, yBot - yTop)} fill="#00E676" opacity="0.07" />
   689→                    <line x1={PAD.left} x2={VB_W - PAD.right} y1={yTop} y2={yTop} stroke="#00E676" strokeOpacity="0.25" strokeDasharray="4 4" strokeWidth="1" />
   690→                    <line x1={PAD.left} x2={VB_W - PAD.right} y1={yBot} y2={yBot} stroke="#00E676" strokeOpacity="0.25" strokeDasharray="4 4" strokeWidth="1" />
   691→                    <text x={VB_W - PAD.right - 4} y={yTop + 11} textAnchor="end" fontSize="9" fill="#00E676" opacity="0.85">
   692→                      zone Rs.{z.low.toFixed(0)}-{z.high.toFixed(0)} · {z.touches} touches
   693→                    </text>
   694→                  </g>
   695→                );
   696→              })}
   697→
   698→              {/* Grid + y labels */}
   699→              {model.gridValues.map((v, i) => (
   700→                <g key={`grid-${i}`}>
   701→                  <line x1={PAD.left} x2={VB_W - PAD.right} y1={model.y(v)} y2={model.y(v)} stroke="rgba(255,255,255,0.05)" strokeWidth="1" />
   702→                  <text x={PAD.left - 6} y={model.y(v) + 3} textAnchor="end" fontSize="10" fill="#64748B">
   703→                    {v.toFixed(v >= 1000 ? 0 : 1)}
   704→                  </text>
   705→                </g>
   706→              ))}
   707→
   708→              {/* Price area + line */}
   709→              <path d={model.area} fill="url(#ppArea)" />
   710→              <path d={model.line} fill="none" stroke="#00E676" strokeWidth="1.6" strokeLinejoin="round" />
   711→
   712→              {/* Evidence markers */}
   713→              {model.evidencePatterns.map((g, gi) => (
   714→                <g key={`ev-${gi}`}>
   715→                  {g.points.map((pt, pi) => (
   716→                    <circle key={pi} cx={pt.x} cy={pt.y} r="3.2" fill={g.color} stroke="#0A0E1A" strokeWidth="1" opacity="0.9" />

   721→              {/* X labels */}

   723→                <text key={`x-${i}`} x={model.x(i)} y={VB_H - 8} textAnchor="middle" fontSize="10" fill="#64748B">

   728→              {/* Hover guide */}

   730→                <g>
   731→                  <line x1={hover.x} x2={hover.x} y1={PAD.top} y2={VB_H - PAD.bottom} stroke="rgba(255,255,255,0.25)" strokeWidth="1" />
   732→                  <circle cx={hover.x} cy={model.y(chart[hover.i].close)} r="3.5" fill="#F4F7FB" stroke="#0A0E1A" strokeWidth="1" />
   733→                </g>

   735→            </svg>
   736→
   737→            {hover && hoverPoint && (
   738→              <div
   739→                className="absolute pointer-events-none bg-[#0A0E1A] border border-white/[0.12] rounded-md px-2.5 py-1.5 text-[10px] shadow-xl z-10"
