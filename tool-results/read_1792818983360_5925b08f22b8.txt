    86→function smaSeries(values: number[], n: number): (number | null)[] {
    87→  const out: (number | null)[] = [];
    88→  let sum = 0;
    89→  for (let i = 0; i < values.length; i++) {
    90→    sum += values[i];
    91→    if (i >= n) sum -= values[i - n];
    92→    out.push(i >= n - 1 ? sum / n : null);
    93→  }
    94→  return out;
    95→}
    96→
    97→function emaSeries(values: number[], n: number): number[] {
    98→  const k = 2 / (n + 1);
    99→  const out: number[] = [];
   100→  let prev = values[0] ?? 0;
   101→  for (let i = 0; i < values.length; i++) {
   102→    prev = i === 0 ? values[0] : values[i] * k + prev * (1 - k);
   103→    out.push(prev);
   104→  }
   105→  return out;
   106→}
   107→
   108→function rsiSeries(closes: number[], n = 14): number[] {
   109→  const out: number[] = [];
   110→  let avgGain = 0, avgLoss = 0;
   111→  for (let i = 0; i < closes.length; i++) {
   112→    if (i === 0) { out.push(50); continue; }
   113→    const diff = closes[i] - closes[i - 1];
   114→    const gain = Math.max(0, diff);
   115→    const loss = Math.max(0, -diff);
   116→    if (i <= n) {
   117→      avgGain = (avgGain * (i - 1) + gain) / i;
   118→      avgLoss = (avgLoss * (i - 1) + loss) / i;
   119→    } else {
   120→      avgGain = (avgGain * (n - 1) + gain) / n;
   121→      avgLoss = (avgLoss * (n - 1) + loss) / n;
   122→    }
   123→    const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   124→    out.push(avgLoss === 0 ? 100 : 100 - 100 / (1 + rs));
   125→  }
   126→  return out;
   127→}

   129→function macdHistogramSeries(closes: number[]): number[] {
   130→  const ema12 = emaSeries(closes, 12);
   131→  const ema26 = emaSeries(closes, 26);
   132→  const macdLine = closes.map((_, i) => ema12[i] - ema26[i]);
   133→  const signal = emaSeries(macdLine, 9);
   134→  return macdLine.map((m, i) => m - signal[i]);
   135→}

   137→interface Pivot { idx: number; price: number; date: string; kind: "high" | "low"; }
   138→
   139→function findPivots(bars: PriceBar[], w = 3): Pivot[] {
   140→  const pivots: Pivot[] = [];
   141→  for (let i = w; i < bars.length - w; i++) {
   142→    let isHigh = true, isLow = true;
   143→    for (let j = i - w; j <= i + w; j++) {
   144→      if (j === i) continue;
   145→      if (bars[j].high >= bars[i].high) isHigh = false;
   146→      if (bars[j].low <= bars[i].low) isLow = false;
   147→    }
   148→    if (isHigh) pivots.push({ idx: i, price: bars[i].high, date: bars[i].date, kind: "high" });
   149→    if (isLow) pivots.push({ idx: i, price: bars[i].low, date: bars[i].date, kind: "low" });
   150→  }
   151→  return pivots;
   152→}

   154→// Least-squares regression on (index, value)
   155→function linreg(values: number[]): { slope: number; r2: number } {
   156→  const n = values.length;
   157→  if (n < 2) return { slope: 0, r2: 0 };
   158→  const meanX = (n - 1) / 2;
   159→  const meanY = values.reduce((s, v) => s + v, 0) / n;
   160→  let num = 0, den = 0, ssTot = 0;
   161→  for (let i = 0; i < n; i++) {
   162→    num += (i - meanX) * (values[i] - meanY);
   163→    den += (i - meanX) ** 2;
   164→    ssTot += (values[i] - meanY) ** 2;
   165→  }
   166→  const slope = den === 0 ? 0 : num / den;
   167→  const r2 = ssTot === 0 ? 0 : num * num / (den * ssTot);
   168→  return { slope, r2 };
   169→}
   170→
   171→function pct(a: number, b: number): number { return b === 0 ? 0 : ((a - b) / b) * 100; }
   172→function fmtP(v: number, d = 1): string { return `${v >= 0 ? "+" : ""}${v.toFixed(d)}%`; }
   173→function fmtPrice(v: number): string { return `Rs. ${v.toFixed(2)}`; }
   174→function fmtVol(v: number): string {
   175→  if (v >= 1e9) return `${(v / 1e9).toFixed(2)}B`;
   176→  if (v >= 1e6) return `${(v / 1e6).toFixed(2)}M`;
   177→  if (v >= 1e3) return `${(v / 1e3).toFixed(1)}K`;
   178→  return v.toFixed(0);
   179→}
   180→
   181→// Confidence scaled to strength tier — same floors/caps as tab 1.
   182→function conf(strength: PatternStrength, base = 0): number {
   183→  const floor = strength === "Strong" ? 78 : strength === "Moderate" ? 58 : 40;
   184→  const cap = strength === "Strong" ? 95 : strength === "Moderate" ? 78 : 58;
   185→  return Math.max(floor, Math.min(cap, floor + base));
   186→}
   187→
   188→function pickStrength(magnitude: number, modTh: number, strongTh: number): PatternStrength {
   189→  if (Math.abs(magnitude) >= strongTh) return "Strong";
   190→  if (Math.abs(magnitude) >= modTh) return "Moderate";
   191→  return "Weak";
   192→}

   196→export function detectStockPatterns(bars: PriceBar[]): { patterns: StockDetectedPattern[]; stats: StockPatternStats; chart: ChartPoint[] } {
   197→  const closes = bars.map((b) => b.close);
   198→  const vols = bars.map((b) => b.volume);
   199→  const n = bars.length;
   200→  const last = bars[n - 1];
   201→  const price = last.close;
   202→
   203→  // ── indicators ──
   204→  const sma20S = smaSeries(closes, 20);
   205→  const sma50S = smaSeries(closes, 50);
   206→  const sma200S = smaSeries(closes, 200);
   207→  const sma20 = sma20S[n - 1];
   208→  const sma50 = sma50S[n - 1];
   209→  const sma200 = sma200S[n - 1];
   210→  const rsiS = rsiSeries(closes, 14);
   211→  const rsi14 = rsiS[n - 1];
   212→  const macdHistS = macdHistogramSeries(closes);
   213→  const macdHist = macdHistS[n - 1];
   214→  const vol20S = smaSeries(vols, 20);
   215→  const avgVol20 = vol20S[n - 1] ?? 0;
   216→
   217→  // 52-week range
   218→  const lookback52w = Math.max(0, n - 252);
   219→  const last52w = bars.slice(lookback52w);
   220→  const high52w = Math.max(...last52w.map((b) => b.high));
   221→  const low52w = Math.min(...last52w.map((b) => b.low));
   222→  const pctFrom52wHigh = pct(price, high52w);
   223→  const pctFrom52wLow = pct(price, low52w);
   224→
   225→  // Returns over horizons
   226→  const changePct1D = n >= 2 ? pct(price, bars[n - 2].close) : 0;
   227→  const changePct1M = n >= 21 ? pct(price, bars[n - 21].close) : changePct1D;
   228→  const changePct3M = n >= 63 ? pct(price, bars[n - 63].close) : changePct1M;
   229→  const changePct12M = n >= 252 ? pct(price, bars[n - 252].close) : pct(price, closes[0]);
   230→
   231→  // Trend — regression over last 180 sessions, annualised
   232→  const trendBars = Math.min(180, n);
   233→  const trendCloses = closes.slice(n - trendBars);
   234→  const { slope, r2 } = linreg(trendCloses);
   235→  // slope = PKR change per session → annualise to a % of the current price
   236→  const trendSlopeAnnualPct = price > 0 ? ((slope * 252) / price) * 100 : 0;
   237→  const trendLabel: StockPatternStats["trendLabel"] =
   238→    Math.abs(trendSlopeAnnualPct) < 5 || r2 < 0.15 ? "Sideways" : trendSlopeAnnualPct > 0 ? "Uptrend" : "Downtrend";
   239→
   240→  // Volatility — annualised std-dev of daily returns (last 90 sessions)
   241→  const rets: number[] = [];
   242→  for (let i = Math.max(1, n - 90); i < n; i++) rets.push((closes[i] - closes[i - 1]) / closes[i - 1]);
   243→  const meanRet = rets.reduce((s, v) => s + v, 0) / (rets.length || 1);
   244→  const vol = Math.sqrt(rets.reduce((s, v) => s + (v - meanRet) ** 2, 0) / (rets.length || 1)) * Math.sqrt(252) * 100;
   245→
   246→  // Pivots + support/resistance from last 250 bars
   247→  const lb = Math.max(0, n - 250);
   248→  const pivotsAll = findPivots(bars.slice(lb), 3).map((p) => ({ ...p, idx: p.idx + lb }));
   249→  const lows = pivotsAll.filter((p) => p.kind === "low" && p.price <= price);
   250→  const highs = pivotsAll.filter((p) => p.kind === "high" && p.price >= price);
   251→  // nearest support = max pivot low below price; nearest resistance = min pivot high above price
   252→  let support: StockPatternStats["support"] = null;
   253→  for (const p of lows) {
   254→    if (!support || p.price > support.price) support = { price: p.price, touches: 1, lastTouch: p.date };
   255→  }
   256→  if (support) {
   257→    const cluster = lows.filter((p) => Math.abs(pct(p.price, support!.price)) < 2);
   258→    support.touches = cluster.length;
   259→    support.lastTouch = cluster[cluster.length - 1]?.date ?? support.lastTouch;
   260→  }
   261→  let resistance: StockPatternStats["resistance"] = null;
   262→  for (const p of highs) {
   263→    if (!resistance || p.price < resistance.price) resistance = { price: p.price, touches: 1, lastTouch: p.date };
   264→  }
   265→  if (resistance) {
   266→    const cluster = highs.filter((p) => Math.abs(pct(p.price, resistance!.price)) < 2);
   267→    resistance.touches = cluster.length;
   268→    resistance.lastTouch = cluster[cluster.length - 1]?.date ?? resistance.lastTouch;
   269→  }
   270→
   271→  // ── pattern detectors ──
   272→  const patterns: StockDetectedPattern[] = [];
   273→  const push = (p: StockDetectedPattern) => patterns.push(p);
   274→
   275→  // 1. Overall trend (regression slope)
   276→  {
   277→    const strength = pickStrength(trendSlopeAnnualPct, 10, 30);
   278→    if (trendLabel === "Uptrend") {
   279→      push({
   280→        type: "TREND_UP",
   281→        label: `Sustained ${strength.toLowerCase()} uptrend`,
   282→        severity: "positive",
   283→        strength,
   284→        confidence: conf(strength, Math.round(r2 * 15)),
   285→        category: "trend",
   286→        detail: `Over the last ${trendBars} trading sessions (${bars[n - trendBars].date} → ${last.date}), the closing price has been climbing at an annualised rate of ${trendSlopeAnnualPct.toFixed(1)}% with a fit quality (R²) of ${r2.toFixed(2)} — ${r2 > 0.6 ? "a very consistent, orderly climb" : r2 > 0.35 ? "a reasonably steady climb" : "a noisy climb with plenty of wiggles"}. The price went from Rs. ${closes[n - trendBars].toFixed(2)} to Rs. ${price.toFixed(2)} over that span (${fmtP(pct(price, closes[n - trendBars]))}).`,
   287→        supportingData: [
   288→          { label: "Annualised slope", value: `${trendSlopeAnnualPct.toFixed(1)}%` },
   289→          { label: "R² fit", value: r2.toFixed(2) },
   290→          { label: "Period", value: `${trendBars} sessions` },
   291→          { label: "Total move", value: fmtP(pct(price, closes[n - trendBars])) },
   292→        ],
   293→        affectedQuarters: [bars[n - trendBars].date, last.date],
   294→        laymanSummary: `The stock has been steadily getting more expensive over the past ${trendBars} trading days — like climbing stairs rather than a bumpy rollercoaster. On average it has gained about ${trendSlopeAnnualPct.toFixed(0)}% per year at this pace.`,
   295→      });
   296→    } else if (trendLabel === "Downtrend") {
   297→      push({
   298→        type: "TREND_DOWN",
   299→        label: `Sustained ${strength.toLowerCase()} downtrend`,
   300→        severity: "negative",
   301→        strength,
   302→        confidence: conf(strength, Math.round(r2 * 15)),
   303→        category: "trend",
   304→        detail: `Over the last ${trendBars} trading sessions (${bars[n - trendBars].date} → ${last.date}), the closing price has been falling at an annualised rate of ${Math.abs(trendSlopeAnnualPct).toFixed(1)}% with a fit quality (R²) of ${r2.toFixed(2)} — ${r2 > 0.6 ? "a consistent, orderly decline" : "a noisy decline with sharp drops"}. The price went from Rs. ${closes[n - trendBars].toFixed(2)} to Rs. ${price.toFixed(2)} (${fmtP(pct(price, closes[n - trendBars]))}).`,
   305→        supportingData: [
   306→          { label: "Annualised slope", value: `${trendSlopeAnnualPct.toFixed(1)}%` },
   307→          { label: "R² fit", value: r2.toFixed(2) },
   308→          { label: "Period", value: `${trendBars} sessions` },
   309→          { label: "Total move", value: fmtP(pct(price, closes[n - trendBars])) },
   310→        ],
   311→        affectedQuarters: [bars[n - trendBars].date, last.date],
   312→        laymanSummary: `The stock has been steadily losing value over the past ${trendBars} trading days. On average it has lost about ${Math.abs(trendSlopeAnnualPct).toFixed(0)}% per year at this pace — buyers who jumped in earlier in this window are likely sitting on losses.`,
   313→      });
   314→    } else {
   315→      push({
   316→        type: "TREND_SIDEWAYS",
   317→        label: "Sideways / range-bound price action",
   318→        severity: "neutral",
   319→        strength: "Moderate",
   320→        confidence: conf("Moderate", 5),
   321→        category: "trend",
   322→        detail: `Over the last ${trendBars} trading sessions the price has moved sideways (annualised slope ${trendSlopeAnnualPct.toFixed(1)}%, R² ${r2.toFixed(2)} — too noisy to call a trend). The price oscillated between roughly Rs. ${Math.min(...trendCloses).toFixed(2)} and Rs. ${Math.max(...trendCloses).toFixed(2)} without committing to a direction.`,
   323→        supportingData: [
   324→          { label: "Annualised slope", value: `${trendSlopeAnnualPct.toFixed(1)}%` },
   325→          { label: "R² fit", value: r2.toFixed(2) },
   326→          { label: "Range low", value: fmtPrice(Math.min(...trendCloses)) },
   327→          { label: "Range high", value: fmtPrice(Math.max(...trendCloses)) },
   328→        ],
   329→        affectedQuarters: [bars[n - trendBars].date, last.date],
   330→        laymanSummary: `The price is stuck in a range — it goes up a bit, comes back down, and repeats. Neither buyers nor sellers are winning. Range-bound stocks often pick a direction after a breakout, so watch the range edges.`,
   331→      });
   332→    }
   333→  }
   334→
   335→  // 2. Price vs SMA200 (the "long-term line in the sand")
   336→  if (sma200 !== null) {
   337→    const dist = pct(price, sma200);
   338→    const strength = pickStrength(dist, 5, 15);
   339→    if (dist > 1) {
   340→      push({
   341→        type: "ABOVE_SMA200",
   342→        label: "Price trading above 200-day average",
   343→        severity: "positive",
   344→        strength,
   345→        confidence: conf(strength, Math.min(12, Math.abs(dist) / 2)),
   346→        category: "trend",
   347→        detail: `The current price (Rs. ${price.toFixed(2)}) is ${dist.toFixed(1)}% above the 200-day simple moving average (Rs. ${sma200.toFixed(2)}). The 200-day average is the most widely watched long-term trend line — trading above it is a classic sign the long-term trend favours buyers. ${dist > 20 ? "The wide gap suggests the stock is quite extended and may be due for a pullback toward the average." : "The gap is moderate, so the stock is extended but not extreme."}`,
   348→        supportingData: [
   349→          { label: "Current price", value: fmtPrice(price) },
   350→          { label: "SMA200", value: fmtPrice(sma200) },
   351→          { label: "Distance", value: `${dist.toFixed(1)}%` },
   352→        ],
   353→        affectedQuarters: [last.date],
   354→        laymanSummary: `The stock costs more today than its average price over the last year — a sign the long-term direction is upward. The 200-day average is like the "fair value line" long-term investors watch.`,
   355→      });
   356→    } else if (dist < -1) {
   357→      push({
   358→        type: "BELOW_SMA200",
   359→        label: "Price trading below 200-day average",
   360→        severity: "negative",
   361→        strength,
   362→        confidence: conf(strength, Math.min(12, Math.abs(dist) / 2)),
   363→        category: "trend",
   364→        detail: `The current price (Rs. ${price.toFixed(2)}) is ${Math.abs(dist).toFixed(1)}% below the 200-day simple moving average (Rs. ${sma200.toFixed(2)}). Trading below the 200-day average is the classic definition of a long-term downtrend and warns that sellers control the bigger picture. ${Math.abs(dist) > 20 ? "The wide gap below suggests heavy selling pressure." : ""}`,
   365→        supportingData: [
   366→          { label: "Current price", value: fmtPrice(price) },
   367→          { label: "SMA200", value: fmtPrice(sma200) },
   368→          { label: "Distance", value: `${dist.toFixed(1)}%` },
   369→        ],
   370→        affectedQuarters: [last.date],
   371→        laymanSummary: `The stock costs less today than its average price over the last year — a warning that the long-term direction is downward and sellers are in control.`,
   372→      });
   373→    }
   374→  }
   375→
   376→  // 3. Golden / death cross (SMA50 vs SMA200) — detect a cross in the last 30 sessions
   377→  if (sma200 !== null && sma50 !== null && n >= 220) {
   378→    let crossIdx = -1, crossDir: "golden" | "death" | null = null;
   379→    for (let i = n - 30; i < n; i++) {
   380→      if (i < 1 || sma50S[i] === null || sma200S[i] === null || sma50S[i - 1] === null || sma200S[i - 1] === null) continue;
   381→      const wasAbove = (sma50S[i - 1] as number) > (sma200S[i - 1] as number);
   382→      const isAbove = (sma50S[i] as number) > (sma200S[i] as number);
   383→      if (isAbove && !wasAbove) { crossIdx = i; crossDir = "golden"; break; }
   384→      if (!isAbove && wasAbove) { crossIdx = i; crossDir = "death"; break; }
   385→    }
   386→    const crossAge = crossIdx >= 0 ? n - 1 - crossIdx : -1;
   387→    if (crossDir === "golden" && crossIdx >= 0) {
   388→      const strength: PatternStrength = crossAge <= 10 ? "Strong" : "Moderate";
   389→      push({
   390→        type: "GOLDEN_CROSS",
   391→        label: "Golden cross — 50-day average crossed above 200-day",
   392→        severity: "positive",
   393→        strength,
   394→        confidence: conf(strength, 8),
   395→        category: "trend",
   396→        detail: `On ${bars[crossIdx].date} the 50-day moving average (Rs. ${(sma50S[crossIdx] as number).toFixed(2)}) crossed above the 200-day moving average (Rs. ${(sma200S[crossIdx] as number).toFixed(2)}) — a "golden cross". This is one of the most widely watched long-term BUY signals in technical analysis: it says the medium-term momentum has flipped upward relative to the long-term trend. Today the 50-day sits at Rs. ${sma50.toFixed(2)} vs the 200-day at Rs. ${sma200.toFixed(2)} (${fmtP(pct(sma50, sma200))}).`,
   397→        supportingData: [
   398→          { label: "Cross date", value: bars[crossIdx].date },
   399→          { label: "SMA50 now", value: fmtPrice(sma50) },
   400→          { label: "SMA200 now", value: fmtPrice(sma200) },
   401→          { label: "Sessions since", value: `${crossAge}` },
   402→        ],
   403→        affectedQuarters: [bars[crossIdx].date],
