   287→   287→// HistoricalOhlcvChart — loads ALL historical price data from askanalyst
   288→   288→// via /api/psx/ohlcv/[symbol] (tries local cache first, then live fetch up to 10y)
   289→   289→// ────────────────────────────────────────────────────────────────────────────
   290→   290→
   291→   291→interface Candle {
   292→   292→  date: number;
   293→   293→  open: number;
   294→   294→  high: number;
   295→   295→  low: number;
   296→   296→  close: number;
   297→   297→  volume: number;
   298→   298→}
   299→   299→
   300→   300→function HistoricalOhlcvChart({ symbol, height = 360 }: { symbol: string; height?: number }) {
   301→   301→  const [data, setData] = useState<Candle[]>([]);
   302→   302→  const [loading, setLoading] = useState(true);
   303→   303→  const [error, setError] = useState<string | null>(null);
   304→   304→  const [source, setSource] = useState<string>("");
   305→   305→  const [range, setRange] = useState<"1m" | "6m" | "1y" | "5y" | "all">("all");
   306→   306→  const [chartType, setChartType] = useState<"line" | "candle">("line");
   307→   307→  const [timeframe, setTimeframe] = useState<Timeframe>("1d");
   308→   308→  const [chartWidth, setChartWidth] = useState(800);
   309→   309→  const chartContainerRef = useRef<HTMLDivElement>(null);
   310→   310→
   311→   311→  useEffect(() => {
   312→   312→    if (!chartContainerRef.current) return;
   313→   313→    const el = chartContainerRef.current;
   314→   314→    const ro = new ResizeObserver((entries) => {
   315→   315→      for (const e of entries) setChartWidth(Math.max(320, Math.floor(e.contentRect.width)));
   316→   316→    });
   317→   317→    ro.observe(el);
   318→   318→    setChartWidth(Math.max(320, el.clientWidth));
   319→   319→    return () => ro.disconnect();
   320→   320→  }, []);
   321→   321→
   322→   322→  useEffect(() => {
   323→   323→    let cancelled = false;
   324→   324→    setLoading(true);
   325→   325→    setError(null);
   326→   326→    setData([]);
   327→   327→
   328→   328→    fetch(`/api/psx/ohlcv/${encodeURIComponent(symbol)}`)
   329→   329→      .then((r) => {
   330→   330→        if (!r.ok) {
   331→   331→          return r.json().then((err) => {
   332→   332→            throw new Error(err?.error || `Upstream ${r.status}`);
   333→   333→          });
   334→   334→        }
   335→   335→        return r.json();
   336→   336→      })
   337→   337→      .then((d) => {
   338→   338→        if (cancelled) return;
   339→   339→        if (!d || !Array.isArray(d.ohlcv) || d.ohlcv.length === 0) {
   340→   340→          setError("No price data available for this symbol");
   341→   341→          setLoading(false);
   342→   342→          return;
   343→   343→        }
   344→   344→        const candles: Candle[] = d.ohlcv
   345→   345→          .filter((c: any) => c && typeof c.date === "number" && !isNaN(c.close))
   346→   346→          .map((c: any) => ({
   347→   347→            date: c.date,
   348→   348→            open: c.open,
   349→   349→            high: c.high,
   350→   350→            low: c.low,
   351→   351→            close: c.close,
   352→   352→            volume: c.volume || 0,
   353→   353→          }));
   354→   354→        if (cancelled) return;
   355→   355→        if (candles.length === 0) {
   356→   356→          setError("No valid rows in upstream data");
   357→   357→          setLoading(false);
   358→   358→          return;
   359→   359→        }
   360→   360→        setData(candles);
   361→   361→        setSource(d.source || "askanalyst.com.pk");
   362→   362→        setLoading(false);
   363→   363→      })
   364→   364→      .catch((err) => {
   365→   365→        if (cancelled) return;
   366→   366→        console.error("[HistoricalOhlcvChart] fetch error:", err);
   367→   367→        setError(err instanceof Error ? err.message : "Unknown error");
   368→   368→        setLoading(false);
   369→   369→      });
   370→   370→
   371→   371→    return () => {
   372→   372→      cancelled = true;
   373→   373→    };
   374→   374→  }, [symbol]);
   375→   375→
   376→   376→  // Apply range filter to the loaded data
   377→   377→  const filtered = (() => {
   378→   378→    if (data.length === 0) return [];
   379→   379→    if (range === "all") return data;
   380→   380→    const now = Date.now();
   381→   381→    const ms: Record<typeof range, number> = {

   675→   675→function computeSMAAndLevels(candles: Candle[]): (Candle & { sma50?: number; sma200?: number })[] {
   676→   676→  return candles.map((c, i) => {
   677→   677→    const sma50 = i >= 49 ? avg(candles.slice(i - 49, i + 1).map((x) => x.close)) : undefined;
   678→   678→    const sma200 = i >= 199 ? avg(candles.slice(i - 199, i + 1).map((x) => x.close)) : undefined;
   679→   679→    return { ...c, sma50, sma200 };
   680→   680→  });
   681→   681→}
   682→   682→
   683→   683→function avg(nums: number[]): number {
   684→   684→  return nums.reduce((s, n) => s + n, 0) / nums.length;
   685→   685→}
   686→   686→
   687→   687→function findSupportResistance(data: (Candle & { sma50?: number; sma200?: number })[]): { support: number | null; resistance: number | null } {
   688→   688→  if (data.length < 20) return { support: null, resistance: null };
   689→   689→  const recent = data.slice(-60, -5);
   690→   690→  if (recent.length === 0) return { support: null, resistance: null };
   691→   691→  const lows = recent.map((c) => c.low);
   692→   692→  const highs = recent.map((c) => c.high);
   693→   693→  const support = Math.min(...lows);
   694→   694→  const resistance = Math.max(...highs);
   695→   695→  return { support, resistance };
   696→   696→}

   698→   698→// ────────────────────────────────────────────────────────────────────────────
   699→   699→// CandlestickChart — pure SVG candlestick chart rendered from OHLCV data.
   700→   700→// Green bodies for up days (close >= open), red for down days (close < open).
   701→   701→// Sub-samples to a max of ~120 candles to keep SVG rendering performant.
   702→   702→// ────────────────────────────────────────────────────────────────────────────
   703→   703→
   704→   704→function aggregateCandles(candles: Candle[], maxBars: number): Candle[] {
   705→   705→  if (candles.length <= maxBars) return candles;
   706→   706→  const binSize = Math.ceil(candles.length / maxBars);
   707→   707→  const out: Candle[] = [];
   708→   708→  for (let i = 0; i < candles.length; i += binSize) {
   709→   709→    const bin = candles.slice(i, i + binSize);
   710→   710→    if (bin.length === 0) continue;
   711→   711→    out.push({
   712→   712→      date: bin[0].date,
   713→   713→      open: bin[0].open,
   714→   714→      high: Math.max(...bin.map((c) => c.high)),
   715→   715→      low: Math.min(...bin.map((c) => c.low)),
   716→   716→      close: bin[bin.length - 1].close,
   717→   717→      volume: bin.reduce((s, c) => s + (c.volume || 0), 0),
   718→   718→    });
   719→   719→  }
   720→   720→  return out;
   721→   721→}
   722→   722→
   723→   723→function CandlestickChart({
   724→   724→  candles,
   725→   725→  support,
   726→   726→  resistance,
   727→   727→  width,
   728→   728→  height,
   729→   729→}: {
   730→   730→  candles: Candle[];
   731→   731→  support: number | null;
   732→   732→  resistance: number | null;
   733→   733→  width: number;
   734→   734→  height: number;
   735→   735→}) {
   736→   736→  const padding = { top: 12, right: 50, bottom: 22, left: 6 };
   737→   737→  const innerW = Math.max(50, width - padding.left - padding.right);
   738→   738→  const innerH = Math.max(50, height - padding.top - padding.bottom);
   739→   739→
   740→   740→  // Sub-sample to ~120 candles for performance on long histories (10y daily)
   741→   741→  const displayed = aggregateCandles(candles, 120);
   742→   742→
   743→   743→  if (displayed.length === 0) {
   744→   744→    return (
   745→   745→      <div className="w-full flex items-center justify-center text-xs text-[#64748B]" style={{ height }}>
   746→   746→        No data to render
   747→   747→      </div>
   748→   748→    );
   749→   749→  }
   750→   750→
   751→   751→  // Y scale — include support/resistance and SMA range in view
   752→   752→  const lows = displayed.map((c) => c.low);
   753→   753→  const highs = displayed.map((c) => c.high);
   754→   754→  let yMin = Math.min(...lows, support ?? Infinity, resistance ?? Infinity);
   755→   755→  let yMax = Math.max(...highs, support ?? -Infinity, resistance ?? -Infinity);
   756→   756→  if (!isFinite(yMin) || !isFinite(yMax)) {
   757→   757→    yMin = Math.min(...lows);
   758→   758→    yMax = Math.max(...highs);
   759→   759→  }
   760→   760→  const yRange = Math.max(0.0001, yMax - yMin);
   761→   761→  const pad = yRange * 0.06;
   762→   762→  yMin -= pad;
   763→   763→  yMax += pad;
   764→   764→  const yScale = yMax - yMin;
   765→   765→
   766→   766→  const xFor = (i: number) => padding.left + (i + 0.5) * (innerW / displayed.length);
   767→   767→  const yFor = (p: number) => padding.top + (1 - (p - yMin) / yScale) * innerH;
   768→   768→
   769→   769→  const cw = Math.max(1, Math.min(12, (innerW / displayed.length) * 0.65));
   770→   770→
   771→   771→  // 5 horizontal gridlines + price labels on right axis
   772→   772→  const gridLines = [0, 0.25, 0.5, 0.75, 1].map((f) => ({
   773→   773→    y: padding.top + f * innerH,
   774→   774→    price: yMax - f * yScale,
   775→   775→  }));
   776→   776→
   777→   777→  // ~6 x-axis date labels
   778→   778→  const labelStep = Math.max(1, Math.ceil(displayed.length / 6));
   779→   779→  const xLabels = displayed
   780→   780→    .map((c, i) => ({ c, i }))
   781→   781→    .filter(({ i }) => i % labelStep === 0 || i === displayed.length - 1);
   782→   782→
   783→   783→  // SMA50 + SMA200 on displayed (aggregated) close
   784→   784→  const sma = (period: number) =>
   785→   785→    displayed.map((_, i) => {
   786→   786→      if (i + 1 < period) return null;
   787→   787→      const slice = displayed.slice(i + 1 - period, i + 1).map((c) => c.close);
   788→   788→      return slice.reduce((s, n) => s + n, 0) / slice.length;
   789→   789→    });
   790→   790→  const sma50 = sma(Math.min(50, Math.max(2, Math.floor(displayed.length / 4))));
   791→   791→  const sma200 = sma(Math.min(200, Math.max(2, Math.floor(displayed.length / 2))));
   792→   792→
   793→   793→  const polylinePoints = (vals: (number | null)[]) =>
   794→   794→    vals
   795→   795→      .map((v, i) => (v == null ? null : `${xFor(i)},${yFor(v)}`))
   796→   796→      .filter((p): p is string => p !== null)
   797→   797→      .join(" ");
   798→   798→
   799→   799→  return (
   800→   800→    <svg width={width} height={height} className="block" role="img" aria-label="Candlestick chart">
   801→   801→      {/* Grid */}
