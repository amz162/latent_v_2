    25→def compute_rsi(p,n=14):
    26→ d=p.diff();g=d.where(d>0,0);l=-d.where(d<0,0)
    27→ ag=g.ewm(alpha=1/n,min_periods=n).mean();al=l.ewm(alpha=1/n,min_periods=n).mean()
    28→ return(100-100/(1+ag/al.replace(0,np.nan))).fillna(50)
    29→def compute_macd(p):
    30→ e12=p.ewm(span=12,adjust=False).mean();e26=p.ewm(span=26,adjust=False).mean()
    31→ m=e12-e26;s=m.ewm(span=9,adjust=False).mean();return m,s,m-s
    32→def compute_bb(p,w=20):
    33→ sma=p.rolling(w).mean();sd=p.rolling(w).std()
    34→ return((sma+2*sd-(sma-2*sd))/sma).fillna(0),((p-(sma-2*sd))/(4*sd).replace(0,np.nan)).fillna(0.5)
    35→def compute_atr(h,l,c,n=14):
    36→ tr=pd.concat([h-l,(h-c.shift()).abs(),(l-c.shift()).abs()],axis=1).max(axis=1)
    37→ return tr.rolling(n).mean().fillna(0)
    38→def compute_obv(c,v):
    39→ s=c.diff().apply(lambda x:1 if x>0 else(-1 if x<0 else 0));return(s*v).cumsum()
    40→def add_features(df):
    41→ df=df.copy().sort_values("date").reset_index(drop=True)
    42→ c=df["close"];h=df["high"];l=df["low"];v=df["volume"]
    43→ df["rsi_14"]=compute_rsi(c);m,s,hh=compute_macd(c)
    44→ df["macd"]=m;df["macd_signal"]=s;df["macd_hist"]=hh
    45→ bw,pb=compute_bb(c);df["bb_width"]=bw;df["bb_pct_b"]=pb
    46→ df["atr_14"]=compute_atr(h,l,c)
    47→ obv=compute_obv(c,v);df["obv_change_pct"]=obv.pct_change(periods=14).replace([np.inf,-np.inf],0).fillna(0)
    48→ for w in[20,50,200]:df[f"price_sma_{w}_ratio"]=c/c.rolling(w,min_periods=1).mean()
    49→ df["volume_sma_20_ratio"]=v/v.rolling(20,min_periods=1).mean().replace(0,1)
    50→ for p in[1,7,14,30]:df[f"return_{p}d"]=c.pct_change(periods=p).replace([np.inf,-np.inf],0).fillna(0)
    51→ df["hl_range_pct"]=((h-l)/c.replace(0,np.nan)).fillna(0)
    52→ df["body_size_pct"]=((df["close"]-df["open"]).abs()/c.replace(0,np.nan)).fillna(0)
    53→ df["volatility_14d"]=c.pct_change().rolling(14).std().fillna(0)
    54→ return df
    55→def add_labels(df):
    56→ # ML integrity (Task 1): per-stock forward returns, NO .fillna(0).

   102→def train_classical(df):
   103→ targets=["target_5d","target_10d","target_21d"]
   104→ # ML integrity (Task 2): date-based chronological split + purge on the FULL df
   105→ # (with NaN targets where immature). Per-horizon NaN drop happens INSIDE the
   106→ # per-horizon loop so per-horizon models keep rows whose target_5d is mature
   107→ # even when target_21d is immature (Phase A1 requirement). The purge uses
   108→ # MAX_HORIZON=21 (safe upper bound for all horizons).
   109→ tr,va,te=chronological_split(df)
   110→ cfg={"XGBoost":lambda:xgb.XGBRegressor(n_estimators=150,max_depth=3,learning_rate=0.05,subsample=0.7,colsample_bytree=0.7,random_state=42),
   111→  "LightGBM":lambda:lgb.LGBMRegressor(n_estimators=150,num_leaves=15,learning_rate=0.05,subsample=0.7,colsample_bytree=0.7,random_state=42,verbose=-1),
   112→  "CatBoost":lambda:cb.CatBoostRegressor(iterations=150,depth=3,learning_rate=0.05,random_state=42,verbose=0,allow_writing_files=False),
   113→  "Random Forest":lambda:RandomForestRegressor(n_estimators=150,max_depth=5,min_samples_leaf=15,max_features=0.7,random_state=42,n_jobs=-1)}
   114→ results={}
   115→ for name,factory in cfg.items():
   116→  print(f"  Training {name}...")
   117→  metrics={};preds={}
   118→  for tc in targets:
   119→   hz=int(tc.split("_")[1].replace("d",""))
   120→   # ML integrity (Phase A1): drop ONLY rows whose THIS-horizon target is NaN
   121→   # (per-horizon drop, not all-horizon). Preserves shorter-horizon rows.
   122→   tr_h=tr.dropna(subset=[tc]);te_h=te.dropna(subset=[tc])
   123→   Xtr=tr_h[FC].values;Xte=te_h[FC].values
   124→   ytr=tr_h[tc].values;yte=te_h[tc].values
   125→   m=factory();m.fit(Xtr,ytr);yp=m.predict(Xte)
   126→   metrics[f"{hz}d"]={"mse":float(mean_squared_error(yte,yp)),"mae":float(mean_absolute_error(yte,yp)),"r2":float(r2_score(yte,yp))}
   127→   for s in STOCKS:
   128→    sdf=df[df["symbol"]==s].sort_values("date").reset_index(drop=True)
   129→    if len(sdf)==0:continue
   130→    latest=sdf.iloc[-1];lf=latest[FC].values.reshape(1,-1)
   131→    pr=float(m.predict(lf)[0]);cp=float(latest["close"])
   132→    if s not in preds:preds[s]={"currentPrice":round(cp,2),"entry":round(cp,2)}
   133→    preds[s][f"ret_{hz}d"]=round(pr,4)
   134→  for s,pd_ in preds.items():
   135→   r5=pd_.get("ret_5d",0);r10=pd_.get("ret_10d",0);r21=pd_.get("ret_21d",0)
   136→   cp=pd_["currentPrice"]
   137→   sdf=df[df["symbol"]==s].sort_values("date").reset_index(drop=True)
   138→   sl=round(float(np.min(sdf["low"].tail(14).values)),2)
   139→   avg=(r5+r10+r21)/3
   140→   sig="BUY"if avg>0.01 else"SELL"if avg<-0.01 else"HOLD"
   141→   d="LONG"if sig!="SELL"else"SHORT"
   142→   pd_.update({"signal":sig,"direction":d,"confidence":min(100,int(abs(avg)*1000)),
   143→    "tp1":round(cp*(1+r5),2),"tp2":round(cp*(1+r10),2),"tp3":round(cp*(1+r21),2),
   144→    "sl":sl,"avgReturn":round(avg,4)})
   145→  results[name]={"trained":True,"version":"2.0.0","metrics":metrics,"predictions":preds}
   146→  print(f"    R²(5d)={metrics['5d']['r2']:.3f}")
   147→ return results

   149→def train_lstm(df):
   150→ try:
   151→  import torch,torch.nn as nn
   152→  from torch.utils.data import DataLoader,TensorDataset
   153→  print("  Training LSTM...")
   154→  # ML integrity (Task 2 + Task C): get the global train_end / val_end dates
   155→  # from chronological_split so every per-stock scaler is fit on the SAME
   156→  # global TRAIN partition (not an arbitrary first-60%-per-stock slice that
   157→  # could straddle the global train/val boundary).
   158→  _tr,_va,_te=chronological_split(df)
   159→  train_end_date=_tr["date"].max();val_end_date=_va["date"].max()
   160→  SEQ=30;seqs=[];targs=[];seq_dates=[]
   161→  for s in STOCKS:
   162→   sdf=df[df["symbol"]==s].sort_values("date").reset_index(drop=True)
   163→   if len(sdf)<SEQ+21:continue
   164→   f=sdf[FC].values
   165→   # ML integrity (Task 3, Task C): fit scaler on the per-stock TRAIN partition
   166→   # (rows with date <= train_end_date) — consistent with chronological_split.
   167→   ti=per_stock_train_idx(sdf,train_end_date)
   168→   if ti<SEQ+5:continue  # stock has too few TRAIN rows
   169→   sc=fit_scaler_on_train(f[:ti]);fs=sc.transform(f)
   170→   for i in range(len(sdf)-SEQ-21):
   171→    di=i+SEQ  # decision bar (after the sequence window — no in-seq future)
   172→    row=sdf.iloc[di]
   173→    tgts=[row["target_5d"],row["target_10d"],row["target_21d"]]
   174→    if any(pd.isna(t) or not np.isfinite(t) for t in tgts):continue
   175→    seqs.append(fs[i:i+SEQ]);targs.append(tgts);seq_dates.append(row["date"])
   176→  if len(seqs)<50:return{"trained":False,"error":"Not enough sequences"}
   177→  X=np.array(seqs,dtype=np.float32);y=np.array(targs,dtype=np.float32)
   178→  # ML integrity (Task 2): date-disjoint train/val/test split by DECISION DATE
   179→  # (replaces the old sp=int(len(X)*0.8) global row split).
   180→  Xtr,ytr,Xva,yva,Xte,yte=split_sequences_by_date(
   181→   X,y,seq_dates,train_end_date,val_end_date
   182→  )
   183→  dl=DataLoader(TensorDataset(torch.from_numpy(Xtr),torch.from_numpy(ytr)),batch_size=32,shuffle=True)
   184→  class M(nn.Module):
   185→   def __init__(s):
   186→    super().__init__();s.lstm=nn.LSTM(len(FC),64,2,batch_first=True,dropout=0.1);s.fc=nn.Linear(64,3)
   187→   def forward(s,x):
   188→    o,_=s.lstm(x);return s.fc(o[:,-1,:])
   189→  model=M();crit=nn.MSELoss();opt=torch.optim.Adam(model.parameters(),lr=0.001)
   190→  model.train()
   191→  for ep in range(30):
   192→   for bx,by in dl:opt.zero_grad();loss=crit(model(bx),by);loss.backward();opt.step()
   193→   if ep%10==0:print(f"    Epoch {ep}: loss={loss.item():.6f}")
   194→  model.eval()
   195→  with torch.no_grad():ypt=model(torch.from_numpy(Xte)).numpy()
   196→  metrics={"5d":{"mse":float(mean_squared_error(yte[:,0],ypt[:,0]))},
   197→   "10d":{"mse":float(mean_squared_error(yte[:,1],ypt[:,1]))},
   198→   "21d":{"mse":float(mean_squared_error(yte[:,2],ypt[:,2]))}}
   199→  def predict_fn(sdf):
   200→   f=sdf[FC].values;sc=StandardScaler();fs=sc.fit_transform(f)
   201→   sq=fs[-SEQ:].reshape(1,SEQ,len(FC))
   202→   with torch.no_grad():p=model(torch.from_numpy(sq.astype(np.float32)))[0].numpy()
   203→   return float(p[0]),float(p[1]),float(p[2])
   204→  preds=gen_preds(df,predict_fn)
   205→  print(f"    Done.")
   206→  return{"trained":True,"version":"2.0.0","metrics":metrics,"predictions":preds}
   207→ except Exception as e:
   208→  print(f"    LSTM FAILED: {e}",file=sys.stderr)
   209→  return{"trained":False,"error":str(e)}

   211→def train_cnn_lstm(df):
   212→ try:
   213→  import torch,torch.nn as nn
   214→  from torch.utils.data import DataLoader,TensorDataset
   215→  print("  Training CNN-LSTM...")
   216→  # ML integrity (Task 2 + Task C): global chronological boundaries so every
   217→  # per-stock scaler is fit on the SAME TRAIN partition.
   218→  _tr,_va,_te=chronological_split(df)
   219→  train_end_date=_tr["date"].max();val_end_date=_va["date"].max()
   220→  SEQ=30;seqs=[];targs=[];seq_dates=[]
   221→  for s in STOCKS:
   222→   sdf=df[df["symbol"]==s].sort_values("date").reset_index(drop=True)
   223→   if len(sdf)<SEQ+21:continue
   224→   f=sdf[FC].values
   225→   # ML integrity (Task 3): fit scaler on the per-stock TRAIN partition only
   226→   # (rows with date <= train_end_date) — replaces the old leaky
   227→   # StandardScaler().fit_transform(f) on the FULL per-stock history.
   228→   ti=per_stock_train_idx(sdf,train_end_date)
   229→   if ti<SEQ+5:continue
   230→   sc=fit_scaler_on_train(f[:ti]);fs=sc.transform(f)
   231→   for i in range(len(sdf)-SEQ-21):
   232→    di=i+SEQ  # decision bar (after the sequence window — no in-seq future)
   233→    row=sdf.iloc[di]
   234→    tgts=[row["target_5d"],row["target_10d"],row["target_21d"]]
   235→    if any(pd.isna(t) or not np.isfinite(t) for t in tgts):continue
   236→    seqs.append(fs[i:i+SEQ]);targs.append(tgts);seq_dates.append(row["date"])
   237→  if len(seqs)<50:return{"trained":False,"error":"Not enough sequences"}
   238→  X=np.array(seqs,dtype=np.float32);y=np.array(targs,dtype=np.float32)
   239→  # ML integrity (Task 2): date-disjoint train/val/test split by DECISION DATE.
   240→  Xtr,ytr,Xva,yva,Xte,yte=split_sequences_by_date(
   241→   X,y,seq_dates,train_end_date,val_end_date
   242→  )
   243→  dl=DataLoader(TensorDataset(torch.from_numpy(Xtr),torch.from_numpy(ytr)),batch_size=32,shuffle=True)
   244→  class CNNLSTM(nn.Module):
   245→   def __init__(s):
   246→    super().__init__()
   247→    s.conv1=nn.Conv1d(len(FC),32,kernel_size=3,padding=1);s.relu=nn.ReLU()
   248→    s.conv2=nn.Conv1d(32,32,kernel_size=3,padding=1)
   249→    s.lstm=nn.LSTM(32,64,2,batch_first=True,dropout=0.1);s.fc=nn.Linear(64,3)
   250→   def forward(s,x):
   251→    x=x.transpose(1,2);x=s.relu(s.conv1(x));x=s.relu(s.conv2(x))
   252→    x=x.transpose(1,2);o,_=s.lstm(x);return s.fc(o[:,-1,:])
   253→  model=CNNLSTM();crit=nn.MSELoss();opt=torch.optim.Adam(model.parameters(),lr=0.001)
   254→  model.train()
   255→  for ep in range(30):
   256→   for bx,by in dl:opt.zero_grad();loss=crit(model(bx),by);loss.backward();opt.step()
   257→   if ep%10==0:print(f"    Epoch {ep}: loss={loss.item():.6f}")
   258→  model.eval()
   259→  with torch.no_grad():ypt=model(torch.from_numpy(Xte)).numpy()
   260→  metrics={"5d":{"mse":float(mean_squared_error(yte[:,0],ypt[:,0]))},
   261→   "10d":{"mse":float(mean_squared_error(yte[:,1],ypt[:,1]))},
   262→   "21d":{"mse":float(mean_squared_error(yte[:,2],ypt[:,2]))}}
   263→  def predict_fn(sdf):
   264→   f=sdf[FC].values;sc=StandardScaler();fs=sc.fit_transform(f)
   265→   sq=fs[-SEQ:].reshape(1,SEQ,len(FC))
   266→   with torch.no_grad():p=model(torch.from_numpy(sq.astype(np.float32)))[0].numpy()
   267→   return float(p[0]),float(p[1]),float(p[2])
   268→  preds=gen_preds(df,predict_fn)
   269→  print(f"    Done.")
   270→  return{"trained":True,"version":"2.0.0","metrics":metrics,"predictions":preds}
   271→ except Exception as e:
   272→  print(f"    CNN-LSTM FAILED: {e}",file=sys.stderr)
   273→  return{"trained":False,"error":str(e)}

   275→def train_transformer_lstm(df):

   319→  metrics={"5d":{"mse":float(mean_squared_error(yte[:,0],ypt[:,0]))},

   321→   "21d":{"mse":float(mean_squared_error(yte[:,2],ypt[:,2]))}}

   324→   sq=fs[-SEQ:].reshape(1,SEQ,len(FC))
   325→   with torch.no_grad():p=model(torch.from_numpy(sq.astype(np.float32)))[0].numpy()
   326→   return float(p[0]),float(p[1]),float(p[2])
   327→  preds=gen_preds(df,predict_fn)
   328→  print(f"    Done.")
   329→  return{"trained":True,"version":"2.0.0","metrics":metrics,"predictions":preds}
   330→ except Exception as e:
   331→  print(f"    Transformer-LSTM FAILED: {e}",file=sys.stderr)
   332→  return{"trained":False,"error":str(e)}

   334→def train_neuralforecast(df):
   335→ from neuralforecast import NeuralForecast
   336→ from neuralforecast.models import PatchTST,NBEATS,TimesNet,Informer
   337→ results={}

   353→   fc=nf.predict()

   356→   for s in STOCKS:

   358→    if len(sdf)==0:continue

   360→    sc=fc[fc["unique_id"]==s]if"unique_id"in fc.columns else fc

   364→    else:tp1,tp2,tp3=cp*1.02,cp*1.03,cp*1.05
   365→    sl=round(float(np.min(sdf["low"].tail(14).values)),2)
   366→    avg=((tp1+tp2+tp3)/3-cp)/cp

   373→   results[name]={"trained":True,"version":"2.0.0","predictions":preds}

   375→  except Exception as e:

   388→def backtest_individual(df, models):
   389→ """Backtest each individual model on the chronological test partition.
   390→
   391→ ML-integrity note (Phase B audit Pattern #10): this is a SIMPLIFIED backtest —
   392→ it applies each model's LATEST per-stock signal to every historical test
   393→ row, NOT a walk-forward re-prediction. For honest walk-forward evaluation
   394→ see scripts/walk_forward_backtesting.py. The test partition itself IS
   395→ chronological (date-based, not row-based) thanks to chronological_split.
   396→ """
   397→ _tr,_va,te=chronological_split(df);te=te.copy()
   398→ bt={}
   399→ for name,md in models.items():
   400→  if not md.get("trained"):continue
   401→  buys=sells=cb_=cs=0;rets=[]
   402→  for s in STOCKS:
   403→   ste=te[te["symbol"]==s].sort_values("date").reset_index(drop=True)
   404→   if len(ste)<21:continue
   405→   for i in range(0,len(ste)-21,5):
   406→    ar=(ste.iloc[min(i+21,len(ste)-1)]["close"]-ste.iloc[i]["close"])/ste.iloc[i]["close"]
   407→    pr=md.get("predictions",{}).get(s,{});sig=pr.get("signal","HOLD")
   408→    if sig=="BUY":buys+=1;rets.append(ar);cb_+=ar>0
   409→    elif sig=="SELL":sells+=1;rets.append(-ar);cs+=ar<0
   410→  tt=buys+sells;wr=(cb_+cs)/max(tt,1)*100;ar_=float(np.mean(rets))*100 if rets else 0
   411→  bt[name]={"totalTrades":int(tt),"buySignals":int(buys),"sellSignals":int(sells),
   412→   "correctBuys":int(cb_),"correctSells":int(cs),
   413→   "winRate":round(float(wr),1),"avgReturnPct":round(float(ar_),2),
   414→   "maxReturnPct":round(float(max(rets)*100 if rets else 0),2),
   415→   "minReturnPct":round(float(min(rets)*100 if rets else 0),2),
   416→   "stars":int(min(5,max(1,int(wr/20)+1)))}
   417→ return bt

   419→def backtest_ensemble(df, models):
   420→ """Backtest the 3 ensemble methods + final consensus on the chronological test partition.
   421→
   422→ ML-integrity note (Phase B audit Pattern #10): SIMPLIFIED backtest — applies
   423→ each model's LATEST per-stock signal to every historical test row, NOT a
   424→ walk-forward re-prediction. For honest walk-forward see
   425→ scripts/walk_forward_backtesting.py. Test partition is date-based.
   426→ """
   427→ _tr,_va,te=chronological_split(df);te=te.copy()
   428→ trained_models={n:m for n,m in models.items() if m.get("trained") and m.get("predictions",{})}
   429→ if not trained_models:return{}
   430→
   431→ # For each test sample, collect all model signals + compute ensemble
   432→ ens_bt={"MAJORITY_VOTE":{"trades":0,"wins":0,"rets":[]},
   433→  "PROBABILITY_WEIGHTED":{"trades":0,"wins":0,"rets":[]},
   434→  "TIER_WEIGHTED":{"trades":0,"wins":0,"rets":[]},
   435→  "FINAL_CONSENSUS":{"trades":0,"wins":0,"rets":[]}}
   436→
   437→ for s in STOCKS:
   438→  ste=te[te["symbol"]==s].sort_values("date").reset_index(drop=True)
   439→  if len(ste)<21:continue
   440→  for i in range(0,len(ste)-21,5):
   441→   ar=(ste.iloc[min(i+21,len(ste)-1)]["close"]-ste.iloc[i]["close"])/ste.iloc[i]["close"]
   442→
   443→   # Collect signals from all trained models
   444→   signals=[];avg_returns=[]
   445→   for mn,md in trained_models.items():
   446→    pr=md.get("predictions",{}).get(s,{})
   447→    sig=pr.get("signal","HOLD")
   448→    ar_m=pr.get("avgReturn",0)
   449→    tier=MODEL_TIERS.get(mn,"TIER_3")
   450→    signals.append((mn,sig,ar_m,tier))
   451→
   452→   if not signals:continue
   453→
   454→   # A. Majority Vote
   455→   buys=sum(1 for _,s,_,_ in signals if s=="BUY")
   456→   sells=sum(1 for _,s,_,_ in signals if s=="SELL")
   457→   holds=sum(1 for _,s,_,_ in signals if s=="HOLD")
   458→   if buys>sells and buys>holds:ens_sig="BUY"
   459→   elif sells>buys and sells>holds:ens_sig="SELL"
   460→   else:ens_sig="HOLD"
   461→
   462→   if ens_sig!="HOLD":
   463→    ens_bt["MAJORITY_VOTE"]["trades"]+=1
   464→    if (ens_sig=="BUY"and ar>0)or(ens_sig=="SELL"and ar<0):ens_bt["MAJORITY_VOTE"]["wins"]+=1
   465→    ens_bt["MAJORITY_VOTE"]["rets"].append(ar if ens_sig=="BUY"else -ar)
   466→
   467→   # B. Probability-Weighted
   468→   avg_ret=sum(ar_m for _,_,ar_m,_ in signals)/len(signals)
   469→   if avg_ret>0.01:ens_sig2="BUY"
   470→   elif avg_ret<-0.01:ens_sig2="SELL"
   471→   else:ens_sig2="HOLD"
   472→
   473→   if ens_sig2!="HOLD":
   474→    ens_bt["PROBABILITY_WEIGHTED"]["trades"]+=1
   475→    if (ens_sig2=="BUY"and ar>0)or(ens_sig2=="SELL"and ar<0):ens_bt["PROBABILITY_WEIGHTED"]["wins"]+=1
   476→    ens_bt["PROBABILITY_WEIGHTED"]["rets"].append(ar if ens_sig2=="BUY"else -ar)
   477→
   478→   # C. Tier-Weighted
   479→   ws=sum(ar_m*TIER_W.get(t,0.25)for _,_,ar_m,t in signals)
   480→   wt=sum(TIER_W.get(t,0.25)for _,_,_,t in signals)
   481→   wr_=ws/wt if wt>0 else 0
   482→   if wr_>0.01:ens_sig3="BUY"
   483→   elif wr_<-0.01:ens_sig3="SELL"
   484→   else:ens_sig3="HOLD"
   485→
   486→   if ens_sig3!="HOLD":
   487→    ens_bt["TIER_WEIGHTED"]["trades"]+=1
   488→    if (ens_sig3=="BUY"and ar>0)or(ens_sig3=="SELL"and ar<0):ens_bt["TIER_WEIGHTED"]["wins"]+=1
   489→    ens_bt["TIER_WEIGHTED"]["rets"].append(ar if ens_sig3=="BUY"else -ar)
   490→
   491→   # Final Consensus (2+ methods agree)
   492→   votes=[ens_sig,ens_sig2,ens_sig3]
   493→   buy_v=sum(1 for v in votes if v=="BUY")
   494→   sell_v=sum(1 for v in votes if v=="SELL")
   495→   if buy_v>=2:fc_sig="BUY"
   496→   elif sell_v>=2:fc_sig="SELL"
   497→   else:fc_sig="HOLD"
   498→
   499→   if fc_sig!="HOLD":
   500→    ens_bt["FINAL_CONSENSUS"]["trades"]+=1
   501→    if (fc_sig=="BUY"and ar>0)or(fc_sig=="SELL"and ar<0):ens_bt["FINAL_CONSENSUS"]["wins"]+=1
   502→    ens_bt["FINAL_CONSENSUS"]["rets"].append(ar if fc_sig=="BUY"else -ar)
   503→
   504→ # Compute stats
   505→ result={}
   506→ for method,data in ens_bt.items():
   507→  t=data["trades"];w=data["wins"];r=data["rets"]
   508→  result[method]={"totalTrades":int(t),"wins":int(w),
   509→   "winRate":round(float(w/max(t,1)*100),1),
   510→   "avgReturnPct":round(float(np.mean(r)*100)if r else 0,2),
   511→   "maxReturnPct":round(float(max(r)*100)if r else 0,2),
   512→   "minReturnPct":round(float(min(r)*100)if r else 0,2),
   513→   "stars":int(min(5,max(1,int(w/max(t,1)*100/20)+1)))if t>0 else 0}
   514→ return result
