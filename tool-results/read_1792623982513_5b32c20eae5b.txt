   286→// ────────────────────────────────────────────────────────────────────────────
   287→// HistoricalOhlcvChart — loads ALL historical price data from askanalyst
   288→// via /api/psx/ohlcv/[symbol] (tries local cache first, then live fetch up to 10y)
   289→// ────────────────────────────────────────────────────────────────────────────
   290→
   291→interface Candle {
   292→  date: number;
   293→  open: number;
   294→  high: number;
   295→  low: number;
   296→  close: number;
   297→  volume: number;
   298→}
   299→
   300→function HistoricalOhlcvChart({ symbol, height = 360 }: { symbol: string; height?: number }) {
   301→  const [data, setData] = useState<Candle[]>([]);
   302→  const [loading, setLoading] = useState(true);
   303→  const [error, setError] = useState<string | null>(null);
   304→  const [source, setSource] = useState<string>("");
   305→  const [range, setRange] = useState<"1m" | "6m" | "1y" | "5y" | "all">("all");
   306→  const [chartType, setChartType] = useState<"line" | "candle">("line");
   307→  const [timeframe, setTimeframe] = useState<Timeframe>("1d");
   308→  const [chartWidth, setChartWidth] = useState(800);
   309→  const chartContainerRef = useRef<HTMLDivElement>(null);
   310→
   311→  useEffect(() => {
   312→    if (!chartContainerRef.current) return;
   313→    const el = chartContainerRef.current;
   314→    const ro = new ResizeObserver((entries) => {
   315→      for (const e of entries) setChartWidth(Math.max(320, Math.floor(e.contentRect.width)));
   316→    });
   317→    ro.observe(el);
   318→    setChartWidth(Math.max(320, el.clientWidth));
   319→    return () => ro.disconnect();
   320→  }, []);
   321→
   322→  useEffect(() => {
   323→    let cancelled = false;
   324→    setLoading(true);
   325→    setError(null);
   326→    setData([]);
   327→
   328→    fetch(`/api/psx/ohlcv/${encodeURIComponent(symbol)}`)
   329→      .then((r) => {
   330→        if (!r.ok) {
   331→          return r.json().then((err) => {
   332→            throw new Error(err?.error || `Upstream ${r.status}`);
   333→          });
   334→        }
   335→        return r.json();
   336→      })
   337→      .then((d) => {
   338→        if (cancelled) return;
   339→        if (!d || !Array.isArray(d.ohlcv) || d.ohlcv.length === 0) {
   340→          setError("No price data available for this symbol");
   341→          setLoading(false);
   342→          return;
   343→        }
   344→        const candles: Candle[] = d.ohlcv
   345→          .filter((c: any) => c && typeof c.date === "number" && !isNaN(c.close))
   346→          .map((c: any) => ({
   347→            date: c.date,
   348→            open: c.open,
   349→            high: c.high,
   350→            low: c.low,
   351→            close: c.close,
   352→            volume: c.volume || 0,
   353→          }));
   354→        if (cancelled) return;
   355→        if (candles.length === 0) {
   356→          setError("No valid rows in upstream data");
   357→          setLoading(false);
   358→          return;
   359→        }
   360→        setData(candles);
   361→        setSource(d.source || "askanalyst.com.pk");
   362→        setLoading(false);
   363→      })
   364→      .catch((err) => {
   365→        if (cancelled) return;
   366→        console.error("[HistoricalOhlcvChart] fetch error:", err);
   367→        setError(err instanceof Error ? err.message : "Unknown error");
   368→        setLoading(false);
   369→      });
   370→
   371→    return () => {
   372→      cancelled = true;
   373→    };
   374→  }, [symbol]);
   375→
   376→  // Apply range filter to the loaded data
   377→  const filtered = (() => {
   378→    if (data.length === 0) return [];
   379→    if (range === "all") return data;
   380→    const now = Date.now();
   381→    const ms: Record<typeof range, number> = {
   382→      "1m": 30 * 24 * 60 * 60 * 1000,
   383→      "6m": 182 * 24 * 60 * 60 * 1000,
   384→      "1y": 365 * 24 * 60 * 60 * 1000,
   385→      "5y": 5 * 365 * 24 * 60 * 60 * 1000,
   386→      all: Number.MAX_SAFE_INTEGER,
   387→    };
   388→    const cutoff = now - ms[range];
   389→    const sliced = data.filter((c) => c.date >= cutoff);
   390→    return sliced.length > 0 ? sliced : data;
   391→  })();
   392→
   393→  const enriched = computeSMAAndLevels(filtered);
   394→  const { support, resistance } = findSupportResistance(enriched);
   395→
   396→  const formatXAxis = (ts: number) => {
   397→    const d = new Date(ts);
   398→    return d.toLocaleDateString("en-PK", { month: "short", year: "2-digit" });
   399→  };
   400→
   401→  if (loading) {
   402→    return (
   403→      <div
   404→        className="w-full rounded-lg bg-[#0A0E1A] border border-white/[0.06] flex flex-col items-center justify-center text-center"
   405→        style={{ height }}
   406→      >
   407→        <Loader2 className="h-8 w-8 text-[#00E676] animate-spin mb-3" />
   408→        <p className="text-xs font-semibold text-[#F4F7FB]">
   409→          Loading all historical data for {symbol}...
   410→        </p>
   411→        <p className="text-[10px] text-[#64748B] mt-1">
   412→          Fetching up to 10 years of daily candles from askanalyst.com.pk
   413→        </p>
   414→        <div className="mt-3 w-48 h-1 bg-white/[0.06] rounded-full overflow-hidden">
   415→          <div className="h-full bg-[#00E676]/60 animate-pulse" style={{ width: "60%" }} />
   416→        </div>
   417→      </div>
   418→    );
   419→  }
   420→
   421→  if (error) {
   422→    return (
   423→      <div
   424→        className="w-full rounded-lg bg-[#0A0E1A] border border-white/[0.06] flex flex-col items-center justify-center text-center p-6"
   425→        style={{ height }}
   426→      >
   427→        <p className="text-sm font-semibold text-[#FF5C5C] mb-1">Chart unavailable</p>
   428→        <p className="text-[10px] text-[#64748B]">{error}</p>
   429→        <p className="text-[10px] text-[#64748B] mt-1 max-w-xs">
   430→          AskAnalyst.com.pk technical chart endpoint may be rate-limited or down, or this symbol is
   431→          not in their coverage. Try another PSX-listed stock.
   432→        </p>
   433→      </div>
   434→    );
   435→  }
   436→
   437→  if (filtered.length === 0) {
   438→    return (
   439→      <div className="w-full flex items-center justify-center text-xs text-[#64748B]" style={{ height }}>
   440→        No data points in selected range
   441→      </div>
   442→    );
   443→  }
   444→
   445→  const firstDate = new Date(filtered[0].date).toLocaleDateString("en-PK", { year: "numeric", month: "short", day: "numeric" });
   446→  const lastDate = new Date(filtered[filtered.length - 1].date).toLocaleDateString("en-PK", { year: "numeric", month: "short", day: "numeric" });
   447→  const firstPrice = filtered[0].close;
   448→  const lastPrice = filtered[filtered.length - 1].close;
   449→  const totalReturn = firstPrice > 0 ? ((lastPrice - firstPrice) / firstPrice) * 100 : 0;
   450→

   675→function computeSMAAndLevels(candles: Candle[]): (Candle & { sma50?: number; sma200?: number })[] {
   676→  return candles.map((c, i) => {
   677→    const sma50 = i >= 49 ? avg(candles.slice(i - 49, i + 1).map((x) => x.close)) : undefined;
   678→    const sma200 = i >= 199 ? avg(candles.slice(i - 199, i + 1).map((x) => x.close)) : undefined;
   679→    return { ...c, sma50, sma200 };
   680→  });
   681→}
   682→
   683→function avg(nums: number[]): number {
   684→  return nums.reduce((s, n) => s + n, 0) / nums.length;
   685→}
   686→
   687→function findSupportResistance(data: (Candle & { sma50?: number; sma200?: number })[]): { support: number | null; resistance: number | null } {
   688→  if (data.length < 20) return { support: null, resistance: null };
   689→  const recent = data.slice(-60, -5);
   690→  if (recent.length === 0) return { support: null, resistance: null };
   691→  const lows = recent.map((c) => c.low);
   692→  const highs = recent.map((c) => c.high);
   693→  const support = Math.min(...lows);
   694→  const resistance = Math.max(...highs);
   695→  return { support, resistance };
   696→}

   698→// ────────────────────────────────────────────────────────────────────────────
   699→// CandlestickChart — pure SVG candlestick chart rendered from OHLCV data.
   700→// Green bodies for up days (close >= open), red for down days (close < open).
   701→// Sub-samples to a max of ~120 candles to keep SVG rendering performant.
   702→// ────────────────────────────────────────────────────────────────────────────
   703→
   704→function aggregateCandles(candles: Candle[], maxBars: number): Candle[] {
   705→  if (candles.length <= maxBars) return candles;
   706→  const binSize = Math.ceil(candles.length / maxBars);
   707→  const out: Candle[] = [];
   708→  for (let i = 0; i < candles.length; i += binSize) {
   709→    const bin = candles.slice(i, i + binSize);
   710→    if (bin.length === 0) continue;
   711→    out.push({
   712→      date: bin[0].date,
   713→      open: bin[0].open,
   714→      high: Math.max(...bin.map((c) => c.high)),
   715→      low: Math.min(...bin.map((c) => c.low)),
   716→      close: bin[bin.length - 1].close,
   717→      volume: bin.reduce((s, c) => s + (c.volume || 0), 0),
   718→    });
   719→  }
   720→  return out;
   721→}
   722→
   723→function CandlestickChart({
   724→  candles,
   725→  support,
   726→  resistance,
   727→  width,
   728→  height,
   729→}: {
   730→  candles: Candle[];
   731→  support: number | null;
   732→  resistance: number | null;
   733→  width: number;
   734→  height: number;
   735→}) {
   736→  const padding = { top: 12, right: 50, bottom: 22, left: 6 };
   737→  const innerW = Math.max(50, width - padding.left - padding.right);
   738→  const innerH = Math.max(50, height - padding.top - padding.bottom);
   739→
   740→  // Sub-sample to ~120 candles for performance on long histories (10y daily)
   741→  const displayed = aggregateCandles(candles, 120);
   742→
   743→  if (displayed.length === 0) {
   744→    return (
   745→      <div className="w-full flex items-center justify-center text-xs text-[#64748B]" style={{ height }}>
   746→        No data to render
   747→      </div>
   748→    );
   749→  }
   750→
   751→  // Y scale — include support/resistance and SMA range in view
   752→  const lows = displayed.map((c) => c.low);
   753→  const highs = displayed.map((c) => c.high);
   754→  let yMin = Math.min(...lows, support ?? Infinity, resistance ?? Infinity);
   755→  let yMax = Math.max(...highs, support ?? -Infinity, resistance ?? -Infinity);
   756→  if (!isFinite(yMin) || !isFinite(yMax)) {
   757→    yMin = Math.min(...lows);
   758→    yMax = Math.max(...highs);
   759→  }
   760→  const yRange = Math.max(0.0001, yMax - yMin);
   761→  const pad = yRange * 0.06;
   762→  yMin -= pad;
   763→  yMax += pad;
   764→  const yScale = yMax - yMin;
   765→
   766→  const xFor = (i: number) => padding.left + (i + 0.5) * (innerW / displayed.length);
   767→  const yFor = (p: number) => padding.top + (1 - (p - yMin) / yScale) * innerH;
   768→
   769→  const cw = Math.max(1, Math.min(12, (innerW / displayed.length) * 0.65));
   770→
   771→  // 5 horizontal gridlines + price labels on right axis
   772→  const gridLines = [0, 0.25, 0.5, 0.75, 1].map((f) => ({
   773→    y: padding.top + f * innerH,
   774→    price: yMax - f * yScale,
   775→  }));
   776→
   777→  // ~6 x-axis date labels
   778→  const labelStep = Math.max(1, Math.ceil(displayed.length / 6));
   779→  const xLabels = displayed
   780→    .map((c, i) => ({ c, i }))
   781→    .filter(({ i }) => i % labelStep === 0 || i === displayed.length - 1);
   782→
   783→  // SMA50 + SMA200 on displayed (aggregated) close
   784→  const sma = (period: number) =>
   785→    displayed.map((_, i) => {
   786→      if (i + 1 < period) return null;
   787→      const slice = displayed.slice(i + 1 - period, i + 1).map((c) => c.close);
   788→      return slice.reduce((s, n) => s + n, 0) / slice.length;
   789→    });
   790→  const sma50 = sma(Math.min(50, Math.max(2, Math.floor(displayed.length / 4))));
   791→  const sma200 = sma(Math.min(200, Math.max(2, Math.floor(displayed.length / 2))));
   792→
   793→  const polylinePoints = (vals: (number | null)[]) =>
   794→    vals
   795→      .map((v, i) => (v == null ? null : `${xFor(i)},${yFor(v)}`))
   796→      .filter((p): p is string => p !== null)
   797→      .join(" ");
   798→
   799→  return (
   800→    <svg width={width} height={height} className="block" role="img" aria-label="Candlestick chart">
   801→      {/* Grid */}
   802→      {gridLines.map((g, i) => (
   803→        <g key={`g${i}`}>
   804→          <line x1={padding.left} y1={g.y} x2={padding.left + innerW} y2={g.y} stroke="rgba(255,255,255,0.04)" strokeWidth={1} />
   805→          <text x={padding.left + innerW + 6} y={g.y + 3} fontSize={10} fill="#64748B" fontFamily="monospace">
   806→            {g.price.toFixed(2)}
   807→          </text>
   808→        </g>
   809→      ))}
   810→
   811→      {/* X-axis date labels */}
   812→      {xLabels.map(({ c, i }) => (
   813→        <text
   814→          key={`xl${i}`}
   815→          x={xFor(i)}
   816→          y={padding.top + innerH + 14}
   817→          fontSize={10}
   818→          fill="#64748B"
   819→          fontFamily="monospace"
   820→          textAnchor="middle"
   821→        >
   822→          {new Date(c.date).toLocaleDateString("en-PK", { month: "short", year: "2-digit" })}
   823→        </text>
   824→      ))}
   825→
   826→      {/* Support line */}
   827→      {support != null && (
   828→        <g>
   829→          <line
   830→            x1={padding.left}
   831→            y1={yFor(support)}
   832→            x2={padding.left + innerW}
   833→            y2={yFor(support)}
   834→            stroke="#FF5C5C"
   835→            strokeWidth={1}
   836→            strokeDasharray="6 4"
   837→            strokeOpacity={0.6}
   838→          />
   839→          <text x={padding.left + 4} y={yFor(support) - 3} fontSize={10} fill="#FF5C5C" fontFamily="monospace">
   840→            S: {support.toFixed(2)}
   841→          </text>
   842→        </g>
   843→      )}
   844→
   845→      {/* Resistance line */}
   846→      {resistance != null && (
   847→        <g>
   848→          <line
   849→            x1={padding.left}
   850→            y1={yFor(resistance)}
   851→            x2={padding.left + innerW}
   852→            y2={yFor(resistance)}
   853→            stroke="#FBBF24"
   854→            strokeWidth={1}
   855→            strokeDasharray="6 4"
   856→            strokeOpacity={0.6}
   857→          />
   858→          <text x={padding.left + 4} y={yFor(resistance) - 3} fontSize={10} fill="#FBBF24" fontFamily="monospace">
   859→            R: {resistance.toFixed(2)}
   860→          </text>
   861→        </g>
   862→      )}
   863→
   864→      {/* SMA50 line */}
   865→      <polyline
   866→        points={polylinePoints(sma50)}
   867→        fill="none"
   868→        stroke="#22E881"
   869→        strokeWidth={1}
   870→        strokeDasharray="3 3"
   871→        strokeOpacity={0.7}
   872→      />
   873→      {/* SMA200 line */}
   874→      <polyline
   875→        points={polylinePoints(sma200)}
   876→        fill="none"
   877→        stroke="#94A3B8"
   878→        strokeWidth={1}
   879→        strokeDasharray="6 3"
   880→        strokeOpacity={0.7}
   881→      />
   882→
   883→      {/* Candles */}
   884→      {displayed.map((c, i) => {
   885→        const x = xFor(i);
   886→        const up = c.close >= c.open;
   887→        const color = up ? "#00E676" : "#FF5C5C";
   888→        const bodyTop = yFor(Math.max(c.open, c.close));
   889→        const bodyBot = yFor(Math.min(c.open, c.close));
   890→        const bodyH = Math.max(1, bodyBot - bodyTop);
   891→        return (
   892→          <g key={`c${i}`}>
   893→            {/* Wick (high to low) */}
   894→            <line x1={x} y1={yFor(c.high)} x2={x} y2={yFor(c.low)} stroke={color} strokeWidth={1} />
   895→            {/* Body (open to close) */}
   896→            <rect
   897→              x={x - cw / 2}
   898→              y={bodyTop}
   899→              width={cw}
   900→              height={bodyH}
   901→              fill={color}
   902→              fillOpacity={up ? 0.85 : 0.85}
   903→              stroke={color}
   904→              strokeWidth={0.5}
   905→            />
   906→          </g>
   907→        );
   908→      })}
   909→    </svg>
   910→  );
   911→}
