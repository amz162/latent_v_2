    45→def compute_rsi(p, n=14):
    46→    d = p.diff(); g = d.where(d > 0, 0); l = -d.where(d < 0, 0)
    47→    ag = g.ewm(alpha=1/n, min_periods=n).mean(); al = l.ewm(alpha=1/n, min_periods=n).mean()
    48→    return (100 - 100 / (1 + ag / al.replace(0, np.nan))).fillna(50)
    49→
    50→
    51→def compute_macd(p):
    52→    e12 = p.ewm(span=12, adjust=False).mean(); e26 = p.ewm(span=26, adjust=False).mean()
    53→    m = e12 - e26; s = m.ewm(span=9, adjust=False).mean(); return m, s, m - s
    54→

    56→def compute_bb(p, w=20):
    57→    sma = p.rolling(w).mean(); sd = p.rolling(w).std()
    58→    return ((sma + 2*sd - (sma - 2*sd)) / sma).fillna(0), \
    59→           ((p - (sma - 2*sd)) / (4*sd).replace(0, np.nan)).fillna(0.5)

    62→def compute_atr(h, l, c, n=14):
    63→    tr = pd.concat([h - l, (h - c.shift()).abs(), (l - c.shift()).abs()], axis=1).max(axis=1)
    64→    return tr.rolling(n).mean().fillna(0)

    67→def compute_obv(c, v):
    68→    s = c.diff().apply(lambda x: 1 if x > 0 else (-1 if x < 0 else 0))
    69→    return (s * v).cumsum()

   122→def apply_atr_rules(signal, current_price, atr):
   123→    """Return (entry, sl, tp1, tp2, tp3) following ATR rules.
   124→
   125→    For BUY: entry = current_price, SL = entry - 1.5×ATR (below), TPs above
   126→    For SELL: entry = current_price, SL = entry + 1.5×ATR (above), TPs below
   127→    For HOLD: returns zeros (model not tradeable)
   128→    """
   129→    if signal == "BUY":
   130→        entry = current_price
   131→        sl = entry - SL_MULT * atr
   132→        tp1 = entry + TP1_MULT * atr
   133→        tp2 = entry + TP2_MULT * atr
   134→        tp3 = entry + TP3_MULT * atr
   135→    elif signal == "SELL":
   136→        entry = current_price
   137→        sl = entry + SL_MULT * atr  # ABOVE entry (fixed!)
   138→        tp1 = entry - TP1_MULT * atr  # below entry (target on downside)
   139→        tp2 = entry - TP2_MULT * atr
   140→        tp3 = entry - TP3_MULT * atr
   141→    else:  # HOLD
   142→        entry = current_price
   143→        sl = current_price
   144→        tp1 = current_price
   145→        tp2 = current_price
   146→        tp3 = current_price
   147→    return entry, sl, tp1, tp2, tp3

   150→def signal_from_avg_return(avg_return):
   151→    """Convert avgReturn to BUY/SELL/HOLD using relaxed thresholds."""
   152→    if avg_return > 0.005:  # Lowered from 0.01 to 0.005 — captures more signals
   153→        return "BUY", "LONG"
   154→    elif avg_return < -0.005:
   155→        return "SELL", "SHORT"
   156→    else:
   157→        return "HOLD", "LONG"

   610→                            exit_price = tp1

   612→                    buys += 1
