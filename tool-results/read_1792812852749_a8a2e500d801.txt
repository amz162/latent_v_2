   136→function fmtPct(v: number, withSign = false): string {
   137→  const sign = withSign && v > 0 ? "+" : "";
   138→  return `${sign}${Number(v).toFixed(2)}%`;
   139→}

   141→function fmtPkr(v: number): string {
   142→  return `Rs. ${Math.round(v).toLocaleString("en-PK")}`;

   423→  const runSimulation = async () => {
   424→    const capital = parseFloat(simCapital.replace(/[,\s]/g, ""));
   425→    const periodValue = parseInt(simPeriodValue, 10);
   426→    const monthlySip = parseFloat(simMonthlySip.replace(/[,\s]/g, "")) || 0;
   427→    const numTrades = parseInt(simNumTrades, 10);
   428→    const minConfidence = parseInt(simMinConfidence, 10);
   429→    if (!Number.isFinite(capital) || capital <= 0) {
   430→      toast.error("Enter a valid capital amount");
   431→      return;
   432→    }
   433→    if (!Number.isFinite(periodValue) || periodValue <= 0) {
   434→      toast.error("Enter a valid time period");
   435→      return;
   436→    }
   437→    if (!Number.isFinite(numTrades) || numTrades <= 0) {
   438→      toast.error("Enter a valid number of trades");
   439→      return;
   440→    }
   441→    if (!Number.isFinite(minConfidence) || minConfidence < 0 || minConfidence > 100) {
   442→      toast.error("Min confidence must be 0–100");
   443→      return;
   444→    }
   445→    setSimLoading(true);
   446→    try {
   447→      const qs =
   448→        `capital=${encodeURIComponent(simCapital)}` +
   449→        `&periodValue=${encodeURIComponent(simPeriodValue)}` +
   450→        `&periodUnit=${encodeURIComponent(simPeriodUnit)}` +
   451→        `&monthlySip=${encodeURIComponent(simMonthlySip)}` +
   452→        `&numTrades=${encodeURIComponent(simNumTrades)}` +
   453→        `&minConfidence=${encodeURIComponent(simMinConfidence)}` +
   454→        `&tradeDirection=${encodeURIComponent(simTradeDirection)}`;
   455→      const r = await fetch(`/api/backtest/investment-simulator?${qs}`);
   456→      const j = await r.json();
   457→      if (j.error) {
   458→        toast.error(j.error);
   459→        return;
   460→      }
   461→      setSimResult(j as SimResponse);
   462→      toast.success(
   463→        `Simulated ${j.config.numTrades} trades over ${j.config.numDays} days — ROI ${j.metrics.roiPct}%`
   464→      );
   465→    } catch (e) {
   466→      toast.error("Simulation failed");
   467→    } finally {
   468→      setSimLoading(false);
   469→    }
   470→  };
   471→

   482→    const losses = total - wins;
   483→    const winRate = fc.winRate ?? 0;
   484→    const avgReturn = fc.avgReturnPct ?? 0;
   485→    const best = fc.maxReturnPct ?? 0;
   486→    const worst = fc.minReturnPct ?? 0;

   488→  }, [ensStats]);

   493→  //
   494→  // Issue 26: fetch LIVE prices for ALL portfolio holdings (not just the 8
   495→  // COVERED_SYMBOLS). Hits /api/psx/live/batch (server wraps getLiveQuote
   496→  // with a 60s in-memory cache). Auto-refreshes every 60s to match the cache
   497→  // TTL. Symbols with no fetchable quote are simply omitted from the response
   498→  // (the row falls back to "no live price").
   499→  const portfolioSymbols = useMemo(
   500→    () => portfolio.filter((p) => p.buyPrice !== null && p.shares > 0).map((p) => p.symbol),
   501→    [portfolio]
   502→  );
   503→  const portfolioSymbolsKey = portfolioSymbols.join(",");
   504→  const [livePriceBatch, setLivePriceBatch] = useState<Record<string, number>>({});
   505→  useEffect(() => {
   506→    if (!portfolioSymbolsKey) {
   507→      setLivePriceBatch({});
   508→      return;
   509→    }
   510→    let cancelled = false;
   511→    const symbolsList = Array.from(new Set(portfolioSymbols));
   512→    const doFetch = () =>
   513→      fetch(`/api/psx/live/batch?symbols=${encodeURIComponent(symbolsList.join(","))}`)
   514→        .then((r) => (r.ok ? r.json() : null))

  1459→function SectorPieChart({ data }: { data: { sector: string; value: number; pct: number }[] }) {
  1460→  const size = 160;
  1461→  const cx = size / 2;
  1462→  const cy = size / 2;
  1463→  const r = 65;
  1464→  const innerR = 30; // donut hole
  1465→
  1466→  // Issue 26: Single-slice (100%) degenerate case — SVG arc spec omits the
  1467→  // slice when start === end (because the path's start and end points are
  1468→  // identical, the arc is "zero-length" and skipped). Render the donut as
  1469→  // two concentric <circle> elements instead so a 1-stock portfolio still
  1470→  // shows a full ring instead of an empty box.
  1471→  if (data.length === 1 && data[0].pct >= 99.999) {
  1472→    const d = data[0];
  1473→    const color = SECTOR_COLORS[0];
  1474→    return (
  1475→      <svg
  1476→        viewBox={`0 0 ${size} ${size}`}
  1477→        width={size}
  1478→        height={size}
  1479→        className="shrink-0"
  1480→        role="img"
  1481→        aria-label={`Sector distribution: ${d.sector} ${d.pct.toFixed(0)}%`}
  1482→      >
  1483→        <circle cx={cx} cy={cy} r={r} fill={color} stroke="#0A0E1A" strokeWidth={1.5} />
  1484→        <circle cx={cx} cy={cy} r={innerR} fill="#0A0E1A" />
  1485→        <title>{`${d.sector}: ${d.pct.toFixed(1)}% (${fmtPkr(d.value)})`}</title>
  1486→      </svg>
  1487→    );
  1488→  }
  1489→
  1490→  // SVG arc paths for each sector slice. Cumulative angle is computed
  1491→  // functionally from the preceding slices' percentages (no mutation) to
  1492→  // satisfy react-hooks/immutability.
  1493→  const slices = data.map((d, i) => {
  1494→    const cumulativeBeforePct = data.slice(0, i).reduce((s, x) => s + x.pct, 0);
  1495→    const startAngle = -Math.PI / 2 + (cumulativeBeforePct / 100) * 2 * Math.PI;
  1496→    const angleSpan = (d.pct / 100) * 2 * Math.PI;
  1497→    const endAngle = startAngle + angleSpan;
  1498→
  1499→    // Outer arc endpoints
  1500→    const x1 = cx + r * Math.cos(startAngle);
  1501→    const y1 = cy + r * Math.sin(startAngle);
  1502→    const x2 = cx + r * Math.cos(endAngle);
  1503→    const y2 = cy + r * Math.sin(endAngle);
  1504→    // Inner arc endpoints (donut)
  1505→    const xi1 = cx + innerR * Math.cos(startAngle);
  1506→    const yi1 = cy + innerR * Math.sin(startAngle);
  1507→    const xi2 = cx + innerR * Math.cos(endAngle);
  1508→    const yi2 = cy + innerR * Math.sin(endAngle);
  1509→
  1510→    const largeArc = angleSpan > Math.PI ? 1 : 0;
  1511→    // Donut slice path: outer arc forward + line in + inner arc backward + close
  1512→    const path = `M ${x1} ${y1} A ${r} ${r} 0 ${largeArc} 1 ${x2} ${y2} L ${xi2} ${yi2} A ${innerR} ${innerR} 0 ${largeArc} 0 ${xi1} ${yi1} Z`;
  1513→    return { path, color: SECTOR_COLORS[i % SECTOR_COLORS.length], ...d };
  1514→  });
  1515→
  1516→  return (
  1517→    <svg
  1518→      viewBox={`0 0 ${size} ${size}`}
  1519→      width={size}
  1520→      height={size}
  1521→      className="shrink-0"
  1522→      role="img"
  1523→      aria-label={`Sector distribution: ${data.map((d) => `${d.sector} ${d.pct.toFixed(0)}%`).join(", ")}`}
  1524→    >
  1525→      {slices.map((s, i) => (
  1526→        <path
  1527→          key={i}
  1528→          d={s.path}
  1529→          fill={s.color}
  1530→          stroke="#0A0E1A"
  1531→          strokeWidth={1.5}
  1532→        >
  1533→          <title>{`${s.sector}: ${s.pct.toFixed(1)}% (${fmtPkr(s.value)})`}</title>
  1534→        </path>
  1535→      ))}
  1536→    </svg>
  1537→  );
  1538→}
