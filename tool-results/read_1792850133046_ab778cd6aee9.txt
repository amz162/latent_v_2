    46→function fmtNum(n: number): string {
    47→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}B`;
    48→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}M`;
    49→  return n.toFixed(0);
    50→}

    52→function fyOf(period: string): string {
    53→  const m = period.match(/FY\s?(\d+)/i);
    54→  return m ? m[1] : period.slice(-2);
    55→}

   268→function StockDeepDive({
   269→  results,
   270→  symbol,
   271→  company,
   272→  onBack,
   273→}: {
   274→  results: QuarterlyResult[];
   275→  symbol: string;
   276→  company: string;
   277→  onBack: () => void;
   278→}) {
   279→  const patterns = detectPatterns(results);
   280→  const quarters = generateQuarterlySummary(results, symbol);
   281→  const yearsBetween =
   282→    results.length >= 2
   283→      ? (new Date(results[results.length - 1].periodEnd).getTime() -
   284→          new Date(results[0].periodEnd).getTime()) /
   285→        (365 * 24 * 60 * 60 * 1000)
   286→      : 0;
   287→  const revCAGR = computeCAGR(results[0].metrics.revenue, results[results.length - 1].metrics.revenue, yearsBetween);
   288→  const profCAGR = computeCAGR(
   289→    results[0].metrics.profitAfterTax,
   290→    results[results.length - 1].metrics.profitAfterTax,
   291→    yearsBetween
   292→  );
   293→  const epsCAGR = computeCAGR(results[0].metrics.eps, results[results.length - 1].metrics.eps, yearsBetween);
   294→  const totalDPS = results.reduce((s, r) => s + r.metrics.dividendPerShare, 0);
   295→  const volatility = computeStdDev(results.map((r) => r.metrics.epsYoYGrowth));
   296→  const revVolatility = computeStdDev(results.map((r) => r.metrics.revenueYoYGrowth));
   297→  const marginSlope = computeSlope(results.map((r) => r.metrics.netMargin));
   298→
   299→  const stats: { label: string; value: string; sub: string; tone?: "pos" | "neg" | "neutral"; explanation: string }[] = [
   300→    { label: "Revenue CAGR", value: `${revCAGR.toFixed(1)}%`, sub: `${yearsBetween.toFixed(1)}y compound`, tone: revCAGR >= 0 ? "pos" : "neg", explanation: "How fast sales grew per year, on average, over the entire period" },
   301→    { label: "Profit CAGR", value: `${profCAGR.toFixed(1)}%`, sub: "PAT growth", tone: profCAGR >= 0 ? "pos" : "neg", explanation: "How fast profit grew per year, on average" },
   302→    { label: "EPS CAGR", value: `${epsCAGR.toFixed(1)}%`, sub: "per-share profit", tone: epsCAGR >= 0 ? "pos" : "neg", explanation: "How fast per-share profit grew per year" },
   303→    { label: "Total DPS", value: `Rs. ${totalDPS.toFixed(2)}`, sub: `${results.length} quarters`, tone: "neutral", explanation: "Total cash dividends paid per share over the entire period" },
   304→    { label: "EPS volatility", value: `${volatility.toFixed(1)}pp`, sub: "EPS YoY std dev", tone: volatility > 15 ? "neg" : "neutral", explanation: "How unpredictable the profit growth is. Higher = more risky/uncertain" },
   305→    { label: "Revenue volatility", value: `${revVolatility.toFixed(1)}pp`, sub: "Rev YoY std dev", tone: revVolatility > 15 ? "neg" : "neutral", explanation: "How unpredictable the sales growth is" },
   306→    { label: "Margin trend slope", value: `${marginSlope >= 0 ? "+" : ""}${marginSlope.toFixed(2)}pp/q`, sub: "net margin trend", tone: marginSlope >= 0 ? "pos" : "neg", explanation: "Whether profit margin is improving (positive) or declining (negative) over time" },
   307→    { label: "Patterns detected", value: `${patterns.length}`, sub: "fundamental signals", tone: "neutral", explanation: "Number of fundamental patterns found in the data" },
   308→  ];
   309→
   310→  return (
   311→    <div className="space-y-4">
   312→      {/* Header with back button */}
   313→      <Card className="bg-[#131829] border-white/[0.08]">
   314→        <CardContent className="p-4">
   315→          <div className="flex items-start gap-3">
   316→            <Button
   317→              variant="outline"
   318→              size="sm"
   319→              onClick={onBack}
   320→              className="border-white/[0.12] text-[#94A3B8] hover:text-[#F4F7FB] shrink-0"
   321→            >
   322→              <ArrowLeft className="h-3.5 w-3.5 mr-1.5" /> Back
   323→            </Button>
   324→            <div className="flex-1 min-w-0">
   325→              <div className="flex items-baseline gap-2 flex-wrap">
   326→                <h2 className="text-xl font-bold text-[#F4F7FB]">{symbol}</h2>
   327→                <span className="text-sm text-[#94A3B8] truncate">{company}</span>
   328→              </div>
   329→              <p className="text-xs text-[#64748B] mt-1">
   330→                Deep dive on {results.length} quarters ·{" "}
   331→                {new Date(results[0].periodEnd).toLocaleDateString("en-PK", { year: "numeric", month: "short" })} →{" "}
   332→                {new Date(results[results.length - 1].periodEnd).toLocaleDateString("en-PK", { year: "numeric", month: "short" })}
   333→              </p>
   334→            </div>
   335→          </div>
   336→        </CardContent>
   337→      </Card>
   338→
   339→      {/* 8 summary stats */}
   340→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   341→        {stats.map((s, i) => (
   342→          <SummaryStat key={i} label={s.label} value={s.value} sub={s.sub} tone={s.tone} explanation={s.explanation} />
   343→        ))}
   344→      </div>
   345→
   346→      {/* Multi-line chart */}
   347→      <Card className="bg-[#131829] border-white/[0.08]">
   348→        <CardHeader>
   349→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   350→            <BarChart3 className="h-4 w-4 text-[#00E676]" />
   351→            Revenue · Profit · EPS trend (normalized 0–100)
   352→          </CardTitle>
   353→        </CardHeader>
   354→        <CardContent>
   355→          <MultiLineChart results={results} />
   356→        </CardContent>
   357→      </Card>
   358→
   359→      {/* Quarterly seasonal pattern */}
   360→      <Card className="bg-[#131829] border-white/[0.08]">
   361→        <CardHeader>
   362→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   363→            <Calendar className="h-4 w-4 text-[#00E676]" />
   364→            Quarterly seasonal pattern
   365→          </CardTitle>
   366→        </CardHeader>
   367→        <CardContent>
   368→          {quarters.length === 0 ? (
   369→            <p className="text-xs text-[#64748B]">
   370→              Need at least 4 quarters (ideally spanning multiple FYs) to compute seasonality.
   371→            </p>
   372→          ) : (
   373→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-3">
   374→              {quarters.map((q) => (
   375→                <QuarterSummaryCard key={q.quarter} q={q} />
   376→              ))}
   377→            </div>
   378→          )}
   379→        </CardContent>
   380→      </Card>
   381→
   382→      {/* Revenue bar chart (CSS) */}
   383→      <Card className="bg-[#131829] border-white/[0.08]">
   384→        <CardHeader>
   385→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   386→            <TrendingUp className="h-4 w-4 text-[#00E676]" />
   387→            Revenue by quarter (PKR, color-coded by FY)
   388→          </CardTitle>
   389→        </CardHeader>
   390→        <CardContent>
   391→          <RevenueBars results={results} />
   392→        </CardContent>
   393→      </Card>
   394→
   395→      {/* EPS bars + net margin overlay */}
   396→      <Card className="bg-[#131829] border-white/[0.08]">
   397→        <CardHeader>
   398→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   399→            <Coins className="h-4 w-4 text-[#00E676]" />
   400→            EPS bars + net margin overlay
   401→          </CardTitle>
   402→        </CardHeader>
   403→        <CardContent>
   404→          <EpsMarginChart results={results} />
   405→        </CardContent>
   406→      </Card>
   407→
   408→      {/* YoY growth chart */}
   409→      <Card className="bg-[#131829] border-white/[0.08]">
   410→        <CardHeader>
   411→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   412→            <Activity className="h-4 w-4 text-[#00E676]" />
   413→            YoY growth — Revenue vs EPS
   414→          </CardTitle>
   415→        </CardHeader>
   416→        <CardContent>
   417→          <YoYGrowthChart results={results} />
   418→        </CardContent>
   419→      </Card>
   420→
   421→      {/* Detected patterns */}
   422→      <Card className="bg-[#131829] border-white/[0.08]">
   423→        <CardHeader>
   424→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2 flex-wrap">
   425→            <Info className="h-4 w-4 text-[#00E676]" />
   426→            Detected patterns ({patterns.length})
   427→            {patterns.length > 0 && (
   428→              <span className="flex items-center gap-1.5 ml-auto text-[10px] font-normal">
   429→                {(() => {
   430→                  const bull = patterns.filter((p) => p.severity === "positive").length;
   431→                  const bear = patterns.filter((p) => p.severity === "negative").length;
   432→                  const neutr = patterns.filter((p) => p.severity === "neutral").length;
   433→                  return (
   434→                    <>
   435→                      {bull > 0 && (
   436→                        <span className="text-[#00E676] font-mono">{bull} bullish</span>
   437→                      )}
   438→                      {bear > 0 && (
   439→                        <span className="text-[#FF5C5C] font-mono">{bear} bearish</span>
   440→                      )}
   441→                      {neutr > 0 && (
   442→                        <span className="text-[#94A3B8] font-mono">{neutr} neutral</span>
   443→                      )}
   444→                      <span className="text-[#64748B]">·</span>

   446→                        {patterns.filter((p) => p.strength === "Strong").length} strong ·{" "}

   449→                      </span>

   458→          {patterns.length === 0 ? (
   459→            <p className="text-xs text-[#64748B]">
   460→              No strong patterns detected yet — need more quarters or larger moves to trigger detection rules.

   476→        </CardHeader>

   478→          <div className="max-h-96 overflow-y-auto custom-scroll rounded-md border border-white/[0.04]">
   479→            <table className="w-full text-xs">
   480→              <thead className="sticky top-0 bg-[#131829] z-10">

   719→// ---- Pure-SVG multi-line chart: revenue (green), profit (blue dashed), EPS (yellow dotted)
   720→function MultiLineChart({ results }: { results: QuarterlyResult[] }) {
   721→  const W = 800;
   722→  const H = 280;
   723→  const P = { top: 20, right: 30, bottom: 36, left: 40 };
   724→  const innerW = W - P.left - P.right;
   725→  const innerH = H - P.top - P.bottom;
   726→
   727→  const revs = results.map((r) => r.metrics.revenue);
   728→  const profits = results.map((r) => r.metrics.profitAfterTax);
   729→  const epss = results.map((r) => r.metrics.eps);
   730→
   731→  const rangeOf = (arr: number[]) => {
   732→    const mn = Math.min(...arr);
   733→    const mx = Math.max(...arr);
   734→    return { mn, mx: mx === mn ? mn + 1 : mx };
   735→  };
   736→  const revR = rangeOf(revs);
   737→  const profR = rangeOf(profits);
   738→  const epsR = rangeOf(epss);
   739→
   740→  const norm = (v: number, r: { mn: number; mx: number }) =>
   741→    r.mx === r.mn ? 0.5 : (v - r.mn) / (r.mx - r.mn);
   742→
   743→  const xAt = (i: number) =>
   744→    P.left + (results.length === 1 ? innerW / 2 : (i / (results.length - 1)) * innerW);
   745→  const yAt = (n: number) => P.top + (1 - n) * innerH;
   746→
   747→  const linePath = (vals: number[], r: { mn: number; mx: number }) =>
   748→    vals
   749→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   750→      .join(" ");
   751→
   752→  const areaPath = (vals: number[], r: { mn: number; mx: number }) => {
   753→    if (vals.length === 0) return "";
   754→    const top = vals
   755→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   756→      .join(" ");
   757→    return `${top} L ${xAt(vals.length - 1).toFixed(1)} ${(P.top + innerH).toFixed(1)} L ${xAt(0).toFixed(
   758→      1
   759→    )} ${(P.top + innerH).toFixed(1)} Z`;
   760→  };
   761→
   762→  const gridLines = Array.from({ length: 5 }, (_, i) => P.top + (i / 4) * innerH);
   763→  const labelStep = Math.max(1, Math.ceil(results.length / 8));
   764→
   765→  const lastRev = revs[revs.length - 1];
   766→  const lastProf = profits[profits.length - 1];
   767→  const lastEps = epss[epss.length - 1];
   768→
   769→  return (
   770→    <div className="w-full">
   771→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-auto" preserveAspectRatio="xMidYMid meet">
   772→        <defs>
   773→          <linearGradient id="revFill" x1="0" y1="0" x2="0" y2="1">
   774→            <stop offset="0%" stopColor="#00E676" stopOpacity="0.25" />
   775→            <stop offset="100%" stopColor="#00E676" stopOpacity="0" />
   776→          </linearGradient>
   777→        </defs>
   778→        {/* Grid */}
   779→        {gridLines.map((y, i) => (
   780→          <line
   781→            key={i}
   782→            x1={P.left}
   783→            y1={y}
   784→            x2={P.left + innerW}
   785→            y2={y}
   786→            stroke="#1E2536"
   787→            strokeWidth="1"
   788→            strokeDasharray="2 4"
   789→          />
   790→        ))}
   791→        {/* X-axis labels */}
   792→        {results.map((r, i) =>
   793→          i % labelStep === 0 ? (
   794→            <text
   795→              key={i}
   796→              x={xAt(i)}
   797→              y={H - 8}
   798→              fill="#64748B"
   799→              fontSize="9"
   800→              textAnchor="middle"
   801→              fontFamily="monospace"
   802→            >
   803→              {r.period}
   804→            </text>
   805→          ) : null
   806→        )}
   807→        {/* Filled area under revenue */}
   808→        <path d={areaPath(revs, revR)} fill="url(#revFill)" />
   809→        {/* Revenue line (green solid) */}
   810→        <path d={linePath(revs, revR)} fill="none" stroke="#00E676" strokeWidth="2" />
   811→        {/* Profit line (blue dashed) */}
   812→        <path d={linePath(profits, profR)} fill="none" stroke="#3B82F6" strokeWidth="2" strokeDasharray="6 4" />
   813→        {/* EPS line (yellow dotted) */}
   814→        <path d={linePath(epss, epsR)} fill="none" stroke="#FBBF24" strokeWidth="2" strokeDasharray="2 3" />
   815→        {/* Endpoint markers */}
   816→        {revs.length > 0 && (
   817→          <circle cx={xAt(revs.length - 1)} cy={yAt(norm(lastRev, revR))} r="3.5" fill="#00E676" />
   818→        )}
   819→        {profits.length > 0 && (
   820→          <circle cx={xAt(profits.length - 1)} cy={yAt(norm(lastProf, profR))} r="3.5" fill="#3B82F6" />
   821→        )}
   822→        {epss.length > 0 && (
   823→          <circle cx={xAt(epss.length - 1)} cy={yAt(norm(lastEps, epsR))} r="3.5" fill="#FBBF24" />
   824→        )}
   825→      </svg>
   826→      {/* Legend */}
   827→      <div className="flex items-center gap-4 mt-3 text-[10px] text-[#94A3B8] flex-wrap justify-center">
   828→        <span className="flex items-center gap-1.5">
   829→          <span className="h-0.5 w-4 bg-[#00E676] inline-block" />
   830→          Revenue (PKR)
   831→        </span>
   832→        <span className="flex items-center gap-1.5">
   833→          <span className="h-0.5 w-4 inline-block" style={{ borderBottom: "2px dashed #3B82F6" }} />

   999→    (r) => r.metrics.revenueYoYGrowth !== 0 || r.metrics.epsYoYGrowth !== 0
  1000→  );

  1003→  }

  1005→  const W = 800;

  1013→    ...items.flatMap((r) => [
  1014→      Math.abs(r.metrics.revenueYoYGrowth),
