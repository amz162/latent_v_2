    43→// ─── Issue 13: Close-trade calculation helpers ─────────────────────────────
    44→// Pakistan capital-gains tax on PSX equities (2024 budget):
    45→//   - holding period ≤ 12 months → 15% CGT on profit (short-term)
    46→//   - holding period > 12 months → 0% CGT (long-term, exempt)
    47→// Losses are not taxed (no tax credit) — CGT applies only on positive profit.
    48→// Broker commission: 0.15% × sell value × 2 (both sides — buy + sell).
    49→interface CloseTarget {
    50→  symbol: string;
    51→  name: string;
    52→  sector?: string | null;
    53→  buyPrice: number;
    54→  shares: number;
    55→  addedAt: string;
    56→}
    57→
    58→interface CloseCalc {
    59→  sellPrice: number;
    60→  buyCost: number;
    61→  sellValue: number;
    62→  commission: number;
    63→  holdingDays: number;
    64→  holdingMonths: number;
    65→  taxRate: number; // 0.15 short-term, 0 long-term
    66→  cgt: number;
    67→  grossPnl: number;
    68→  netPnl: number;
    69→  netPnlPct: number;
    70→}
    71→
    72→function computeCloseMetrics(
    73→  buyPrice: number,
    74→  shares: number,
    75→  sellPrice: number,
    76→  addedAt: string
    77→): CloseCalc {
    78→  const buyCost = buyPrice * shares;
    79→  const sellValue = sellPrice * shares;
    80→  // Broker commission = 0.15% × sell value, applied on both sides (buy + sell).
    81→  // Using sellValue as the base for both sides per spec ("0.15% × sell value (both sides)").
    82→  const commission = 0.0015 * sellValue * 2;
    83→  const holdingDays = Math.max(
    84→    0,
    85→    Math.floor((Date.now() - new Date(addedAt).getTime()) / (1000 * 60 * 60 * 24))
    86→  );
    87→  const holdingMonths = Math.floor(holdingDays / 30);
    88→  // CGT: 15% for ≤12 months (short-term), 0% for >12 months (long-term)
    89→  const taxRate = holdingDays <= 365 ? 0.15 : 0;
    90→  const grossPnl = sellValue - buyCost;
    91→  // CGT only on positive profit (no tax credit for losses)
    92→  const cgt = grossPnl > 0 ? grossPnl * taxRate : 0;
    93→  const netPnl = grossPnl - commission - cgt;
    94→  const netPnlPct = buyCost > 0 ? (netPnl / buyCost) * 100 : 0;
    95→  return {
    96→    sellPrice,
    97→    buyCost,
    98→    sellValue,
    99→    commission,
   100→    holdingDays,
   101→    holdingMonths,
   102→    taxRate,
   103→    cgt,
   104→    grossPnl,
   105→    netPnl,
   106→    netPnlPct,
   107→  };
   108→}

   110→// ─── Issue 15: batched LIVE quote fetcher for portfolio holdings ────────────
   111→// Hits our own /api/psx/live/batch endpoint, which server-side calls
   112→// `getLiveQuote` (60s in-memory cache per symbol) for each portfolio symbol
   113→// in parallel via Promise.all. Returns the latest LIVE PSX price for each
   114→// symbol, plus isLoading / refresh / lastUpdated for the UI.
   115→//
   116→// Used by PortfolioTab to:
   117→//   - render per-position live P&L (green/red) on open holdings (Issue 15)
   118→//   - compute the total portfolio P&L summary card (Issue 15)
   119→//   - show per-row live price on watch-only items (free upgrade)
   120→//   - skeleton-while-loading + manual Refresh button + 60s auto-refresh
   121→//
   122→// Implementation note: setState is only called inside async .then callbacks
   123→// (NOT synchronously in the effect body) to avoid React 19's
   124→// `react-hooks/set-state-in-effect` cascading-render warning. The "loading"
   125→// state is derived from `state.key !== key` (data is stale until the new
   126→// fetch for the current symbol-set resolves) plus an explicit `loading`
   127→// flag set inside the `refresh` event handler.
   128→function usePortfolioLiveQuotes(symbols: string[]): {
   129→  prices: Record<string, number>;
   130→  isLoading: boolean;
   131→  refresh: () => void;
   132→  lastUpdated: Date | null;
   133→} {
   134→  const key = symbols.join(",");
   135→  const [state, setState] = useState<{
   136→    key: string;
   137→    prices: Record<string, number>;
   138→    lastUpdated: Date | null;
   139→    loading: boolean;
   140→  }>({ key: "", prices: {}, lastUpdated: null, loading: false });
   141→  const [refreshTick, setRefreshTick] = useState(0);
   142→
   143→  useEffect(() => {
   144→    if (!key) return;
   145→    let cancelled = false;
   146→    const symbolsList = key.split(",");
   147→    fetch(`/api/psx/live/batch?symbols=${encodeURIComponent(symbolsList.join(","))}`)
   148→      .then((r) => (r.ok ? r.json() : null))
   149→      .then((d) => {
   150→        if (cancelled) return;
   151→        if (d && d.prices && typeof d.prices === "object") {
   152→          setState({
   153→            key,
   154→            prices: d.prices as Record<string, number>,
   155→            lastUpdated: new Date(d.fetchedAt || Date.now()),
   156→            loading: false,
   157→          });
   158→        } else {
   159→          // Even if no prices resolved, mark this key as "fetched" so
   160→          // isLoading flips to false and rows show "no live price" instead
   161→          // of a skeleton.
   162→          setState({ key, prices: {}, lastUpdated: new Date(), loading: false });
   163→        }
   164→      })
   165→      .catch(() => {
   166→        if (cancelled) return;
   167→        // Keep stale prices if any; mark this key as attempted so the
   168→        // skeleton goes away.
   169→        setState((s) => ({ key, prices: s.key === key ? s.prices : {}, lastUpdated: new Date(), loading: false }));
   170→      });
   171→    return () => {
   172→      cancelled = true;
   173→    };
   174→  }, [key, refreshTick]);
   175→
   176→  // Auto-refresh every 60s — matches getLiveQuote's 60s cache TTL so the
   177→  // dashboard P&L stays current without manual button clicks.
   178→  useEffect(() => {
   179→    if (!key) return;
   180→    const id = setInterval(() => setRefreshTick((t) => t + 1), 60_000);
   181→    return () => clearInterval(id);
   182→  }, [key]);
   183→
   184→  const refresh = useCallback(() => {
   185→    setRefreshTick((t) => t + 1);
   186→    setState((s) => ({ ...s, loading: true }));
   187→  }, []);
   188→
   189→  // Derive public state:
   190→  //   - prices: valid only when state.key === key (otherwise stale → empty)
   191→  //   - isLoading: true if we have no data yet for the current key OR a
   192→  //     refresh is explicitly in progress
   193→  const hasData = state.key === key;
   194→  const prices = hasData ? state.prices : {};
   195→  const lastUpdated = hasData ? state.lastUpdated : null;
   196→  const isLoading = !!key && (!hasData || state.loading);
   197→  return { prices, isLoading, refresh, lastUpdated };
   198→}

   200→export function PortfolioTab() {

   240→      type: "info",

   247→    setSelectedSymbol("");
   248→    setSelectedName("");
   249→    setSelectedSector(undefined);

   252→  };
   253→
   254→  const sorted = [...portfolio].sort((a, b) => a.order - b.order);
   255→
   256→  // ─── Issue 10: separate HOLDINGS from WATCHLIST ─────────────────────────
   257→  // A position is a "holding" ONLY if it has shares > 0 AND a non-null
   258→  // buyPrice (a real, open position). Watch-only items (shares=0 OR
   259→  // buyPrice=null) do NOT count toward the holdings summary / total P&L.
   260→  // The store distinguishes via the shares/buyPrice fields — see
   261→  // PortfolioPosition shape in src/lib/latent/store.ts.
   262→  const isHolding = (p: PortfolioPosition) =>
   263→    p.shares > 0 && p.buyPrice !== null && p.buyPrice !== undefined;
   264→  // Pre-compute (position, sortedIdx) pairs so each rendered row knows its
   265→  // original index in `sorted` for the reorder controls (moveUp/moveDown).
   266→  const holdingsWithIdx = useMemo(
   267→    () =>
   268→      sorted
   269→        .map((p, idx) => ({ p, idx }))
   270→        .filter(({ p }) => isHolding(p)),
   271→    [sorted]
   272→  );
   273→  const watchlistWithIdx = useMemo(
   274→    () =>
   275→      sorted
   276→        .map((p, idx) => ({ p, idx }))
   277→        .filter(({ p }) => !isHolding(p)),
   278→    [sorted]
   279→  );
   280→  const holdings = useMemo(() => holdingsWithIdx.map(({ p }) => p), [holdingsWithIdx]);
   281→  const watchlist = useMemo(() => watchlistWithIdx.map(({ p }) => p), [watchlistWithIdx]);

   283→  // Issue 15: fetch LIVE prices via getLiveQuote (batched server-side) for

   296→  // via /api/psx/live/batch) for the symbol, then re-fetches a fresh quote

   298→  // Preserves the close-trade flow: CGT + broker commission calc, holding

   300→  const openCloseDialog = (p: PortfolioPosition) => {

   304→    }

   315→    const cached = livePrices[p.symbol];
   316→    setCloseSellPrice(
   317→      typeof cached === "number"
   318→        ? cached.toFixed(2)
   319→        : (p.buyPrice as number).toFixed(2)
   320→    );
   321→
   322→    // Re-fetch a fresh live quote for the close-trade dialog so the user has
   323→    // the most recent fill price (uses getLiveQuote's 60s cache server-side).
   324→    setClosePriceLoading(true);
   325→    fetch(`/api/psx/live?symbol=${encodeURIComponent(p.symbol)}`)
   326→      .then((r) => (r.ok ? r.json() : null))
   327→      .then((d) => {
   328→        if (!d) return;
   329→        if (typeof d.price === "number" && d.price > 0) {
   330→          setCloseSellPrice(d.price.toFixed(2));
   331→        } else if (typeof d.close === "number" && d.close > 0) {
   332→          setCloseSellPrice(d.close.toFixed(2));
   333→        }
   334→      })
   335→      .catch(() => {
   336→        // ignore — user can manually enter the sell price
   337→      })
   338→      .finally(() => setClosePriceLoading(false));
   339→  };
   340→
   341→  // Compute close metrics reactively from current sellPrice input
   342→  const closeCalc = useMemo<CloseCalc | null>(() => {
   343→    if (!closeTarget) return null;
   344→    const sp = parseFloat(closeSellPrice.replace(/[,\s]/g, ""));
   345→    if (!Number.isFinite(sp) || sp <= 0) return null;
   346→    return computeCloseMetrics(
   347→      closeTarget.buyPrice,
   348→      closeTarget.shares,
   349→      sp,
   350→      closeTarget.addedAt
   351→    );
   352→  }, [closeTarget, closeSellPrice]);
   353→
   354→  const confirmCloseTrade = () => {
   355→    if (!closeTarget || !closeCalc) {
   356→      toast.error("Enter a valid sell price to close");
   357→      return;
   358→    }
   359→    setCloseSubmitting(true);
   360→    const t = closeTarget;
   361→    const c = closeCalc;
   362→    const closedTrade = {
   363→      id: `ct_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   364→      symbol: t.symbol,
   365→      name: t.name,
   366→      sector: t.sector ?? null,
   367→      buyPrice: t.buyPrice,
   368→      shares: t.shares,
   369→      sellPrice: c.sellPrice,
   370→      buyCost: c.buyCost,
   371→      sellValue: c.sellValue,
   372→      commission: c.commission,
   373→      cgt: c.cgt,
   374→      grossPnl: c.grossPnl,
   375→      netPnl: c.netPnl,
   376→      netPnlPct: c.netPnlPct,
   377→      holdingDays: c.holdingDays,
   378→      taxRate: c.taxRate,
   379→      closedAt: new Date().toISOString(),
   380→    };
   381→    closePositionFromPortfolio(t.symbol, closedTrade);
   382→    pushNotification({
   383→      type: "info",
   384→      title: `${t.symbol} closed — Net P&L PKR ${Math.round(c.netPnl).toLocaleString()}`,
   385→      body: `Sold ${t.shares} sh @ ${c.sellPrice.toFixed(2)} · commission PKR ${Math.round(c.commission).toLocaleString()} · CGT PKR ${Math.round(c.cgt).toLocaleString()} (${(c.taxRate * 100).toFixed(0)}% bracket) · holding ${c.holdingDays}d. Position reset to watch-only.`,
   386→      linkView: "tracker",
   387→    });
   388→    toast.success(
   389→      `${t.symbol} closed — Net P&L PKR ${Math.round(c.netPnl).toLocaleString()} (${c.netPnlPct >= 0 ? "+" : ""}${c.netPnlPct.toFixed(2)}%). Position reset to watch-only.`
   390→    );
   391→    setCloseTarget(null);
   392→    setCloseSellPrice("");
   393→    setCloseSubmitting(false);
   394→  };

   396→  // ─── Issue 15: Holdings-only totals (watchlist does NOT count) ──────────

   400→  const totalCost = useMemo(
   401→    () => holdings.reduce((sum, p) => sum + p.shares * (p.buyPrice as number), 0),
   402→    [holdings]
   403→  );
   404→  const totalValue = useMemo(
   405→    () =>
   406→      holdings.reduce((sum, p) => {
   407→        const live = livePrices[p.symbol];
   408→        if (typeof live === "number") return sum + p.shares * live;
   409→        // Fallback to MOCK_STOCKS price when fetch hasn't resolved or failed
   410→        const stock = MOCK_STOCKS.find((s) => s.symbol === p.symbol);
   411→        return stock ? sum + p.shares * stock.price : sum;
   412→      }, 0),
   413→    [holdings, livePrices]
   414→  );
   415→  const totalPnL = totalValue - totalCost;
   416→  const totalPnLPct = totalCost > 0 ? (totalPnL / totalCost) * 100 : 0;
   417→  const hasLivePrices = Object.keys(livePrices).length > 0;
   418→
   419→  const moveUp = (idx: number) => {
   420→    if (idx === 0) return;
   421→    reorder(idx, idx - 1);
   422→  };
   423→  const moveDown = (idx: number) => {
   424→    if (idx === sorted.length - 1) return;
   425→    reorder(idx, idx + 1);
   426→  };

   437→            </p>

   443→                Add stock

  1114→interface InterpretedNewsItem {
  1115→  title?: string;
  1116→  link?: string;
  1117→  source?: string;
  1118→  publishedAt?: string;
  1119→  summary?: string;
  1120→  symbol: string;
  1121→  impact: "positive" | "negative" | "neutral";
  1122→  impactStrength: "weak" | "moderate" | "strong";
  1123→  laymanExplanation: string;
  1124→  shortTermOutlook: string;
  1125→  affectedStocks: string[];
  1126→}
  1127→
  1128→interface NewsDigestEntry {
  1129→  symbol: string;
  1130→  stockName: string;
  1131→  newsCount: number;
  1132→  interpretedItems: InterpretedNewsItem[];
  1133→}
  1134→
  1135→function PortfolioNewsDigest({ symbols }: { symbols: string[] }) {
  1136→  const [digest, setDigest] = useState<NewsDigestEntry[]>([]);
  1137→  const [loading, setLoading] = useState(false);
  1138→  const [error, setError] = useState<string | null>(null);
  1139→  const setView = useLatentStore((s) => s.setView);
  1140→  const pushNotification = useLatentStore((s) => s.pushNotification);
  1141→
  1142→  const loadDigest = async () => {
  1143→    if (symbols.length === 0) return;
  1144→    setLoading(true);
  1145→    setError(null);
  1146→    try {
  1147→      const res = await fetch(`/api/portfolio/news-digest?symbols=${symbols.join(",")}&limit=2`);
  1148→      if (!res.ok) {
  1149→        const err = await res.json().catch(() => ({}));
  1150→        throw new Error(err.detail || err.error || `Failed (${res.status})`);
  1151→      }
  1152→      const data = await res.json();
  1153→      setDigest(data.digest || []);
  1154→      const total = data.totalInterpreted || 0;
  1155→      if (total > 0) {
  1156→        pushNotification({
  1157→          type: "news",
  1158→          title: `${total} news items interpreted`,
  1159→          body: `Auto-analyzed news for ${symbols.length} portfolio stock${symbols.length === 1 ? "" : "s"}`,
  1160→          linkView: "portfolio",
  1161→        });
  1162→      }
  1163→    } catch (err) {
  1164→      setError(err instanceof Error ? err.message : "Failed to load news digest");
  1165→    } finally {
  1166→      setLoading(false);
  1167→    }
  1168→  };
  1169→
  1170→  useEffect(() => {
  1171→    loadDigest();
  1172→  }, [symbols.join(",")]);
  1173→
  1174→  const impactColor = (i: string) =>
  1175→    i === "positive" ? "text-[#00E676]" : i === "negative" ? "text-[#FF5C5C]" : "text-[#94A3B8]";
  1176→
  1177→  return (
  1178→    <Card className="bg-[#131829] border-white/[0.08] mt-4">
  1179→      <CardContent className="p-5">
  1180→        <div className="flex items-center justify-between mb-4">
  1181→          <div>
  1182→            <h3 className="font-semibold text-[#F4F7FB] flex items-center gap-2">
  1183→              <TrendingUp className="h-4 w-4 text-[#00E676]" />
  1184→              Auto-News Digest for Your Portfolio
  1185→            </h3>
  1186→            <p className="text-xs text-[#94A3B8] mt-0.5">
  1187→              For each stock below: news fetched from AskAnalyst + GLM-4 interprets impact in layman terms.
  1188→            </p>
  1189→          </div>
  1190→          <Button
  1191→            variant="outline"
  1192→            size="sm"
  1193→            onClick={loadDigest}
  1194→            disabled={loading}
  1195→            className="border-white/[0.12] text-[#94A3B8] h-7"
  1196→          >
  1197→            {loading ? (
  1198→              <Loader2 className="h-3 w-3 mr-1.5 animate-spin" />
  1199→            ) : (
  1200→              <RefreshCw className="h-3 w-3 mr-1.5" />
  1201→            )}
  1202→            Refresh
  1203→          </Button>
  1204→        </div>
  1205→
  1206→        {loading && digest.length === 0 && (
  1207→          <div className="space-y-3">
  1208→            {[1, 2, 3].map((i) => <Skeleton key={i} className="h-24 w-full bg-white/[0.04]" />)}
  1209→          </div>
  1210→        )}
  1211→
  1212→        {error && (
  1213→          <div className="rounded-md bg-[#FF5C5C]/[0.07] border border-[#FF5C5C]/20 p-3 text-xs text-[#FF5C5C]">
  1214→            <span className="font-semibold">Error:</span> {error}
  1215→          </div>
  1216→        )}
  1217→
  1218→        {!loading && !error && digest.length > 0 && (
  1219→          <div className="space-y-4">
  1220→            {digest.map((entry, di) => (
  1221→              <div key={`${entry.symbol}-${di}`} className="border-l-2 border-[#00E676]/30 pl-3">
  1222→                <div className="flex items-center gap-2 mb-2">
  1223→                  <div className="h-7 w-7 rounded-md bg-[#00E676]/10 flex items-center justify-center font-mono font-bold text-[#00E676] text-[10px]">
  1224→                    {entry.symbol.slice(0, 4)}
  1225→                  </div>
  1226→                  <span className="font-semibold text-[#F4F7FB] text-sm">{entry.symbol}</span>
  1227→                  <span className="text-[11px] text-[#94A3B8] truncate flex-1">{entry.stockName}</span>
  1228→                  <Badge variant="outline" className="text-[9px] py-0 h-4 border-white/[0.12] text-[#64748B]">
