    42→// ─── Issue 13: Close-trade calculation helpers ─────────────────────────────
    43→// Pakistan capital-gains tax on PSX equities (2024 budget):
    44→//   - holding period ≤ 12 months → 15% CGT on profit (short-term)
    45→//   - holding period > 12 months → 0% CGT (long-term, exempt)
    46→// Losses are not taxed (no tax credit) — CGT applies only on positive profit.
    47→// Broker commission: 0.15% × sell value × 2 (both sides — buy + sell).
    48→interface CloseTarget {
    49→  symbol: string;
    50→  name: string;
    51→  sector?: string | null;
    52→  buyPrice: number;
    53→  shares: number;
    54→  addedAt: string;
    55→}
    56→
    57→interface CloseCalc {
    58→  sellPrice: number;
    59→  buyCost: number;
    60→  sellValue: number;
    61→  commission: number;
    62→  holdingDays: number;
    63→  holdingMonths: number;
    64→  taxRate: number; // 0.15 short-term, 0 long-term
    65→  cgt: number;
    66→  grossPnl: number;
    67→  netPnl: number;
    68→  netPnlPct: number;
    69→}
    70→
    71→function computeCloseMetrics(
    72→  buyPrice: number,
    73→  shares: number,
    74→  sellPrice: number,
    75→  addedAt: string
    76→): CloseCalc {
    77→  const buyCost = buyPrice * shares;
    78→  const sellValue = sellPrice * shares;
    79→  // Broker commission = 0.15% × sell value, applied on both sides (buy + sell).
    80→  // Using sellValue as the base for both sides per spec ("0.15% × sell value (both sides)").
    81→  const commission = 0.0015 * sellValue * 2;
    82→  const holdingDays = Math.max(
    83→    0,
    84→    Math.floor((Date.now() - new Date(addedAt).getTime()) / (1000 * 60 * 60 * 24))
    85→  );
    86→  const holdingMonths = Math.floor(holdingDays / 30);
    87→  // CGT: 15% for ≤12 months (short-term), 0% for >12 months (long-term)
    88→  const taxRate = holdingDays <= 365 ? 0.15 : 0;
    89→  const grossPnl = sellValue - buyCost;
    90→  // CGT only on positive profit (no tax credit for losses)
    91→  const cgt = grossPnl > 0 ? grossPnl * taxRate : 0;
    92→  const netPnl = grossPnl - commission - cgt;
    93→  const netPnlPct = buyCost > 0 ? (netPnl / buyCost) * 100 : 0;
    94→  return {
    95→    sellPrice,
    96→    buyCost,
    97→    sellValue,
    98→    commission,
    99→    holdingDays,
   100→    holdingMonths,
   101→    taxRate,
   102→    cgt,
   103→    grossPnl,
   104→    netPnl,
   105→    netPnlPct,
   106→  };
   107→}

   109→// ─── Issue 15: batched LIVE quote fetcher for portfolio holdings ────────────
   110→// Hits our own /api/psx/live/batch endpoint, which server-side calls
   111→// `getLiveQuote` (60s in-memory cache per symbol) for each portfolio symbol
   112→// in parallel via Promise.all. Returns the latest LIVE PSX price for each
   113→// symbol, plus isLoading / refresh / lastUpdated for the UI.
   114→//
   115→// Used by PortfolioTab to:
   116→//   - render per-position live P&L (green/red) on open holdings (Issue 15)
   117→//   - compute the total portfolio P&L summary card (Issue 15)
   118→//   - show per-row live price on watch-only items (free upgrade)
   119→//   - skeleton-while-loading + manual Refresh button + 60s auto-refresh
   120→//
   121→// Implementation note: setState is only called inside async .then callbacks
   122→// (NOT synchronously in the effect body) to avoid React 19's
   123→// `react-hooks/set-state-in-effect` cascading-render warning. The "loading"
   124→// state is derived from `state.key !== key` (data is stale until the new
   125→// fetch for the current symbol-set resolves) plus an explicit `loading`
   126→// flag set inside the `refresh` event handler.
   127→function usePortfolioLiveQuotes(symbols: string[]): {
   128→  prices: Record<string, number>;
   129→  isLoading: boolean;
   130→  refresh: () => void;
   131→  lastUpdated: Date | null;
   132→} {
   133→  const key = symbols.join(",");
   134→  const [state, setState] = useState<{
   135→    key: string;
   136→    prices: Record<string, number>;
   137→    lastUpdated: Date | null;
   138→    loading: boolean;
   139→  }>({ key: "", prices: {}, lastUpdated: null, loading: false });
   140→  const [refreshTick, setRefreshTick] = useState(0);
   141→
   142→  useEffect(() => {
   143→    if (!key) return;
   144→    let cancelled = false;
   145→    const symbolsList = key.split(",");
   146→    fetch(`/api/psx/live/batch?symbols=${encodeURIComponent(symbolsList.join(","))}`)
   147→      .then((r) => (r.ok ? r.json() : null))
   148→      .then((d) => {
   149→        if (cancelled) return;
   150→        if (d && d.prices && typeof d.prices === "object") {
   151→          setState({
   152→            key,
   153→            prices: d.prices as Record<string, number>,
   154→            lastUpdated: new Date(d.fetchedAt || Date.now()),
   155→            loading: false,
   156→          });
   157→        } else {
   158→          // Even if no prices resolved, mark this key as "fetched" so
   159→          // isLoading flips to false and rows show "no live price" instead
   160→          // of a skeleton.
   161→          setState({ key, prices: {}, lastUpdated: new Date(), loading: false });
   162→        }
   163→      })
   164→      .catch(() => {
   165→        if (cancelled) return;
   166→        // Keep stale prices if any; mark this key as attempted so the
   167→        // skeleton goes away.
   168→        setState((s) => ({ key, prices: s.key === key ? s.prices : {}, lastUpdated: new Date(), loading: false }));
   169→      });
   170→    return () => {
   171→      cancelled = true;
   172→    };
   173→  }, [key, refreshTick]);
   174→
   175→  // Auto-refresh every 60s — matches getLiveQuote's 60s cache TTL so the
   176→  // dashboard P&L stays current without manual button clicks.
   177→  useEffect(() => {
   178→    if (!key) return;
   179→    const id = setInterval(() => setRefreshTick((t) => t + 1), 60_000);
   180→    return () => clearInterval(id);
   181→  }, [key]);
   182→
   183→  const refresh = useCallback(() => {
   184→    setRefreshTick((t) => t + 1);
   185→    setState((s) => ({ ...s, loading: true }));
   186→  }, []);
   187→
   188→  // Derive public state:
   189→  //   - prices: valid only when state.key === key (otherwise stale → empty)
   190→  //   - isLoading: true if we have no data yet for the current key OR a
   191→  //     refresh is explicitly in progress
   192→  const hasData = state.key === key;
   193→  const prices = hasData ? state.prices : {};
   194→  const lastUpdated = hasData ? state.lastUpdated : null;
   195→  const isLoading = !!key && (!hasData || state.loading);
   196→  return { prices, isLoading, refresh, lastUpdated };
   197→}

   255→  // ─── Issue 10: separate HOLDINGS from WATCHLIST ─────────────────────────
   256→  // A position is a "holding" ONLY if it has shares > 0 AND a non-null

   266→    () =>
   267→      sorted
   268→        .map((p, idx) => ({ p, idx }))

   332→        }
   333→      })
   334→      .catch(() => {
   335→        // ignore — user can manually enter the sell price

   337→      .finally(() => setClosePriceLoading(false));

   347→      closeTarget.shares,
   348→      sp,
   349→      closeTarget.addedAt

   351→  }, [closeTarget, closeSellPrice]);

   353→  const confirmCloseTrade = () => {

   359→    const t = closeTarget;

   362→      id: `ct_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   363→      symbol: t.symbol,
   364→      name: t.name,
   365→      sector: t.sector ?? null,
   366→      buyPrice: t.buyPrice,
   367→      shares: t.shares,
   368→      sellPrice: c.sellPrice,

   371→      commission: c.commission,
   372→      cgt: c.cgt,
   373→      grossPnl: c.grossPnl,
   374→      netPnl: c.netPnl,
   375→      netPnlPct: c.netPnlPct,
   376→      holdingDays: c.holdingDays,
   377→      taxRate: c.taxRate,
   378→      closedAt: new Date().toISOString(),
   379→    };
   380→    closePositionFromPortfolio(t.symbol, closedTrade);
   381→    pushNotification({
   382→      type: "info",
   383→      title: `${t.symbol} closed — Net P&L PKR ${Math.round(c.netPnl).toLocaleString()}`,
   384→      body: `Sold ${t.shares} sh @ ${c.sellPrice.toFixed(2)} · commission PKR ${Math.round(c.commission).toLocaleString()} · CGT PKR ${Math.round(c.cgt).toLocaleString()} (${(c.taxRate * 100).toFixed(0)}% bracket) · holding ${c.holdingDays}d. Position reset to watch-only.`,
   385→      linkView: "tracker",
   386→    });
   387→    toast.success(
   388→      `${t.symbol} closed — Net P&L PKR ${Math.round(c.netPnl).toLocaleString()} (${c.netPnlPct >= 0 ? "+" : ""}${c.netPnlPct.toFixed(2)}%). Position reset to watch-only.`
   389→    );
   390→    setCloseTarget(null);
   391→    setCloseSellPrice("");
   392→    setCloseSubmitting(false);

   401→    [holdings]
   402→  );

   404→    () =>

   410→        return stock ? sum + p.shares * stock.price : sum;
   411→      }, 0),
   412→    [holdings, livePrices]
   413→  );
   414→  const totalPnL = totalValue - totalCost;
   415→  const totalPnLPct = totalCost > 0 ? (totalPnL / totalCost) * 100 : 0;
   416→  const hasLivePrices = Object.keys(livePrices).length > 0;
   417→
   418→  const moveUp = (idx: number) => {
   419→    if (idx === 0) return;
   420→    reorder(idx, idx - 1);
   421→  };
   422→  const moveDown = (idx: number) => {
   423→    if (idx === sorted.length - 1) return;
   424→    reorder(idx, idx + 1);
   425→  };

  1112→interface InterpretedNewsItem {
  1113→  title?: string;
  1114→  link?: string;
  1115→  source?: string;
  1116→  publishedAt?: string;
  1117→  summary?: string;
  1118→  symbol: string;
  1119→  impact: "positive" | "negative" | "neutral";
  1120→  impactStrength: "weak" | "moderate" | "strong";
  1121→  laymanExplanation: string;
  1122→  shortTermOutlook: string;
  1123→  affectedStocks: string[];
  1124→}
  1125→
  1126→interface NewsDigestEntry {
  1127→  symbol: string;
  1128→  stockName: string;
  1129→  newsCount: number;
  1130→  interpretedItems: InterpretedNewsItem[];
  1131→}
  1132→
  1133→function PortfolioNewsDigest({ symbols }: { symbols: string[] }) {
  1134→  const [digest, setDigest] = useState<NewsDigestEntry[]>([]);
  1135→  const [loading, setLoading] = useState(false);
  1136→  const [error, setError] = useState<string | null>(null);
  1137→  const setView = useLatentStore((s) => s.setView);
  1138→  const pushNotification = useLatentStore((s) => s.pushNotification);
  1139→
  1140→  const loadDigest = async () => {
  1141→    if (symbols.length === 0) return;
  1142→    setLoading(true);
  1143→    setError(null);
  1144→    try {
  1145→      const res = await fetch(`/api/portfolio/news-digest?symbols=${symbols.join(",")}&limit=2`);
  1146→      if (!res.ok) {
  1147→        const err = await res.json().catch(() => ({}));
  1148→        throw new Error(err.detail || err.error || `Failed (${res.status})`);
  1149→      }
  1150→      const data = await res.json();
  1151→      setDigest(data.digest || []);
  1152→      const total = data.totalInterpreted || 0;
  1153→      if (total > 0) {
  1154→        pushNotification({
  1155→          type: "news",
  1156→          title: `${total} news items interpreted`,
  1157→          body: `Auto-analyzed news for ${symbols.length} portfolio stock${symbols.length === 1 ? "" : "s"}`,
  1158→          linkView: "portfolio",
  1159→        });
  1160→      }
  1161→    } catch (err) {
  1162→      setError(err instanceof Error ? err.message : "Failed to load news digest");
  1163→    } finally {
  1164→      setLoading(false);
  1165→    }
  1166→  };
  1167→
  1168→  useEffect(() => {
  1169→    loadDigest();
  1170→  }, [symbols.join(",")]);
  1171→
  1172→  const impactColor = (i: string) =>
  1173→    i === "positive" ? "text-[#00E676]" : i === "negative" ? "text-[#FF5C5C]" : "text-[#94A3B8]";
  1174→
  1175→  return (
  1176→    <Card className="bg-[#131829] border-white/[0.08] mt-4">
  1177→      <CardContent className="p-5">
  1178→        <div className="flex items-center justify-between mb-4">
  1179→          <div>
  1180→            <h3 className="font-semibold text-[#F4F7FB] flex items-center gap-2">
  1181→              <TrendingUp className="h-4 w-4 text-[#00E676]" />
  1182→              Auto-News Digest for Your Portfolio
  1183→            </h3>
  1184→            <p className="text-xs text-[#94A3B8] mt-0.5">
  1185→              For each stock below: news fetched from AskAnalyst + GLM-4 interprets impact in layman terms.
  1186→            </p>
  1187→          </div>
  1188→          <Button
  1189→            variant="outline"
  1190→            size="sm"
  1191→            onClick={loadDigest}
  1192→            disabled={loading}
  1193→            className="border-white/[0.12] text-[#94A3B8] h-7"
  1194→          >
  1195→            {loading ? (
  1196→              <Loader2 className="h-3 w-3 mr-1.5 animate-spin" />
  1197→            ) : (
  1198→              <RefreshCw className="h-3 w-3 mr-1.5" />
  1199→            )}
  1200→            Refresh
  1201→          </Button>
  1202→        </div>
  1203→
  1204→        {loading && digest.length === 0 && (
  1205→          <div className="space-y-3">
  1206→            {[1, 2, 3].map((i) => <Skeleton key={i} className="h-24 w-full bg-white/[0.04]" />)}
  1207→          </div>
  1208→        )}
  1209→
  1210→        {error && (
  1211→          <div className="rounded-md bg-[#FF5C5C]/[0.07] border border-[#FF5C5C]/20 p-3 text-xs text-[#FF5C5C]">
  1212→            <span className="font-semibold">Error:</span> {error}
  1213→          </div>
  1214→        )}
  1215→
  1216→        {!loading && !error && digest.length > 0 && (
  1217→          <div className="space-y-4">
  1218→            {digest.map((entry, di) => (

  1222→                    {entry.symbol.slice(0, 4)}
  1223→                  </div>
