    49→    49→// Parse the dividend amount (PKR per share) from an announcement title.
    50→    50→// Handles patterns like: "Rs. 4.50 per share", "PKR 6.0/share", "10.5 per share",
    51→    51→// "Final Dividend Rs.7.50 per share", etc.
    52→    52→function parseDividendAmount(title: string): number | null {
    53→    53→  const t = title.toLowerCase();
    54→    54→  // Negative lookahead-free patterns ordered most-specific → least-specific
    55→    55→  const patterns: RegExp[] = [
    56→    56→    /(?:rs\.?|pkr|rupees)\s*\=?\s*(\d+(?:\.\d+)?)\s*(?:per\s*share|\/\s*share|p\.?s\.?|ps)?/i,
    57→    57→    /(\d+(?:\.\d+)?)\s*(?:per\s*share|\/\s*share)/i,
    58→    58→    /dividend[^0-9]*(\d+(?:\.\d+)?)/i,
    59→    59→    /(\d+(?:\.\d+)?)\s*(?:dividend|payout)/i,
    60→    60→  ];
    61→    61→  for (const p of patterns) {
    62→    62→    const m = title.match(p);
    63→    63→    if (m && m[1]) {
    64→    64→      const amt = parseFloat(m[1]);
    65→    65→      if (!isNaN(amt) && amt > 0 && amt < 1000) return amt;
    66→    66→    }
    67→    67→  }
    68→    68→  // Also check summary field if no match in title (handled by caller)
    69→    69→  void t;
    70→    70→  return null;
    71→    71→}
    72→    72→
    73→    73→function classifyType(title: string): DividendRow["type"] {
    74→    74→  const t = title.toLowerCase();
    75→    75→  if (/final/.test(t)) return "Final";
    76→    76→  if (/interim/.test(t)) return "Interim";
    77→    77→  if (/special/.test(t)) return "Special";
    78→    78→  if (/bonus/.test(t)) return "Bonus";
    79→    79→  return "Unknown";
    80→    80→}
