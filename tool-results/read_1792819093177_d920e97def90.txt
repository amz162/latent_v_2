   123→function fmtPrice(v: number | null | undefined): string {
   124→  if (v === null || v === undefined || !isFinite(v)) return "—";
   125→  return `Rs. ${v.toFixed(2)}`;
   126→}
   127→function fmtVolume(v: number): string {
   128→  if (v >= 1e9) return `${(v / 1e9).toFixed(2)}B`;
   129→  if (v >= 1e6) return `${(v / 1e6).toFixed(2)}M`;
   130→  if (v >= 1e3) return `${(v / 1e3).toFixed(1)}K`;
   131→  return v.toFixed(0);
   132→}

   153→  const [tab, setTab] = useState<"results" | "stock">("results");
   154→
   155→  // ── Tab 1 state: quarterly RESULT patterns ──
   156→  const [results, setResults] = useState<QuarterlyResult[]>([]);
   157→  const [company, setCompany] = useState<string>("");
   158→  const [loading, setLoading] = useState(false);
   159→  const [error, setError] = useState<string | null>(null);
   160→
   161→  // ── Tab 2 state: stock PRICE patterns ──
   162→  const [stockData, setStockData] = useState<StockPatternResponse | null>(null);
   163→  const [stockLoading, setStockLoading] = useState(false);
   164→  const [stockError, setStockError] = useState<string | null>(null);
   165→
   166→  const loadResults = (symbol: string) => {
   167→    setLoading(true);
   168→    setError(null);
   169→    setResults([]);
   170→    fetch(`/api/psx/financial-results/${encodeURIComponent(symbol)}`)
   171→      .then((r) => r.json())
   172→      .then((d) => {
   173→        if (d.error) throw new Error(d.error);
   174→        if (!d.results || d.results.length === 0) {
   175→          throw new Error("No quarterly data available for this stock yet.");
   176→        }
   177→        setResults(d.results as QuarterlyResult[]);
   178→        setCompany(d.company || symbol);
   179→        toast.success(`Loaded ${d.results.length} quarters for ${symbol}`);
   180→      })

   182→        const msg = err instanceof Error ? err.message : "Unknown error";
   183→        setError(msg);
   184→        toast.error(`Could not load ${symbol}: ${msg}`);

   186→      .finally(() => setLoading(false));
   187→  };
   188→
   189→  const loadStockPatterns = (symbol: string) => {

   203→        const msg = err instanceof Error ? err.message : "Unknown error";
   204→        setStockError(msg);
   205→        toast.error(`Could not analyze ${symbol}: ${msg}`);

   208→  };

   213→      setResults([]);
   214→      setCompany("");

   216→      setStockData(null);

   218→      return;

   474→function StockDeepDive({
   475→  results,
   476→  symbol,
   477→  company,
   478→  onBack,
   479→}: {
   480→  results: QuarterlyResult[];
   481→  symbol: string;
   482→  company: string;
   483→  onBack: () => void;
   484→}) {
   485→  const patterns = detectPatterns(results);
   486→  const quarters = generateQuarterlySummary(results, symbol);
   487→  const yearsBetween =
   488→    results.length >= 2
   489→      ? (new Date(results[results.length - 1].periodEnd).getTime() -
   490→          new Date(results[0].periodEnd).getTime()) /
   491→        (365 * 24 * 60 * 60 * 1000)
   492→      : 0;
   493→  const revCAGR = computeCAGR(results[0].metrics.revenue, results[results.length - 1].metrics.revenue, yearsBetween);
   494→  const profCAGR = computeCAGR(
   495→    results[0].metrics.profitAfterTax,
   496→    results[results.length - 1].metrics.profitAfterTax,
   497→    yearsBetween
   498→  );
   499→  const epsCAGR = computeCAGR(results[0].metrics.eps, results[results.length - 1].metrics.eps, yearsBetween);
   500→  const totalDPS = results.reduce((s, r) => s + r.metrics.dividendPerShare, 0);
   501→  const volatility = computeStdDev(results.map((r) => r.metrics.epsYoYGrowth));
   502→  const revVolatility = computeStdDev(results.map((r) => r.metrics.revenueYoYGrowth));
   503→  const marginSlope = computeSlope(results.map((r) => r.metrics.netMargin));
   504→
   505→  const stats: { label: string; value: string; sub: string; tone?: "pos" | "neg" | "neutral"; explanation: string }[] = [

   815→function PatternRow({ pattern }: { pattern: DetectedPattern & { category?: string } }) {
   816→  const isPositive = pattern.severity === "positive";
   817→  const isNegative = pattern.severity === "negative";
   818→  // Tab 2 (stock patterns) passes an extra `category` field — render it as a
   819→  // small gray badge so the user can see which pattern family it belongs to.
   820→  const CATEGORY_LABELS: Record<string, string> = {
   821→    trend: "Trend",
   822→    momentum: "Momentum",
   823→    breakout: "Breakout",
   824→    structure: "Structure",
   825→    volatility: "Volatility",
   826→    volume: "Volume",
   827→    candlestick: "Candlestick",
   828→  };
   829→  const icon = isPositive ? (
   830→    <CheckCircle2 className="h-3.5 w-3.5 text-[#00E676]" />
   831→  ) : isNegative ? (
   832→    <AlertTriangle className="h-3.5 w-3.5 text-[#FF5C5C]" />
   833→  ) : (
   834→    <Info className="h-3.5 w-3.5 text-[#94A3B8]" />
   835→  );
   836→  const sevClass = isPositive
   837→    ? "border-[#00E676]/30 text-[#00E676] bg-[#00E676]/[0.06]"
   838→    : isNegative
   839→    ? "border-[#FF5C5C]/30 text-[#FF5C5C] bg-[#FF5C5C]/[0.06]"
   840→    : "border-white/[0.08] text-[#94A3B8] bg-white/[0.02]";
   841→  const borderClass = isPositive
   842→    ? "border-[#00E676]/20"
   843→    : isNegative
   844→    ? "border-[#FF5C5C]/20"
   845→    : "border-white/[0.06]";
   846→
   847→  // Issue 9: strength tier — Strong / Moderate / Weak — colour-coded so the
   848→  // user can scan a list of patterns at a glance and pick out only the ones
   849→  // that genuinely matter. Strong = solid green/red; Moderate = softer tint;
   850→  // Weak = muted gray so it visually de-emphasises less informative signals.
   851→  const strength: PatternStrength | undefined = pattern.strength;
   852→  const strengthClass =
   853→    strength === "Strong"
   854→      ? isPositive
   855→        ? "border-[#00E676] bg-[#00E676] text-[#0A0E1A] font-bold"
   856→        : "border-[#FF5C5C] bg-[#FF5C5C] text-[#0A0E1A] font-bold"
   857→      : strength === "Moderate"
   858→      ? isPositive
   859→        ? "border-[#00E676]/50 text-[#00E676] bg-[#00E676]/[0.12]"
   860→        : "border-[#FF5C5C]/50 text-[#FF5C5C] bg-[#FF5C5C]/[0.12]"
   861→      : "border-white/[0.12] text-[#94A3B8] bg-white/[0.04]";
   862→
   863→  // Issue 9: confidence bar — visualises the 0-100 score alongside the
   864→  // discrete strength tier, so the user sees both signals.
   865→  const confPct = Math.max(0, Math.min(100, Math.round(pattern.confidence)));
   866→  const confColor = isPositive ? "#00E676" : isNegative ? "#FF5C5C" : "#94A3B8";
   867→
   868→  return (
   869→    <div className={`rounded-lg border ${borderClass} bg-[#0A0E1A] p-3`}>
   870→      <div className="flex items-start justify-between gap-2 mb-1.5">
   871→        <div className="flex items-center gap-2 min-w-0">
   872→          {icon}
   873→          <span className="text-sm font-semibold text-[#F4F7FB] truncate">{pattern.label}</span>
   874→        </div>
   875→        <div className="flex items-center gap-1.5 shrink-0 flex-wrap justify-end">
   876→          {pattern.category && (
   877→            <Badge variant="outline" className="text-[9px] uppercase border-white/[0.08] text-[#64748B]">
   878→              {CATEGORY_LABELS[pattern.category] || pattern.category}
   879→            </Badge>
   880→          )}
   881→          {strength && (
   882→            <Badge variant="outline" className={`text-[9px] uppercase ${strengthClass}`}>
   883→              {strength}
   884→            </Badge>
   885→          )}
   886→          <Badge variant="outline" className={`text-[9px] uppercase ${sevClass}`}>
   887→            {isPositive ? "bullish" : isNegative ? "bearish" : "neutral"}
   888→          </Badge>
   889→        </div>
   890→      </div>
   891→
   892→      {/* Detail + confidence bar */}
   893→      <p className="text-xs text-[#94A3B8] mb-2 leading-relaxed">{pattern.detail}</p>
   894→      <div className="flex items-center gap-2 mb-2">
   895→        <div className="flex-1 h-1.5 rounded-full bg-white/[0.05] overflow-hidden">
   896→          <div
   897→            className="h-full rounded-full transition-all"
   898→            style={{ width: `${confPct}%`, backgroundColor: confColor }}
   899→          />
   900→        </div>
   901→        <span className="text-[9px] font-mono text-[#64748B] shrink-0">{confPct}% conf</span>
   902→      </div>
   903→
   904→      {/* Issue 9: supporting data points — small grid the user can verify at a glance */}
   905→      {pattern.supportingData && pattern.supportingData.length > 0 && (
   906→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5 mb-2 rounded-md bg-white/[0.02] border border-white/[0.04] p-2">
   907→          {pattern.supportingData.map((d: SupportingDataPoint, i: number) => (
   908→            <div key={i} className="flex flex-col min-w-0">
   909→              <span className="text-[9px] uppercase tracking-wider text-[#64748B] truncate">{d.label}</span>
   910→              <span className="text-[11px] font-mono text-[#F4F7FB] truncate" title={d.value}>{d.value}</span>
   911→            </div>
   912→          ))}
   913→        </div>
   914→      )}
   915→
   916→      {pattern.laymanSummary && (
   917→        <div className="mb-1.5 rounded-md bg-[#00E676]/[0.05] border border-[#00E676]/20 p-2 flex items-start gap-2 leading-relaxed">
   918→          <Info className="h-3.5 w-3.5 mt-0.5 shrink-0 text-[#00E676]" />
   919→          <div>
   920→            <span className="text-[9px] uppercase tracking-wider text-[#00E676] font-semibold block mb-0.5">
   921→              In plain English
   922→            </span>
   923→            <p className="text-xs text-[#00E676] font-medium">{pattern.laymanSummary}</p>
   924→          </div>
   925→        </div>
   926→      )}
   927→      {pattern.affectedQuarters.length > 0 && (
   928→        <div className="flex items-center gap-1 flex-wrap mt-2 pt-2 border-t border-white/[0.04]">
   929→          <span className="text-[10px] text-[#64748B]">Triggered:</span>
   930→          {pattern.affectedQuarters.map((q, i) => (
   931→            <Badge key={i} variant="outline" className="text-[9px] font-mono border-white/[0.08] text-[#64748B]">
   932→              {q}
   933→            </Badge>
   934→          ))}
   935→        </div>

   941→// ---- Pure-SVG multi-line chart: revenue (green), profit (blue dashed), EPS (yellow dotted)
   942→function MultiLineChart({ results }: { results: QuarterlyResult[] }) {
   943→  const W = 800;
   944→  const H = 280;
   945→  const P = { top: 20, right: 30, bottom: 36, left: 40 };
   946→  const innerW = W - P.left - P.right;
   947→  const innerH = H - P.top - P.bottom;
   948→
   949→  const revs = results.map((r) => r.metrics.revenue);
   950→  const profits = results.map((r) => r.metrics.profitAfterTax);
   951→  const epss = results.map((r) => r.metrics.eps);
   952→
   953→  const rangeOf = (arr: number[]) => {
   954→    const mn = Math.min(...arr);
   955→    const mx = Math.max(...arr);
   956→    return { mn, mx: mx === mn ? mn + 1 : mx };
   957→  };
   958→  const revR = rangeOf(revs);
   959→  const profR = rangeOf(profits);
   960→  const epsR = rangeOf(epss);
   961→
   962→  const norm = (v: number, r: { mn: number; mx: number }) =>
   963→    r.mx === r.mn ? 0.5 : (v - r.mn) / (r.mx - r.mn);
   964→
   965→  const xAt = (i: number) =>
   966→    P.left + (results.length === 1 ? innerW / 2 : (i / (results.length - 1)) * innerW);
   967→  const yAt = (n: number) => P.top + (1 - n) * innerH;
   968→
   969→  const linePath = (vals: number[], r: { mn: number; mx: number }) =>
   970→    vals
   971→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   972→      .join(" ");

   974→  const areaPath = (vals: number[], r: { mn: number; mx: number }) => {
   975→    if (vals.length === 0) return "";

   977→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   978→      .join(" ");
   979→    return `${top} L ${xAt(vals.length - 1).toFixed(1)} ${(P.top + innerH).toFixed(1)} L ${xAt(0).toFixed(
   980→      1
   981→    )} ${(P.top + innerH).toFixed(1)} Z`;
   982→  };
   983→
   984→  const gridLines = Array.from({ length: 5 }, (_, i) => P.top + (i / 4) * innerH);
   985→  const labelStep = Math.max(1, Math.ceil(results.length / 8));
   986→
   987→  const lastRev = revs[revs.length - 1];
   988→  const lastProf = profits[profits.length - 1];
   989→  const lastEps = epss[epss.length - 1];
   990→
   991→  return (
   992→    <div className="w-full">
   993→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-auto" preserveAspectRatio="xMidYMid meet">

   995→          <linearGradient id="revFill" x1="0" y1="0" x2="0" y2="1">
   996→            <stop offset="0%" stopColor="#00E676" stopOpacity="0.25" />
   997→            <stop offset="100%" stopColor="#00E676" stopOpacity="0" />
   998→          </linearGradient>
   999→        </defs>

  1001→        {gridLines.map((y, i) => (
  1002→          <line
  1003→            key={i}
  1004→            x1={P.left}
  1005→            y1={y}
  1006→            x2={P.left + innerW}
  1007→            y2={y}
  1008→            stroke="#1E2536"
  1009→            strokeWidth="1"
  1010→            strokeDasharray="2 4"
  1011→          />
  1012→        ))}
  1013→        {/* X-axis labels */}
  1014→        {results.map((r, i) =>
  1015→          i % labelStep === 0 ? (
  1016→            <text
  1017→              key={i}
  1018→              x={xAt(i)}
  1019→              y={H - 8}
  1020→              fill="#64748B"
  1021→              fontSize="9"
  1022→              textAnchor="middle"
  1023→              fontFamily="monospace"
  1024→            >
  1025→              {r.period}
  1026→            </text>
  1027→          ) : null
  1028→        )}
  1029→        {/* Filled area under revenue */}
  1030→        <path d={areaPath(revs, revR)} fill="url(#revFill)" />
  1031→        {/* Revenue line (green solid) */}
  1032→        <path d={linePath(revs, revR)} fill="none" stroke="#00E676" strokeWidth="2" />
  1033→        {/* Profit line (blue dashed) */}
  1034→        <path d={linePath(profits, profR)} fill="none" stroke="#3B82F6" strokeWidth="2" strokeDasharray="6 4" />
  1035→        {/* EPS line (yellow dotted) */}

  1037→        {/* Endpoint markers */}
  1038→        {revs.length > 0 && (
  1039→          <circle cx={xAt(revs.length - 1)} cy={yAt(norm(lastRev, revR))} r="3.5" fill="#00E676" />

  1656→// ── Price chart: SVG line chart with SMA20/50/200 overlays + volume bars.
  1657→// Also draws dashed support/resistance lines when they lie within the range.
  1658→function PriceChart({
  1659→  chart,
  1660→  support,
  1661→  resistance,
  1662→}: {
  1663→  chart: StockChartPoint[];
  1664→  support: { price: number; touches: number; lastTouch: string } | null;
  1665→  resistance: { price: number; touches: number; lastTouch: string } | null;
  1666→}) {
  1667→  const pts = chart.slice(-180);
  1668→  if (pts.length === 0) return <p className="text-xs text-[#64748B]">No chart data.</p>;
  1669→
  1670→  const W = 800;
  1671→  const H = 360;
  1672→  const P = { top: 16, right: 62, bottom: 30, left: 52 };
  1673→  const innerW = W - P.left - P.right;
  1674→  const priceH = 240;
  1675→  const volH = 54;
  1676→  const volTop = P.top + priceH + 14;
  1677→
  1678→  const closes = pts.map((p) => p.close);
  1679→  const allVals = closes.concat(
  1680→    pts.map((p) => p.sma20).filter((v): v is number => v !== null),
  1681→    pts.map((p) => p.sma50).filter((v): v is number => v !== null),
  1682→    pts.map((p) => p.sma200).filter((v): v is number => v !== null),
  1683→    support ? [support.price] : [],
  1684→    resistance ? [resistance.price] : []
  1685→  );
  1686→  let mn = Math.min(...allVals);
  1687→  let mx = Math.max(...allVals);
  1688→  const pad = (mx - mn) * 0.06 || mx * 0.02 || 1;
  1689→  mn -= pad;
  1690→  mx += pad;
  1691→
  1692→  const xAt = (i: number) => P.left + (pts.length === 1 ? innerW / 2 : (i / (pts.length - 1)) * innerW);
  1693→  const yAt = (v: number) => P.top + (1 - (v - mn) / (mx - mn)) * priceH;
  1694→
  1695→  const maxVol = Math.max(...pts.map((p) => p.volume), 1);
  1696→  const volY = (v: number) => volTop + volH - (v / maxVol) * volH;
  1697→
  1698→  // Build a continuous path from (possibly null-padded) values — SMA200 has
  1699→  // nulls at the start of the series until enough bars accumulate.
  1700→  const segPath = (get: (p: StockChartPoint) => number | null): string[] => {
  1701→    const segs: string[] = [];
  1702→    let cur: string[] = [];
  1703→    pts.forEach((p, i) => {
  1704→      const v = get(p);
  1705→      if (v === null || v === undefined) {
  1706→        if (cur.length > 0) segs.push(cur.join(" "));
  1707→        cur = [];
  1708→        return;
  1709→      }
  1710→      cur.push(`${cur.length === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(v).toFixed(1)}`);
  1711→    });
  1712→    if (cur.length > 0) segs.push(cur.join(" "));
  1713→    return segs.filter((s) => s.length > 0);
  1714→  };
  1715→
  1716→  const closePath = segPath((p) => p.close).join(" ");
  1717→  const sma20Segs = segPath((p) => p.sma20);
  1718→  const sma50Segs = segPath((p) => p.sma50);
  1719→  const sma200Segs = segPath((p) => p.sma200);
  1720→
  1721→  // Price area fill under close
  1722→  const areaPath = `${closePath} L ${xAt(pts.length - 1).toFixed(1)} ${(P.top + priceH).toFixed(1)} L ${xAt(0).toFixed(1)} ${(P.top + priceH).toFixed(1)} Z`;
  1723→
  1724→  const gridLines = Array.from({ length: 5 }, (_, i) => P.top + (i / 4) * priceH);
  1725→  const gridVals = Array.from({ length: 5 }, (_, i) => mx - ((mx - mn) * i) / 4);
  1726→  const labelStep = Math.max(1, Math.ceil(pts.length / 7));
  1727→
  1728→  const lastClose = pts[pts.length - 1].close;
  1729→
  1730→  return (
  1731→    <div className="w-full">
  1732→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-auto" preserveAspectRatio="xMidYMid meet">
  1733→        <defs>
  1734→          <linearGradient id="priceFill" x1="0" y1="0" x2="0" y2="1">
  1735→            <stop offset="0%" stopColor="#00E676" stopOpacity="0.18" />
  1736→            <stop offset="100%" stopColor="#00E676" stopOpacity="0" />
  1737→          </linearGradient>
  1738→        </defs>
  1739→        {/* Grid + y labels */}
  1740→        {gridLines.map((y, i) => (
  1741→          <g key={i}>
  1742→            <line x1={P.left} y1={y} x2={P.left + innerW} y2={y} stroke="#1E2536" strokeWidth="1" strokeDasharray="2 4" />
  1743→            <text x={P.left - 6} y={y + 3} fill="#64748B" fontSize="9" textAnchor="end" fontFamily="monospace">
  1744→              {gridVals[i].toFixed(0)}
  1745→            </text>
  1746→          </g>
  1747→        ))}
  1748→        {/* X labels */}
  1749→        {pts.map((p, i) =>
  1750→          i % labelStep === 0 ? (
  1751→            <text
  1752→              key={i}
  1753→              x={xAt(i)}
  1754→              y={H - 8}
  1755→              fill="#64748B"
  1756→              fontSize="9"
  1757→              textAnchor="middle"
  1758→              fontFamily="monospace"
  1759→            >
  1760→              {p.date.slice(2)}
  1761→            </text>
  1762→          ) : null
  1763→        )}
  1764→        {/* Support / resistance lines */}
  1765→        {support && support.price >= mn && support.price <= mx && (
  1766→          <g>
  1767→            <line
  1768→              x1={P.left}
  1769→              y1={yAt(support.price)}
  1770→              x2={P.left + innerW}
  1771→              y2={yAt(support.price)}
  1772→              stroke="#00E676"
  1773→              strokeWidth="1"
  1774→              strokeDasharray="5 4"
  1775→              opacity="0.55"
  1776→            />
  1777→            <text x={P.left + innerW + 4} y={yAt(support.price) + 3} fill="#00E676" fontSize="8.5" fontFamily="monospace">
  1778→              S {support.price.toFixed(0)}
  1779→            </text>
  1780→          </g>
  1781→        )}
  1782→        {resistance && resistance.price >= mn && resistance.price <= mx && (
  1783→          <g>
  1784→            <line
  1785→              x1={P.left}
  1786→              y1={yAt(resistance.price)}
  1787→              x2={P.left + innerW}
  1788→              y2={yAt(resistance.price)}
  1789→              stroke="#FF5C5C"
  1790→              strokeWidth="1"
  1791→              strokeDasharray="5 4"
  1792→              opacity="0.55"
  1793→            />
  1794→            <text x={P.left + innerW + 4} y={yAt(resistance.price) + 3} fill="#FF5C5C" fontSize="8.5" fontFamily="monospace">
  1795→              R {resistance.price.toFixed(0)}
  1796→            </text>
  1797→          </g>
  1798→        )}
  1799→        {/* Volume bars */}
  1800→        {pts.map((p, i) => {
  1801→          const up = i === 0 ? p.close >= (pts[0]?.close ?? p.close) : p.close >= pts[i - 1].close;
  1802→          const y = volY(p.volume);
  1803→          return (
  1804→            <rect
  1805→              key={`v${i}`}
  1806→              x={xAt(i) - innerW / pts.length / 2}
  1807→              y={y}
  1808→              width={Math.max(0.8, innerW / pts.length)}
  1809→              height={volTop + volH - y}
  1810→              fill={up ? "#00E676" : "#FF5C5C"}
  1811→              opacity="0.25"
  1812→            />
  1813→          );
  1814→        })}
  1815→        {/* Price area + close line */}
  1816→        <path d={areaPath} fill="url(#priceFill)" />
  1817→        <path d={closePath} fill="none" stroke="#00E676" strokeWidth="2" />
  1818→        {/* SMA overlays */}
  1819→        {sma20Segs.map((d, i) => (
  1820→          <path key={`s20-${i}`} d={d} fill="none" stroke="#FBBF24" strokeWidth="1.5" opacity="0.9" />
  1821→        ))}
  1822→        {sma50Segs.map((d, i) => (
  1823→          <path key={`s50-${i}`} d={d} fill="none" stroke="#38BDF8" strokeWidth="1.5" strokeDasharray="5 3" opacity="0.85" />
  1824→        ))}
  1825→        {sma200Segs.map((d, i) => (
  1826→          <path key={`s200-${i}`} d={d} fill="none" stroke="#94A3B8" strokeWidth="1.5" strokeDasharray="2 3" opacity="0.8" />
  1827→        ))}
  1828→        {/* Endpoint marker */}
