    45→function fmtNum(n: number): string {
    46→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}B`;
    47→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}M`;
    48→  return n.toFixed(0);
    49→}

    51→function fyOf(period: string): string {
    52→  const m = period.match(/FY\s?(\d+)/i);
    53→  return m ? m[1] : period.slice(-2);
    54→}

   264→function StockDeepDive({
   265→  results,
   266→  symbol,
   267→  company,
   268→  onBack,
   269→}: {
   270→  results: QuarterlyResult[];
   271→  symbol: string;
   272→  company: string;
   273→  onBack: () => void;
   274→}) {
   275→  const patterns = detectPatterns(results);
   276→  const quarters = generateQuarterlySummary(results, symbol);
   277→  const yearsBetween =
   278→    results.length >= 2
   279→      ? (new Date(results[results.length - 1].periodEnd).getTime() -
   280→          new Date(results[0].periodEnd).getTime()) /
   281→        (365 * 24 * 60 * 60 * 1000)
   282→      : 0;
   283→  const revCAGR = computeCAGR(results[0].metrics.revenue, results[results.length - 1].metrics.revenue, yearsBetween);
   284→  const profCAGR = computeCAGR(
   285→    results[0].metrics.profitAfterTax,
   286→    results[results.length - 1].metrics.profitAfterTax,
   287→    yearsBetween
   288→  );
   289→  const epsCAGR = computeCAGR(results[0].metrics.eps, results[results.length - 1].metrics.eps, yearsBetween);
   290→  const totalDPS = results.reduce((s, r) => s + r.metrics.dividendPerShare, 0);
   291→  const volatility = computeStdDev(results.map((r) => r.metrics.epsYoYGrowth));
   292→  const revVolatility = computeStdDev(results.map((r) => r.metrics.revenueYoYGrowth));
   293→  const marginSlope = computeSlope(results.map((r) => r.metrics.netMargin));
   294→
   295→  const stats: { label: string; value: string; sub: string; tone?: "pos" | "neg" | "neutral"; explanation: string }[] = [
   296→    { label: "Revenue CAGR", value: `${revCAGR.toFixed(1)}%`, sub: `${yearsBetween.toFixed(1)}y compound`, tone: revCAGR >= 0 ? "pos" : "neg", explanation: "How fast sales grew per year, on average, over the entire period" },
   297→    { label: "Profit CAGR", value: `${profCAGR.toFixed(1)}%`, sub: "PAT growth", tone: profCAGR >= 0 ? "pos" : "neg", explanation: "How fast profit grew per year, on average" },
   298→    { label: "EPS CAGR", value: `${epsCAGR.toFixed(1)}%`, sub: "per-share profit", tone: epsCAGR >= 0 ? "pos" : "neg", explanation: "How fast per-share profit grew per year" },
   299→    { label: "Total DPS", value: `Rs. ${totalDPS.toFixed(2)}`, sub: `${results.length} quarters`, tone: "neutral", explanation: "Total cash dividends paid per share over the entire period" },
   300→    { label: "EPS volatility", value: `${volatility.toFixed(1)}pp`, sub: "EPS YoY std dev", tone: volatility > 15 ? "neg" : "neutral", explanation: "How unpredictable the profit growth is. Higher = more risky/uncertain" },
   301→    { label: "Revenue volatility", value: `${revVolatility.toFixed(1)}pp`, sub: "Rev YoY std dev", tone: revVolatility > 15 ? "neg" : "neutral", explanation: "How unpredictable the sales growth is" },
   302→    { label: "Margin trend slope", value: `${marginSlope >= 0 ? "+" : ""}${marginSlope.toFixed(2)}pp/q`, sub: "net margin trend", tone: marginSlope >= 0 ? "pos" : "neg", explanation: "Whether profit margin is improving (positive) or declining (negative) over time" },
   303→    { label: "Patterns detected", value: `${patterns.length}`, sub: "fundamental signals", tone: "neutral", explanation: "Number of fundamental patterns found in the data" },
   304→  ];
   305→
   306→  return (
   307→    <div className="space-y-4">
   308→      {/* Header with back button */}
   309→      <Card className="bg-[#131829] border-white/[0.08]">
   310→        <CardContent className="p-4">
   311→          <div className="flex items-start gap-3">
   312→            <Button
   313→              variant="outline"
   314→              size="sm"
   315→              onClick={onBack}
   316→              className="border-white/[0.12] text-[#94A3B8] hover:text-[#F4F7FB] shrink-0"
   317→            >
   318→              <ArrowLeft className="h-3.5 w-3.5 mr-1.5" /> Back
   319→            </Button>
   320→            <div className="flex-1 min-w-0">
   321→              <div className="flex items-baseline gap-2 flex-wrap">
   322→                <h2 className="text-xl font-bold text-[#F4F7FB]">{symbol}</h2>
   323→                <span className="text-sm text-[#94A3B8] truncate">{company}</span>
   324→              </div>
   325→              <p className="text-xs text-[#64748B] mt-1">
   326→                Deep dive on {results.length} quarters ·{" "}
   327→                {new Date(results[0].periodEnd).toLocaleDateString("en-PK", { year: "numeric", month: "short" })} →{" "}
   328→                {new Date(results[results.length - 1].periodEnd).toLocaleDateString("en-PK", { year: "numeric", month: "short" })}
   329→              </p>
   330→            </div>
   331→          </div>
   332→        </CardContent>
   333→      </Card>
   334→
   335→      {/* 8 summary stats */}
   336→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   337→        {stats.map((s, i) => (
   338→          <SummaryStat key={i} label={s.label} value={s.value} sub={s.sub} tone={s.tone} explanation={s.explanation} />
   339→        ))}
   340→      </div>
   341→
   342→      {/* Multi-line chart */}
   343→      <Card className="bg-[#131829] border-white/[0.08]">
   344→        <CardHeader>
   345→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   346→            <BarChart3 className="h-4 w-4 text-[#00E676]" />
   347→            Revenue · Profit · EPS trend (normalized 0–100)
   348→          </CardTitle>
   349→        </CardHeader>
   350→        <CardContent>
   351→          <MultiLineChart results={results} />
   352→        </CardContent>
   353→      </Card>
   354→
   355→      {/* Quarterly seasonal pattern */}
   356→      <Card className="bg-[#131829] border-white/[0.08]">
   357→        <CardHeader>
   358→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   359→            <Calendar className="h-4 w-4 text-[#00E676]" />
   360→            Quarterly seasonal pattern
   361→          </CardTitle>
   362→        </CardHeader>
   363→        <CardContent>
   364→          {quarters.length === 0 ? (
   365→            <p className="text-xs text-[#64748B]">
   366→              Need at least 4 quarters (ideally spanning multiple FYs) to compute seasonality.
   367→            </p>
   368→          ) : (
   369→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-3">
   370→              {quarters.map((q) => (
   371→                <QuarterSummaryCard key={q.quarter} q={q} />
   372→              ))}
   373→            </div>
   374→          )}
   375→        </CardContent>
   376→      </Card>
   377→
   378→      {/* Revenue bar chart (CSS) */}
   379→      <Card className="bg-[#131829] border-white/[0.08]">
   380→        <CardHeader>
   381→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   382→            <TrendingUp className="h-4 w-4 text-[#00E676]" />
   383→            Revenue by quarter (PKR, color-coded by FY)
   384→          </CardTitle>
   385→        </CardHeader>
   386→        <CardContent>
   387→          <RevenueBars results={results} />
   388→        </CardContent>
   389→      </Card>
   390→
   391→      {/* EPS bars + net margin overlay */}
   392→      <Card className="bg-[#131829] border-white/[0.08]">
   393→        <CardHeader>
   394→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   395→            <Coins className="h-4 w-4 text-[#00E676]" />
   396→            EPS bars + net margin overlay

   398→        </CardHeader>
   399→        <CardContent>
   400→          <EpsMarginChart results={results} />
   401→        </CardContent>
   402→      </Card>
   403→
   404→      {/* YoY growth chart */}
   405→      <Card className="bg-[#131829] border-white/[0.08]">
   406→        <CardHeader>
   407→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2">
   408→            <Activity className="h-4 w-4 text-[#00E676]" />
   409→            YoY growth — Revenue vs EPS
   410→          </CardTitle>

   413→          <YoYGrowthChart results={results} />

   417→      {/* Detected patterns */}
   418→      <Card className="bg-[#131829] border-white/[0.08]">
   419→        <CardHeader>
   420→          <CardTitle className="text-base text-[#F4F7FB] flex items-center gap-2 flex-wrap">
   421→            <Info className="h-4 w-4 text-[#00E676]" />
   422→            Detected patterns ({patterns.length})

   424→              <span className="flex items-center gap-1.5 ml-auto text-[10px] font-normal">
   425→                {(() => {
   426→                  const bull = patterns.filter((p) => p.severity === "positive").length;
   427→                  const bear = patterns.filter((p) => p.severity === "negative").length;
   428→                  const neutr = patterns.filter((p) => p.severity === "neutral").length;
   429→                  return (
   430→                    <>
   431→                      {bull > 0 && (
   432→                        <span className="text-[#00E676] font-mono">{bull} bullish</span>
   433→                      )}
   434→                      {bear > 0 && (
   435→                        <span className="text-[#FF5C5C] font-mono">{bear} bearish</span>
   436→                      )}
   437→                      {neutr > 0 && (
   438→                        <span className="text-[#94A3B8] font-mono">{neutr} neutral</span>
   439→                      )}
   440→                      <span className="text-[#64748B]">·</span>
   441→                      <span className="text-[#64748B]">
   442→                        {patterns.filter((p) => p.strength === "Strong").length} strong ·{" "}

   715→// ---- Pure-SVG multi-line chart: revenue (green), profit (blue dashed), EPS (yellow dotted)
   716→function MultiLineChart({ results }: { results: QuarterlyResult[] }) {
   717→  const W = 800;
   718→  const H = 280;
   719→  const P = { top: 20, right: 30, bottom: 36, left: 40 };
   720→  const innerW = W - P.left - P.right;
   721→  const innerH = H - P.top - P.bottom;
   722→
   723→  const revs = results.map((r) => r.metrics.revenue);
   724→  const profits = results.map((r) => r.metrics.profitAfterTax);
   725→  const epss = results.map((r) => r.metrics.eps);
   726→
   727→  const rangeOf = (arr: number[]) => {
   728→    const mn = Math.min(...arr);
   729→    const mx = Math.max(...arr);
   730→    return { mn, mx: mx === mn ? mn + 1 : mx };
   731→  };
   732→  const revR = rangeOf(revs);
   733→  const profR = rangeOf(profits);
   734→  const epsR = rangeOf(epss);
   735→
   736→  const norm = (v: number, r: { mn: number; mx: number }) =>
   737→    r.mx === r.mn ? 0.5 : (v - r.mn) / (r.mx - r.mn);
   738→
   739→  const xAt = (i: number) =>
   740→    P.left + (results.length === 1 ? innerW / 2 : (i / (results.length - 1)) * innerW);
   741→  const yAt = (n: number) => P.top + (1 - n) * innerH;
   742→
   743→  const linePath = (vals: number[], r: { mn: number; mx: number }) =>
   744→    vals
   745→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   746→      .join(" ");
   747→
   748→  const areaPath = (vals: number[], r: { mn: number; mx: number }) => {
   749→    if (vals.length === 0) return "";
   750→    const top = vals
   751→      .map((v, i) => `${i === 0 ? "M" : "L"} ${xAt(i).toFixed(1)} ${yAt(norm(v, r)).toFixed(1)}`)
   752→      .join(" ");
   753→    return `${top} L ${xAt(vals.length - 1).toFixed(1)} ${(P.top + innerH).toFixed(1)} L ${xAt(0).toFixed(
   754→      1
   755→    )} ${(P.top + innerH).toFixed(1)} Z`;
   756→  };
   757→
   758→  const gridLines = Array.from({ length: 5 }, (_, i) => P.top + (i / 4) * innerH);
   759→  const labelStep = Math.max(1, Math.ceil(results.length / 8));
   760→
   761→  const lastRev = revs[revs.length - 1];
   762→  const lastProf = profits[profits.length - 1];
   763→  const lastEps = epss[epss.length - 1];
   764→
   765→  return (
   766→    <div className="w-full">
   767→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-auto" preserveAspectRatio="xMidYMid meet">
   768→        <defs>
   769→          <linearGradient id="revFill" x1="0" y1="0" x2="0" y2="1">
   770→            <stop offset="0%" stopColor="#00E676" stopOpacity="0.25" />
   771→            <stop offset="100%" stopColor="#00E676" stopOpacity="0" />
   772→          </linearGradient>
   773→        </defs>
   774→        {/* Grid */}
   775→        {gridLines.map((y, i) => (
   776→          <line
   777→            key={i}
   778→            x1={P.left}
   779→            y1={y}
   780→            x2={P.left + innerW}
   781→            y2={y}
   782→            stroke="#1E2536"
   783→            strokeWidth="1"
   784→            strokeDasharray="2 4"
   785→          />
   786→        ))}
   787→        {/* X-axis labels */}
   788→        {results.map((r, i) =>
   789→          i % labelStep === 0 ? (
   790→            <text
   791→              key={i}
   792→              x={xAt(i)}
   793→              y={H - 8}
   794→              fill="#64748B"
   795→              fontSize="9"
   796→              textAnchor="middle"
   797→              fontFamily="monospace"
   798→            >
   799→              {r.period}
   800→            </text>
   801→          ) : null
   802→        )}
   803→        {/* Filled area under revenue */}
   804→        <path d={areaPath(revs, revR)} fill="url(#revFill)" />
   805→        {/* Revenue line (green solid) */}
   806→        <path d={linePath(revs, revR)} fill="none" stroke="#00E676" strokeWidth="2" />
   807→        {/* Profit line (blue dashed) */}
   808→        <path d={linePath(profits, profR)} fill="none" stroke="#3B82F6" strokeWidth="2" strokeDasharray="6 4" />
   809→        {/* EPS line (yellow dotted) */}
   810→        <path d={linePath(epss, epsR)} fill="none" stroke="#FBBF24" strokeWidth="2" strokeDasharray="2 3" />
   811→        {/* Endpoint markers */}
   812→        {revs.length > 0 && (
   813→          <circle cx={xAt(revs.length - 1)} cy={yAt(norm(lastRev, revR))} r="3.5" fill="#00E676" />
   814→        )}
   815→        {profits.length > 0 && (
   816→          <circle cx={xAt(profits.length - 1)} cy={yAt(norm(lastProf, profR))} r="3.5" fill="#3B82F6" />
   817→        )}
   818→        {epss.length > 0 && (
   819→          <circle cx={xAt(epss.length - 1)} cy={yAt(norm(lastEps, epsR))} r="3.5" fill="#FBBF24" />
   820→        )}
   821→      </svg>
   822→      {/* Legend */}
   823→      <div className="flex items-center gap-4 mt-3 text-[10px] text-[#94A3B8] flex-wrap justify-center">
   824→        <span className="flex items-center gap-1.5">
   825→          <span className="h-0.5 w-4 bg-[#00E676] inline-block" />
   826→          Revenue (PKR)
   827→        </span>
   828→        <span className="flex items-center gap-1.5">
   829→          <span className="h-0.5 w-4 inline-block" style={{ borderBottom: "2px dashed #3B82F6" }} />
   830→          Profit after tax (PKR)
   831→        </span>
   832→        <span className="flex items-center gap-1.5">
   833→          <span className="h-0.5 w-4 inline-block" style={{ borderBottom: "2px dotted #FBBF24" }} />
   834→          EPS (PKR)

   993→function YoYGrowthChart({ results }: { results: QuarterlyResult[] }) {
   994→  const items = results.filter(
   995→    (r) => r.metrics.revenueYoYGrowth !== 0 || r.metrics.epsYoYGrowth !== 0
   996→  );
   997→  if (items.length === 0) {
   998→    return <p className="text-xs text-[#64748B]">No YoY growth data available (need 4+ quarters).</p>;
   999→  }
  1000→
  1001→  const W = 800;
  1002→  const H = 240;
  1003→  const P = { top: 20, right: 30, bottom: 30, left: 40 };
  1004→  const innerW = W - P.left - P.right;
  1005→  const innerH = H - P.top - P.bottom;
  1006→  const zeroY = P.top + innerH / 2;
  1007→
  1008→  const maxAbs = Math.max(
  1009→    ...items.flatMap((r) => [
  1010→      Math.abs(r.metrics.revenueYoYGrowth),
  1011→      Math.abs(r.metrics.epsYoYGrowth),
  1012→    ]),
  1013→    1
  1014→  );
  1015→
  1016→  const step = innerW / Math.max(1, items.length);
  1017→  const barW = step * 0.3;
  1018→  const gap = step * 0.1;
  1019→  const labelStep = Math.max(1, Math.ceil(items.length / 8));
  1020→
  1021→  return (
  1022→    <div className="w-full overflow-x-auto">
  1023→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-auto" preserveAspectRatio="xMidYMid meet">

  1031→          strokeWidth="1"
