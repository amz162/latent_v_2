    43→// ─── Issue 13: Close-trade calculation helpers ─────────────────────────────
    44→// Pakistan capital-gains tax on PSX equities (2024 budget):
    45→//   - holding period ≤ 12 months → 15% CGT on profit (short-term)
    46→//   - holding period > 12 months → 0% CGT (long-term, exempt)
    47→// Losses are not taxed (no tax credit) — CGT applies only on positive profit.
    48→// Broker commission: 0.15% × sell value × 2 (both sides — buy + sell).
    49→interface CloseTarget {
    50→  symbol: string;
    51→  name: string;
    52→  sector?: string | null;
    53→  buyPrice: number;
    54→  shares: number;
    55→  addedAt: string;
    56→  // Issue 28-B: snapshot of the lots behind the (averaged) buy price.
    57→  buys: BuyLot[];
    58→}
    59→
    60→interface CloseCalc {
    61→  sellPrice: number;
    62→  buyCost: number;
    63→  sellValue: number;
    64→  commission: number;
    65→  holdingDays: number;
    66→  holdingMonths: number;
    67→  taxRate: number; // 0.15 short-term, 0 long-term
    68→  cgt: number;
    69→  grossPnl: number;
    70→  netPnl: number;
    71→  netPnlPct: number;
    72→}
    73→
    74→function computeCloseMetrics(
    75→  buyPrice: number,
    76→  shares: number,
    77→  sellPrice: number,
    78→  addedAt: string
    79→): CloseCalc {
    80→  const buyCost = buyPrice * shares;
    81→  const sellValue = sellPrice * shares;
    82→  // Broker commission = 0.15% × sell value, applied on both sides (buy + sell).
    83→  // Using sellValue as the base for both sides per spec ("0.15% × sell value (both sides)").
    84→  const commission = 0.0015 * sellValue * 2;
    85→  const holdingDays = Math.max(
    86→    0,
    87→    Math.floor((Date.now() - new Date(addedAt).getTime()) / (1000 * 60 * 60 * 24))
    88→  );
    89→  const holdingMonths = Math.floor(holdingDays / 30);
    90→  // CGT: 15% for ≤12 months (short-term), 0% for >12 months (long-term)
    91→  const taxRate = holdingDays <= 365 ? 0.15 : 0;
    92→  const grossPnl = sellValue - buyCost;
    93→  // CGT only on positive profit (no tax credit for losses)
    94→  const cgt = grossPnl > 0 ? grossPnl * taxRate : 0;
    95→  const netPnl = grossPnl - commission - cgt;
    96→  const netPnlPct = buyCost > 0 ? (netPnl / buyCost) * 100 : 0;
    97→  return {
    98→    sellPrice,
    99→    buyCost,
   100→    sellValue,
   101→    commission,
   102→    holdingDays,
   103→    holdingMonths,
   104→    taxRate,
   105→    cgt,
   106→    grossPnl,
   107→    netPnl,
   108→    netPnlPct,
   109→  };
   110→}

   112→// ─── Issue 15: batched LIVE quote fetcher for portfolio holdings ────────────
   113→// Hits our own /api/psx/live/batch endpoint, which server-side calls
   114→// `getLiveQuote` (60s in-memory cache per symbol) for each portfolio symbol
   115→// in parallel via Promise.all. Returns the latest LIVE PSX price for each
   116→// symbol, plus isLoading / refresh / lastUpdated for the UI.
   117→//
   118→// Used by PortfolioTab to:
   119→//   - render per-position live P&L (green/red) on open holdings (Issue 15)
   120→//   - compute the total portfolio P&L summary card (Issue 15)
   121→//   - show per-row live price on watch-only items (free upgrade)
   122→//   - skeleton-while-loading + manual Refresh button + 60s auto-refresh
   123→//
   124→// Implementation note: setState is only called inside async .then callbacks
   125→// (NOT synchronously in the effect body) to avoid React 19's
   126→// `react-hooks/set-state-in-effect` cascading-render warning. The "loading"
   127→// state is derived from `state.key !== key` (data is stale until the new
   128→// fetch for the current symbol-set resolves) plus an explicit `loading`
   129→// flag set inside the `refresh` event handler.
   130→function usePortfolioLiveQuotes(symbols: string[]): {
   131→  prices: Record<string, number>;
   132→  isLoading: boolean;
   133→  refresh: () => void;
   134→  lastUpdated: Date | null;
   135→} {
   136→  const key = symbols.join(",");
   137→  const [state, setState] = useState<{
   138→    key: string;
   139→    prices: Record<string, number>;
   140→    lastUpdated: Date | null;
   141→    loading: boolean;
   142→  }>({ key: "", prices: {}, lastUpdated: null, loading: false });
   143→  const [refreshTick, setRefreshTick] = useState(0);
   144→
   145→  useEffect(() => {
   146→    if (!key) return;
   147→    let cancelled = false;
   148→    const symbolsList = key.split(",");
   149→    fetch(`/api/psx/live/batch?symbols=${encodeURIComponent(symbolsList.join(","))}`)
   150→      .then((r) => (r.ok ? r.json() : null))
   151→      .then((d) => {
   152→        if (cancelled) return;
   153→        if (d && d.prices && typeof d.prices === "object") {
   154→          setState({
   155→            key,
   156→            prices: d.prices as Record<string, number>,
   157→            lastUpdated: new Date(d.fetchedAt || Date.now()),
   158→            loading: false,
   159→          });
   160→        } else {
   161→          // Even if no prices resolved, mark this key as "fetched" so
   162→          // isLoading flips to false and rows show "no live price" instead
   163→          // of a skeleton.
   164→          setState({ key, prices: {}, lastUpdated: new Date(), loading: false });
   165→        }
   166→      })
   167→      .catch(() => {
   168→        if (cancelled) return;
   169→        // Keep stale prices if any; mark this key as attempted so the
   170→        // skeleton goes away.
   171→        setState((s) => ({ key, prices: s.key === key ? s.prices : {}, lastUpdated: new Date(), loading: false }));
   172→      });
   173→    return () => {
   174→      cancelled = true;
   175→    };
   176→  }, [key, refreshTick]);
   177→
   178→  // Auto-refresh every 60s — matches getLiveQuote's 60s cache TTL so the
   179→  // dashboard P&L stays current without manual button clicks.
   180→  useEffect(() => {
   181→    if (!key) return;
   182→    const id = setInterval(() => setRefreshTick((t) => t + 1), 60_000);
   183→    return () => clearInterval(id);
   184→  }, [key]);
   185→
   186→  const refresh = useCallback(() => {
   187→    setRefreshTick((t) => t + 1);
   188→    setState((s) => ({ ...s, loading: true }));
   189→  }, []);
   190→
   191→  // Derive public state:
   192→  //   - prices: valid only when state.key === key (otherwise stale → empty)
   193→  //   - isLoading: true if we have no data yet for the current key OR a
   194→  //     refresh is explicitly in progress
   195→  const hasData = state.key === key;
   196→  const prices = hasData ? state.prices : {};
   197→  const lastUpdated = hasData ? state.lastUpdated : null;
   198→  const isLoading = !!key && (!hasData || state.loading);
   199→  return { prices, isLoading, refresh, lastUpdated };
   200→}

   243→    const sharesEntered = shares.trim() !== "";

   246→    const parsedPrice = priceEntered ? parseFloat(buyPrice) : NaN;

   250→      toast.error("Enter a valid number of shares (greater than zero)");
   251→      return;
   252→    }
   253→    if (priceEntered && (!Number.isFinite(parsedPrice) || parsedPrice <= 0)) {
   254→      toast.error("Enter a valid buy price (greater than zero)");
   255→      return;
   256→    }
   257→
   258→    const addingPosition = parsedShares > 0;
   259→
   260→    // ─── Issue 28-B: one row per symbol — duplicate guard ───
   261→    const existing = portfolio.find((p) => p.symbol === selectedSymbol);
   262→    const existingIsHolding =
   263→      !!existing &&
   264→      existing.shares > 0 &&
   265→      existing.buyPrice !== null &&
   266→      existing.buyPrice !== undefined;
   267→
   268→    if (!addingPosition) {
   269→      // Issue 19 behavior: until the user specifies shares (volume), the
   270→      // stock is added as "watch only" (no P&L tracking). A symbol may only
   271→      // exist ONCE anywhere in the portfolio array — as a holding OR as a
   272→      // watch-only item — so block duplicate watch-only adds with a clear
   273→      // message and keep the dialog open so the user sees it.

   330→          linkView: "portfolio",
   331→        });
   332→      } else {
   333→        toast.success(
   334→          `${selectedSymbol} moved from watchlist to portfolio (${parsedShares} shares @ ${parsedPrice.toFixed(2)})`
   335→        );
   336→        pushNotification({
   337→          type: "info",
   338→          title: `${selectedSymbol} moved to portfolio`,
   339→          body: `${selectedSymbol} moved from watchlist to portfolio (${parsedShares} shares @ PKR ${parsedPrice.toFixed(2)})`,
   340→          linkView: "portfolio",
   341→        });
   342→      }
   343→    } else {
   344→      toast.success(`${selectedSymbol} added to watchlist`);
   345→      pushNotification({
   346→        type: "info",
   347→        title: "Stock added to watchlist",
   348→        body: `${selectedSymbol} added (watch only)`,
   349→        linkView: "portfolio",

   354→    setSelectedSymbol("");
   355→    setSelectedName("");
   356→    setSelectedSector(undefined);
   357→    setBuyPrice("");
   358→    setShares("");

   361→  const sorted = [...portfolio].sort((a, b) => a.order - b.order);
   362→
   363→  // ─── Issue 10: separate HOLDINGS from WATCHLIST ─────────────────────────
   364→  // A position is a "holding" ONLY if it has shares > 0 AND a non-null
   365→  // buyPrice (a real, open position). Watch-only items (shares=0 OR
   366→  // buyPrice=null) do NOT count toward the holdings summary / total P&L.
   367→  // The store distinguishes via the shares/buyPrice fields — see
   368→  // PortfolioPosition shape in src/lib/latent/store.ts.

   370→    p.shares > 0 && p.buyPrice !== null && p.buyPrice !== undefined;

   482→      grossPnl: c.grossPnl,
   483→      netPnl: c.netPnl,
   484→      netPnlPct: c.netPnlPct,
   485→      holdingDays: c.holdingDays,
   486→      taxRate: c.taxRate,
   487→      closedAt: new Date().toISOString(),
   488→      // Issue 28-B: snapshot of the buy lots behind the averaged buy price.

   495→      body: `Sold ${t.shares} sh @ ${c.sellPrice.toFixed(2)} · commission PKR ${Math.round(c.commission).toLocaleString()} · CGT PKR ${Math.round(c.cgt).toLocaleString()} (${(c.taxRate * 100).toFixed(0)}% bracket) · holding ${c.holdingDays}d. Position reset to watch-only.`,
   496→      linkView: "tracker",
   497→    });
   498→    toast.success(
   499→      `${t.symbol} closed — Net P&L PKR ${Math.round(c.netPnl).toLocaleString()} (${c.netPnlPct >= 0 ? "+" : ""}${c.netPnlPct.toFixed(2)}%). Position reset to watch-only.`

   506→  // ─── Issue 15: Holdings-only totals (watchlist does NOT count) ──────────
   507→  // Per-position: current value = shares × livePrice, P&L = (livePrice −
   508→  // buyPrice) × shares, P&L% = ((livePrice − buyPrice)/buyPrice) × 100.
   509→  // Totals: sum of all holdings' cost/value/P&L → portfolio summary cards.
   510→  const totalCost = useMemo(
   511→    () => holdings.reduce((sum, p) => sum + p.shares * (p.buyPrice as number), 0),
   512→    [holdings]
   513→  );
   514→  const totalValue = useMemo(
   515→    () =>

  1320→      const total = data.totalInterpreted || 0;
  1321→      if (total > 0) {

  1358→            size="sm"

  1364→              <Loader2 className="h-3 w-3 mr-1.5 animate-spin" />

  1366→              <RefreshCw className="h-3 w-3 mr-1.5" />
  1367→            )}
  1368→            Refresh
  1369→          </Button>
  1370→        </div>
  1371→
  1372→        {loading && digest.length === 0 && (
  1373→          <div className="space-y-3">
  1374→            {[1, 2, 3].map((i) => <Skeleton key={i} className="h-24 w-full bg-white/[0.04]" />)}
  1375→          </div>

  1377→
  1378→        {error && (
  1379→          <div className="rounded-md bg-[#FF5C5C]/[0.07] border border-[#FF5C5C]/20 p-3 text-xs text-[#FF5C5C]">
