    49→    49→  modelBuyCount?: number; // how many of 22 trained models voted BUY
    50→    50→  modelSellCount?: number; // how many of 22 trained models voted SELL
    51→    51→  modelHoldCount?: number; // how many of 22 trained models voted HOLD
    52→    52→}
    53→    53→
    54→    54→// Filter policy:
    55→    55→//   - Only BUY and SELL signals (HOLD suppressed)
    56→    56→//   - R:R ≥ 1:2 (reward ≥ 2× risk), using TP3 (21-day target) as the reward target
    57→    57→// The batch endpoint /api/signals/all applies this filter server-side, so the
    58→    58→// SignalSection list and the Dashboard's "active signals" KPI always agree.
    59→    59→const MIN_RISK_REWARD = 1.99; // 1.99 to avoid floating-point edge cases
    60→    60→
    61→    61→function computeRiskReward(entry: number, target: number, stop: number): number {
    62→    62→  const reward = Math.abs(target - entry);
    63→    63→  const risk = Math.abs(entry - stop);
    64→    64→  if (risk <= 0) return 0;
    65→    65→  return reward / risk;
    66→    66→}
    67→    67→
    68→    68→// Build a one-line layman summary from the ensemble consensus fields.
    69→    69→function buildReason(
    70→    70→  symbol: string,
    71→    71→  direction: Direction,
    72→    72→  confidence: number,
    73→    73→  rr: number,
    74→    74→  avgReturn?: number,
    75→    75→  modelBuyCount?: number,
    76→    76→  modelSellCount?: number,
    77→    77→  modelHoldCount?: number
    78→    78→): string {
    79→    79→  const verb = direction === "BUY" ? "Bullish" : "Bearish";
    80→    80→  const ret =
    81→    81→    typeof avgReturn === "number" && isFinite(avgReturn)
    82→    82→      ? ` · projected ${avgReturn >= 0 ? "+" : ""}${(avgReturn * 100).toFixed(1)}% avg return`
    83→    83→      : "";
    84→    84→  const modelVote =
    85→    85→    typeof modelBuyCount === "number" &&
    86→    86→    typeof modelSellCount === "number" &&
    87→    87→    typeof modelHoldCount === "number"
    88→    88→      ? ` · ${modelBuyCount + modelSellCount + modelHoldCount}-model consensus: ${modelBuyCount} BUY / ${modelSellCount} SELL / ${modelHoldCount} HOLD`
    89→    89→      : "";
    90→    90→  return `${verb} ensemble consensus on ${symbol} — confidence ${confidence}/100, R:R 1:${rr.toFixed(2)} (TP3 target, 21-day). SL = 1.5 × ATR (14-day).${ret}${modelVote}`;
    91→    91→}
