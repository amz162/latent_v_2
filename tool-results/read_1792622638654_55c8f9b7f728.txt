    42→// ─── Issue 13: Close-trade calculation helpers ─────────────────────────────
    43→// Pakistan capital-gains tax on PSX equities (2024 budget):
    44→//   - holding period ≤ 12 months → 15% CGT on profit (short-term)
    45→//   - holding period > 12 months → 0% CGT (long-term, exempt)
    46→// Losses are not taxed (no tax credit) — CGT applies only on positive profit.
    47→// Broker commission: 0.15% × sell value × 2 (both sides — buy + sell).
    48→interface CloseTarget {
    49→  symbol: string;
    50→  name: string;
    51→  sector?: string | null;
    52→  buyPrice: number;
    53→  shares: number;
    54→  addedAt: string;
    55→}
    56→
    57→interface CloseCalc {
    58→  sellPrice: number;
    59→  buyCost: number;
    60→  sellValue: number;
    61→  commission: number;
    62→  holdingDays: number;
    63→  holdingMonths: number;
    64→  taxRate: number; // 0.15 short-term, 0 long-term
    65→  cgt: number;
    66→  grossPnl: number;
    67→  netPnl: number;
    68→  netPnlPct: number;
    69→}
    70→
    71→function computeCloseMetrics(
    72→  buyPrice: number,
    73→  shares: number,
    74→  sellPrice: number,
    75→  addedAt: string
    76→): CloseCalc {
    77→  const buyCost = buyPrice * shares;
    78→  const sellValue = sellPrice * shares;
    79→  // Broker commission = 0.15% × sell value, applied on both sides (buy + sell).
    80→  // Using sellValue as the base for both sides per spec ("0.15% × sell value (both sides)").
    81→  const commission = 0.0015 * sellValue * 2;
    82→  const holdingDays = Math.max(
    83→    0,
    84→    Math.floor((Date.now() - new Date(addedAt).getTime()) / (1000 * 60 * 60 * 24))
    85→  );
    86→  const holdingMonths = Math.floor(holdingDays / 30);
    87→  // CGT: 15% for ≤12 months (short-term), 0% for >12 months (long-term)
    88→  const taxRate = holdingDays <= 365 ? 0.15 : 0;
    89→  const grossPnl = sellValue - buyCost;
    90→  // CGT only on positive profit (no tax credit for losses)
    91→  const cgt = grossPnl > 0 ? grossPnl * taxRate : 0;
    92→  const netPnl = grossPnl - commission - cgt;
    93→  const netPnlPct = buyCost > 0 ? (netPnl / buyCost) * 100 : 0;
    94→  return {
    95→    sellPrice,
    96→    buyCost,
    97→    sellValue,
    98→    commission,
    99→    holdingDays,
   100→    holdingMonths,
   101→    taxRate,
   102→    cgt,
   103→    grossPnl,
   104→    netPnl,
   105→    netPnlPct,
   106→  };
   107→}

   109→// Fetch live prices for portfolio symbols via /api/psx/ohlcv/[symbol].
   110→// Returns a map of symbol → latest close price. Symbols with no fetchable price
   111→// are simply missing from the map (caller falls back to MOCK_STOCKS or shows "—").
   112→// Issue 14: previous rendering did `MOCK_STOCKS.find((s) => s.symbol === p.symbol)`
   113→// and `if (!stock) return null` — that skipped ALL non-demo stocks. We now fetch
   114→// live OHLCV for every portfolio symbol so ANY of the 361+ PSX-listed stocks
   115→// is displayable with a current price.
   116→function usePortfolioLivePrices(symbols: string[]): Record<string, number> {
   117→  const [prices, setPrices] = useState<Record<string, number>>({});
   118→  const key = symbols.join(",");
   119→  useEffect(() => {
   120→    if (!key) return;
   121→    let cancelled = false;
   122→    Promise.all(
   123→      symbols.map((sym) =>
   124→        fetch(`/api/psx/ohlcv/${encodeURIComponent(sym)}`)
   125→          .then((r) => (r.ok ? r.json() : null))
   126→          .then((d) => {
   127→            if (!d) return null;
   128→            if (Array.isArray(d.ohlcv) && d.ohlcv.length > 0) {
   129→              const last = d.ohlcv[d.ohlcv.length - 1];
   130→              if (last && typeof last.close === "number" && last.close > 0) {
   131→                return { sym, price: last.close };
   132→              }
   133→            }
   134→            return null;
   135→          })
   136→          .catch(() => null)
   137→      )
   138→    ).then((results) => {
   139→      if (cancelled) return;
   140→      const m: Record<string, number> = {};
   141→      for (const r of results) {
   142→        if (r && typeof r.price === "number" && r.price > 0) m[r.sym] = r.price;
   143→      }
   144→      setPrices(m);
   145→    });
   146→    return () => {
   147→      cancelled = true;
   148→    };
   149→  }, [key]);
   150→  return prices;
   151→}

   833→// ────────────────────────────────────────────────────────────────────────────
   834→// Per-Stock News Digest panel (auto-fetched + LLM-interpreted news for portfolio)
   835→// ────────────────────────────────────────────────────────────────────────────
   836→
   837→interface InterpretedNewsItem {
   838→  title?: string;
   839→  link?: string;
   840→  source?: string;
   841→  publishedAt?: string;
   842→  summary?: string;
   843→  symbol: string;
   844→  impact: "positive" | "negative" | "neutral";
   845→  impactStrength: "weak" | "moderate" | "strong";
   846→  laymanExplanation: string;
   847→  shortTermOutlook: string;
   848→  affectedStocks: string[];
   849→}
   850→
   851→interface NewsDigestEntry {
   852→  symbol: string;
   853→  stockName: string;
   854→  newsCount: number;
   855→  interpretedItems: InterpretedNewsItem[];
   856→}
   857→
   858→function PortfolioNewsDigest({ symbols }: { symbols: string[] }) {
   859→  const [digest, setDigest] = useState<NewsDigestEntry[]>([]);
   860→  const [loading, setLoading] = useState(false);
   861→  const [error, setError] = useState<string | null>(null);
   862→  const setView = useLatentStore((s) => s.setView);
   863→  const pushNotification = useLatentStore((s) => s.pushNotification);
   864→
   865→  const loadDigest = async () => {
   866→    if (symbols.length === 0) return;
   867→    setLoading(true);
   868→    setError(null);
   869→    try {
   870→      const res = await fetch(`/api/portfolio/news-digest?symbols=${symbols.join(",")}&limit=2`);
   871→      if (!res.ok) {
   872→        const err = await res.json().catch(() => ({}));
   873→        throw new Error(err.detail || err.error || `Failed (${res.status})`);
   874→      }
   875→      const data = await res.json();
   876→      setDigest(data.digest || []);
   877→      const total = data.totalInterpreted || 0;
   878→      if (total > 0) {
   879→        pushNotification({
   880→          type: "news",
   881→          title: `${total} news items interpreted`,
   882→          body: `Auto-analyzed news for ${symbols.length} portfolio stock${symbols.length === 1 ? "" : "s"}`,
   883→          linkView: "portfolio",
   884→        });
   885→      }
   886→    } catch (err) {
   887→      setError(err instanceof Error ? err.message : "Failed to load news digest");
   888→    } finally {
   889→      setLoading(false);
   890→    }
   891→  };
   892→
   893→  useEffect(() => {
   894→    loadDigest();
   895→  }, [symbols.join(",")]);
   896→
   897→  const impactColor = (i: string) =>
   898→    i === "positive" ? "text-[#00E676]" : i === "negative" ? "text-[#FF5C5C]" : "text-[#94A3B8]";
   899→
   900→  return (
   901→    <Card className="bg-[#131829] border-white/[0.08] mt-4">
   902→      <CardContent className="p-5">
   903→        <div className="flex items-center justify-between mb-4">
   904→          <div>
   905→            <h3 className="font-semibold text-[#F4F7FB] flex items-center gap-2">
   906→              <TrendingUp className="h-4 w-4 text-[#00E676]" />
   907→              Auto-News Digest for Your Portfolio
   908→            </h3>
   909→            <p className="text-xs text-[#94A3B8] mt-0.5">
   910→              For each stock below: news fetched from AskAnalyst + GLM-4 interprets impact in layman terms.
   911→            </p>
   912→          </div>
   913→          <Button
   914→            variant="outline"
   915→            size="sm"
   916→            onClick={loadDigest}
   917→            disabled={loading}
   918→            className="border-white/[0.12] text-[#94A3B8] h-7"
   919→          >
   920→            {loading ? (
   921→              <Loader2 className="h-3 w-3 mr-1.5 animate-spin" />
   922→            ) : (
   923→              <RefreshCw className="h-3 w-3 mr-1.5" />
   924→            )}
   925→            Refresh
   926→          </Button>
   927→        </div>
   928→
   929→        {loading && digest.length === 0 && (
   930→          <div className="space-y-3">
   931→            {[1, 2, 3].map((i) => <Skeleton key={i} className="h-24 w-full bg-white/[0.04]" />)}
   932→          </div>
   933→        )}
   934→
   935→        {error && (
   936→          <div className="rounded-md bg-[#FF5C5C]/[0.07] border border-[#FF5C5C]/20 p-3 text-xs text-[#FF5C5C]">
   937→            <span className="font-semibold">Error:</span> {error}
   938→          </div>
   939→        )}
   940→
   941→        {!loading && !error && digest.length > 0 && (
   942→          <div className="space-y-4">
   943→            {digest.map((entry) => (
   944→              <div key={entry.symbol} className="border-l-2 border-[#00E676]/30 pl-3">
   945→                <div className="flex items-center gap-2 mb-2">
   946→                  <div className="h-7 w-7 rounded-md bg-[#00E676]/10 flex items-center justify-center font-mono font-bold text-[#00E676] text-[10px]">
   947→                    {entry.symbol.slice(0, 4)}
   948→                  </div>
