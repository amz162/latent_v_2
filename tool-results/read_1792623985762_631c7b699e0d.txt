    57→function fmtMoney(n: number): string {
    58→  if (!isFinite(n) || isNaN(n)) return "Rs 0";
    59→  const sign = n < 0 ? "-" : "";
    60→  const abs = Math.abs(n);
    61→  if (abs >= 1_00_00_000) return `${sign}Rs ${(abs / 1_00_00_000).toFixed(2)} Cr`;
    62→  if (abs >= 1_00_000) return `${sign}Rs ${(abs / 1_00_000).toFixed(2)} Lac`;
    63→  if (abs >= 1_000) return `${sign}Rs ${abs.toLocaleString("en-PK", { maximumFractionDigits: 0 })}`;
    64→  return `${sign}Rs ${abs.toFixed(2)}`;
    65→}
    66→
    67→function fmtNum(n: number, d = 2): string {
    68→  if (!isFinite(n) || isNaN(n)) return "0";
    69→  return n.toLocaleString("en-PK", { maximumFractionDigits: d, minimumFractionDigits: d });
    70→}
    71→
    72→function monthsBetween(buy: string, sell: string): number {
    73→  if (!buy || !sell) return 0;
    74→  const b = new Date(buy);
    75→  const s = new Date(sell);
    76→  if (isNaN(b.getTime()) || isNaN(s.getTime())) return 0;
    77→  const diff = (s.getTime() - b.getTime()) / (1000 * 60 * 60 * 24);
    78→  return diff / 30.4375; // avg month length
    79→}

    85→type FilerStatus = "filer" | "non_filer";
    86→
    87→function getCgtRateClean(months: number, status: FilerStatus): { rate: number; band: string } {
    88→  if (months <= 12) return { rate: status === "filer" ? 15 : 22.5, band: "≤ 12 months (Short-term)" };
    89→  if (months <= 24) return { rate: status === "filer" ? 12.5 : 18.75, band: "12–24 months (Mid-term)" };
    90→  return { rate: 0, band: "> 24 months (Long-term — exempt)" };
    91→}

   511→    const m = parseFloat(monthly) || 0;
   512→    const a = (parseFloat(annual) || 0) / 100;
   513→    const y = parseInt(years) || 0;
   514→    const su = (parseFloat(stepUp) || 0) / 100;
   515→
   516→    const monthlyRate = Math.pow(1 + a, 1 / 12) - 1;
   517→    const yearlyRows: { year: number; monthlyThisYear: number; contributed: number; corpus: number; cumulativeContributed: number }[] = [];
   518→    let corpus = 0;
   519→    let totalContributed = 0;
   520→    let monthlyThisYear = m;
   521→
   522→    for (let yy = 1; yy <= y; yy++) {
   523→      const yearContribution = monthlyThisYear * 12;
   524→      // Grow corpus + this year's monthly contributions compounded monthly
   525→      let yearEnd = corpus;
   526→      for (let mo = 0; mo < 12; mo++) {

   915→// Map a long PSX sector name (e.g. "Commercial Banks", "Oil & Gas Marketing")
   916→// into the short code the fair-value API uses (bank / fertilizer / cement / oilgas / power / default).
   917→function toSectorCode(sectorName: string): string {
   918→  const s = (sectorName || "").toLowerCase();
   919→  if (!s) return "default";
   920→  if (s.includes("bank")) return "bank";
   921→  if (s.includes("fertil")) return "fertilizer";
   922→  if (s.includes("cement")) return "cement";
   923→  if (s.includes("oil") || s.includes("gas")) return "oilgas";
   924→  if (s.includes("power") || s.includes("energy")) return "power";
   925→  return "default";
   926→}

   990→// Verdict string → friendly label + tone color
   991→function verdictInfo(
   992→  verdict?: string
   993→): { label: string; tone: "pos" | "warn" | "neutral" | "neg" } {
   994→  const v = (verdict || "").toUpperCase();
   995→  if (v.includes("UNDERVALUED") && !v.includes("SLIGHT"))
   996→    return { label: "UNDERVALUED", tone: "pos" };
   997→  if (v.includes("SLIGHTLY UNDERVALUED"))
   998→    return { label: "SLIGHTLY UNDERVALUED", tone: "pos" };
   999→  if (v.includes("FAIRLY VALUED"))
  1000→    return { label: "FAIRLY VALUED", tone: "neutral" };
  1001→  if (v.includes("OVERVALUED"))
  1002→    return { label: "OVERVALUED", tone: "neg" };
  1003→  return { label: verdict || "—", tone: "neutral" };
  1004→}

  1773→
  1774→function PositionSizeCalculator() {
  1775→  const [capital, setCapital] = useState("500000");
  1776→  const [entry, setEntry] = useState("100");
  1777→  const [sl, setSl] = useState("95");
  1778→  const [risk, setRisk] = useState("1.5");
  1779→  const [target, setTarget] = useState("115");
  1780→
  1781→  const r = useMemo(() => {
  1782→    const cap = parseFloat(capital) || 0;
  1783→    const e = parseFloat(entry) || 0;
  1784→    const s = parseFloat(sl) || 0;
  1785→    const rk = (parseFloat(risk) || 0) / 100;
  1786→    const tg = parseFloat(target) || 0;
  1787→
  1788→    const isLong = s < e;
  1789→    const riskPerShare = Math.abs(e - s);
  1790→    const riskAmount = cap * rk;
  1791→    const shares = riskPerShare > 0 ? riskAmount / riskPerShare : 0;
  1792→    const positionValue = shares * e;
  1793→    const rewardPerShare = isLong ? Math.abs(tg - e) : Math.abs(e - tg);
  1794→    const rrRatio = riskPerShare > 0 ? rewardPerShare / riskPerShare : 0;
  1795→    const potentialProfit = shares * rewardPerShare;
  1796→    const potentialLoss = shares * riskPerShare;
  1797→    const pctAtTarget = e > 0 ? (rewardPerShare / e) * 100 : 0;
  1798→    const pctAtSl = e > 0 ? (riskPerShare / e) * 100 : 0;
  1799→    const capitalDeployedPct = cap > 0 ? (positionValue / cap) * 100 : 0;
  1800→
  1801→    // Quality
  1802→    let quality: { label: string; tone: "neg" | "warn" | "ok" | "good" | "great" } = { label: "Poor", tone: "neg" };
  1803→    if (rrRatio >= 3) quality = { label: "Excellent", tone: "great" };
  1804→    else if (rrRatio >= 2) quality = { label: "Very Good", tone: "good" };
  1805→    else if (rrRatio >= 1.5) quality = { label: "Good", tone: "ok" };
  1806→    else if (rrRatio >= 1) quality = { label: "Fair", tone: "warn" };
  1807→
  1808→    return {
  1809→      isLong,
  1810→      riskPerShare,
  1811→      riskAmount,
  1812→      shares,
  1813→      positionValue,
  1814→      rewardPerShare,
  1815→      rrRatio,
  1816→      potentialProfit,
  1817→      potentialLoss,
  1818→      pctAtTarget,
  1819→      pctAtSl,
  1820→      capitalDeployedPct,
  1821→      quality,
  1822→    };
  1823→  }, [capital, entry, sl, risk, target]);
  1824→
