   135→function fmtPct(v: number, withSign = false): string {
   136→  const sign = withSign && v > 0 ? "+" : "";
   137→  return `${sign}${Number(v).toFixed(2)}%`;
   138→}
   139→
   140→function fmtPkr(v: number): string {
   141→  return `Rs. ${Math.round(v).toLocaleString("en-PK")}`;

   299→  // Per-stock signals from finalConsensus — KEPT (used by the Signal Tracker

   326→
   327→  // ─── Issue 13 — Investment Simulator (redesigned) ──────────────────────
   328→  // Uses /api/backtest/investment-simulator which:
   329→  //  - Trades on ALL final-consensus signals (~284) from /api/signals/all
   330→  //    (not just the 8 hardcoded COVERED_SYMBOLS).
   331→  //  - Chooses stocks RANDOMLY (seeded shuffle for determinism).
   332→  //  - Trades on ALL stock data in the OhlcvBar DB (100 KSE-100 stocks).
   333→  //  - Each trade uses the final-ensemble-consensus signal (entry/SL/TP1/2/3).
   334→  //  - Holds for the user's time-period input (day/month/year → trading days).
   335→  //  - Optional monthly SIP adds to the total capital pool.
   336→  interface SimTrade {
   337→    tradeNo: number;
   338→    symbol: string;
   339→    side: "BUY" | "SELL";
   340→    direction: "LONG" | "SHORT";
   341→    entryDate: number;
   342→    entryPrice: number;
   343→    exitDate: number;
   344→    exitPrice: number;
   345→    exitReason: "TP1_HIT" | "TP2_HIT" | "TP3_HIT" | "SL_HIT" | "TIME_UP";
   346→    sl: number;
   347→    tp1: number;
   348→    tp2: number;
   349→    tp3: number;
   350→    shares: number;
   351→    notional: number;
   352→    netPnl: number;
   353→    netPnlPct: number;
   354→    isWin: boolean;
   355→  }
   356→  interface SimMetrics {
   357→    startingCapital: number;
   358→    amountPerTrade: number;
   359→    totalTrades: number;
   360→    numDays: number;
   361→    winTrades: number;
   362→    lossTrades: number;
   363→    winRate: number;
   364→    avgWinPct: number;
   365→    avgLossPct: number;
   366→    grossProfit: number;
   367→    grossLoss: number;
   368→    totalCosts: number;
   369→    grossBeforeCosts: number;
   370→    netPnL: number;
   371→    finalValue: number;
   372→    roiPct: number;
   373→    maxDrawdownPct: number;
   374→    profitFactor: number;
   375→    sharpe: number;
   376→    sortino: number;
   377→    equityCurve: { tradeNo: number; value: number }[];
   378→  }
   379→  interface SimConfig {
   380→    initialCapital: number;
   381→    periodValue: number;
   382→    periodUnit: PeriodUnit;
   383→    monthlySip: number;
   384→    numDays: number;
   385→    numMonths: number;
   386→    totalSipContribution: number;
   387→    totalCapital: number;
   388→    amountPerTrade: number;
   389→    numSignalsAvailable: number;
   390→    numTrades: number;
   391→    seed: number;
   392→  }
   393→  interface SimResponse {
   394→    config: SimConfig;
   395→    metrics: SimMetrics;
   396→    trades: SimTrade[];
   397→  }
   398→
   399→  const [simCapital, setSimCapital] = useState<string>("100000");
   400→  const [simPeriodValue, setSimPeriodValue] = useState<string>("3");
   401→  const [simPeriodUnit, setSimPeriodUnit] = useState<PeriodUnit>("month");
   402→  const [simMonthlySip, setSimMonthlySip] = useState<string>("0");
   403→  const [simResult, setSimResult] = useState<SimResponse | null>(null);
   404→  const [simLoading, setSimLoading] = useState(false);
   405→
   406→  const runSimulation = async () => {
   407→    const capital = parseFloat(simCapital.replace(/[,\s]/g, ""));
   408→    const periodValue = parseInt(simPeriodValue, 10);
   409→    const monthlySip = parseFloat(simMonthlySip.replace(/[,\s]/g, "")) || 0;
   410→    if (!Number.isFinite(capital) || capital <= 0) {
   411→      toast.error("Enter a valid capital amount");
   412→      return;
   413→    }
   414→    if (!Number.isFinite(periodValue) || periodValue <= 0) {
   415→      toast.error("Enter a valid time period");
   416→      return;
   417→    }
   418→    setSimLoading(true);
   419→    try {
   420→      const qs = `capital=${encodeURIComponent(simCapital)}&periodValue=${encodeURIComponent(simPeriodValue)}&periodUnit=${encodeURIComponent(simPeriodUnit)}&monthlySip=${encodeURIComponent(simMonthlySip)}`;
   421→      const r = await fetch(`/api/backtest/investment-simulator?${qs}`);
   422→      const j = await r.json();
   423→      if (j.error) {
   424→        toast.error(j.error);
   425→        return;
   426→      }
   427→      setSimResult(j as SimResponse);
   428→      toast.success(
   429→        `Simulated ${j.config.numTrades} trades over ${j.config.numDays} days — ROI ${j.metrics.roiPct}%`
   430→      );
   431→    } catch (e) {
   432→      toast.error("Simulation failed");
   433→    } finally {
   434→      setSimLoading(false);
   435→    }
   436→  };
   437→
   438→  const resetSimulation = () => {
   439→    setSimResult(null);
   440→  };
   441→
   442→  // Aggregate top-card stats — uses ensembleBacktesting.FINAL_CONSENSUS.
   443→  const signalStats = useMemo(() => {
   444→    const fc = ensStats?.fc;
   445→    if (!fc) return null;
   446→    const total = fc.totalTrades ?? 0;
   447→    const wins = fc.wins ?? Math.round((total * (fc.winRate || 0)) / 100);
   448→    const losses = total - wins;
   449→    const winRate = fc.winRate ?? 0;
   450→    const avgReturn = fc.avgReturnPct ?? 0;
   451→    const best = fc.maxReturnPct ?? 0;
   452→    const worst = fc.minReturnPct ?? 0;
   453→    return { total, wins, losses, winRate, avgReturn, best, worst };
   454→  }, [ensStats]);
   455→
   456→  // Portfolio tracker — only positions with buyPrice != null && shares > 0
   457→  // are tracked. Closed positions remain in the list and contribute to P&L.
   458→  // Issue 23: keep closed trades visible + reflect them in top card stats.
   459→  const livePriceMap = useMemo(() => {
   460→    const m: Record<string, number> = {};
   461→    for (const s of aiSignals) {
   462→      if (s.symbol && typeof s.currentPrice === "number") m[s.symbol] = s.currentPrice;
   463→    }
   464→    return m;
   465→  }, [aiSignals]);
   466→
   467→  const portfolioPositions = useMemo(
   468→    () => portfolio.filter((p) => p.buyPrice !== null && p.shares > 0),
   469→    [portfolio]
   470→  );
   471→
   472→  const portfolioStats = useMemo(() => {
   473→    let totalValue = 0;
   474→    let totalInvested = 0;
   475→    let totalPnl = 0;
   476→    let openCount = 0;
   477→    let hasLiveForOpen = false;
   478→    let hasAnyPnl = false;
   479→
   480→    // Open positions (from portfolio store, shares > 0)
   481→    for (const p of portfolioPositions) {
   482→      const buy = (p.buyPrice as number) || 0;
   483→      const invested = buy * p.shares;
   484→      totalInvested += invested;
   485→      openCount++;
   486→      const live = livePriceMap[p.symbol];
   487→      if (typeof live === "number") {
   488→        const val = live * p.shares;
   489→        totalValue += val;
   490→        totalPnl += val - invested;
   491→        hasLiveForOpen = true;
   492→        hasAnyPnl = true;
   493→      }
   494→    }
   495→
   496→    // Issue 18: closed trades from the close flow (closedTrades in store)
   497→    // add their net P&L to the totals. They have no remaining market value
   498→    // (shares were sold), but their net P&L contributes to total P&L.
   499→    let closedCount = 0;
   500→    let closedWins = 0;
   501→    let closedNetPnl = 0;
   502→    for (const t of closedTrades) {
   503→      closedCount++;
   504→      closedNetPnl += t.netPnl;
   505→      if (t.netPnl > 0) closedWins++;
   506→    }
   507→    totalPnl += closedNetPnl;
   508→    if (closedCount > 0) hasAnyPnl = true;
   509→
   510→    const roiPct = totalInvested > 0 ? (totalPnl / totalInvested) * 100 : 0;
   511→    const winRate = closedCount > 0 ? (closedWins / closedCount) * 100 : null;
   512→
   513→    return {
   514→      total: openCount + closedCount,
   515→      totalValue,
   516→      totalInvested,
   517→      totalPnl,
   518→      roiPct,
   519→      openCount,
   520→      closedCount,
   521→      closedWins,
   522→      winRate,
   523→      hasLiveForOpen,
   524→      hasAnyPnl,
   525→    };
   526→  }, [portfolioPositions, livePriceMap, closedTrades]);
   527→
   528→  // Issue 18: sector-wise holding distribution for the pie chart.
   529→  // For each open position: value = (livePrice || buyPrice) × shares.
   530→  // Sector is taken from position.sector (stored at add-time) or from the
   531→  // sectorMap fetched on mount, else falls back to "Other".
   532→  const sectorData = useMemo(() => {

  1236→// ─── Issue 18: SVG-based donut pie chart for sector distribution ──────────
  1237→const SECTOR_COLORS = [
  1238→  "#00E676", "#FBBF24", "#FF5C5C", "#22E881", "#4ADE80",
  1239→  "#94A3B8", "#F59E0B", "#10B981", "#EF4444", "#8B5CF6",
  1240→  "#3B82F6", "#EC4899", "#14B8A6", "#A855F7", "#6366F1",
  1241→];
  1242→
  1243→function SectorPieChart({ data }: { data: { sector: string; value: number; pct: number }[] }) {
  1244→  const size = 160;
  1245→  const cx = size / 2;
  1246→  const cy = size / 2;
  1247→  const r = 65;
  1248→  const innerR = 30; // donut hole
  1249→
  1250→  // SVG arc paths for each sector slice. Cumulative angle is computed
  1251→  // functionally from the preceding slices' percentages (no mutation) to
  1252→  // satisfy react-hooks/immutability.
  1253→  const slices = data.map((d, i) => {
  1254→    const cumulativeBeforePct = data.slice(0, i).reduce((s, x) => s + x.pct, 0);
  1255→    const startAngle = -Math.PI / 2 + (cumulativeBeforePct / 100) * 2 * Math.PI;
  1256→    const angleSpan = (d.pct / 100) * 2 * Math.PI;
  1257→    const endAngle = startAngle + angleSpan;
  1258→
  1259→    // Outer arc endpoints
  1260→    const x1 = cx + r * Math.cos(startAngle);
  1261→    const y1 = cy + r * Math.sin(startAngle);
  1262→    const x2 = cx + r * Math.cos(endAngle);
  1263→    const y2 = cy + r * Math.sin(endAngle);
  1264→    // Inner arc endpoints (donut)
  1265→    const xi1 = cx + innerR * Math.cos(startAngle);
  1266→    const yi1 = cy + innerR * Math.sin(startAngle);
  1267→    const xi2 = cx + innerR * Math.cos(endAngle);
  1268→    const yi2 = cy + innerR * Math.sin(endAngle);
  1269→
  1270→    const largeArc = angleSpan > Math.PI ? 1 : 0;
  1271→    // Donut slice path: outer arc forward + line in + inner arc backward + close
  1272→    const path = `M ${x1} ${y1} A ${r} ${r} 0 ${largeArc} 1 ${x2} ${y2} L ${xi2} ${yi2} A ${innerR} ${innerR} 0 ${largeArc} 0 ${xi1} ${yi1} Z`;
  1273→    return { path, color: SECTOR_COLORS[i % SECTOR_COLORS.length], ...d };
  1274→  });
  1275→
  1276→  return (
  1277→    <svg
  1278→      viewBox={`0 0 ${size} ${size}`}
  1279→      width={size}
  1280→      height={size}
  1281→      className="shrink-0"
  1282→      role="img"
  1283→      aria-label={`Sector distribution: ${data.map((d) => `${d.sector} ${d.pct.toFixed(0)}%`).join(", ")}`}
  1284→    >
  1285→      {slices.map((s, i) => (
  1286→        <path
  1287→          key={i}
  1288→          d={s.path}
  1289→          fill={s.color}
  1290→          stroke="#0A0E1A"
  1291→          strokeWidth={1.5}
  1292→        >
  1293→          <title>{`${s.sector}: ${s.pct.toFixed(1)}% (${fmtPkr(s.value)})`}</title>
  1294→        </path>
  1295→      ))}
  1296→    </svg>
  1297→  );
  1298→}
